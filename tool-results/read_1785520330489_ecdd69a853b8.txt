    34→async function api(method, urlPath, body = null, extraHeaders = {}) {
    35→  const opts = {
    36→    method,
    37→    headers: {
    38→      'Authorization': `Bearer ${TOKEN}`,
    39→      'Content-Type': 'application/json',
    40→      ...extraHeaders
    41→    }
    42→  };
    43→  if (body) opts.body = JSON.stringify(body);
    44→  try {
    45→    const res = await fetch(`${BASE}${urlPath}`, opts);
    46→    let data;
    47→    try { data = await res.json(); } catch (e) { data = null; }
    48→    return { status: res.status, data, ok: res.ok };
    49→  } catch (e) {
    50→    return { status: 0, data: { error: e.message }, ok: false };
    51→  }
    52→}

    58→function rec(id, desc, expected, actual, evidence = '') {
    59→  const pass = String(actual) === String(expected);
    60→  let ev = '';
    61→  if (typeof evidence === 'string') {
    62→    ev = evidence.length > 500 ? evidence.substring(0, 500) + '...' : evidence;
    63→  } else if (evidence != null && typeof evidence === 'object') {
    64→    ev = JSON.stringify(evidence);
    65→    if (ev.length > 500) ev = ev.substring(0, 500) + '...';
    66→  }
    67→  RESULTS.push({ id, desc, expected, actual, pass, evidence: ev || '' });
    68→  const icon = pass ? 'PASS' : 'FAIL';
    69→  console.log(`  [${icon}] ${id}: ${desc} | Expected: ${expected} | Got: ${actual}`);
