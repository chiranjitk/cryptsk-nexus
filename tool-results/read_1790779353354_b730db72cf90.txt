    40→// parseIPWithPrefix parses "10.10.10.1/24" into (net.IP, prefixLen, error).
    41→// Used for fallback handling when ip_types.ParseAddressWithPrefix is not viable.
    42→func parseIPWithPrefix(ipWithPrefix string) (net.IP, uint8, error) {
    43→        parts := strings.Split(ipWithPrefix, "/")
    44→        if len(parts) != 2 {
    45→                return nil, 0, fmt.Errorf("invalid IP/prefix: %s", ipWithPrefix)
    46→        }
    47→        ipAddr, err := parseIP(parts[0])
    48→        if err != nil {
    49→                return nil, 0, err
    50→        }
    51→        prefixLen, err := strconv.Atoi(parts[1])
    52→        if err != nil {
    53→                return nil, 0, fmt.Errorf("invalid prefix: %s", parts[1])
    54→        }
    55→        return ipAddr, uint8(prefixLen), nil
    56→}
    57→
    58→// parseIP parses a single IPv4/IPv6 address string.
    59→func parseIP(ipStr string) (net.IP, error) {
    60→        ipAddr := net.ParseIP(ipStr)
    61→        if ipAddr == nil {
    62→                return nil, fmt.Errorf("invalid IP: %s", ipStr)
    63→        }
    64→        return ipAddr, nil
    65→}

    95→func (c *VPPLiveClient) Connect() error {
    96→        c.mu.Lock()
    97→        defer c.mu.Unlock()
    98→
    99→        conn, err := govpp.Connect(VPPAPISock)
   100→        if err != nil {
   101→                return fmt.Errorf("failed to connect to VPP at %s: %w", VPPAPISock, err)
   102→        }
   103→
   104→        ch, err := conn.NewAPIChannel()
   105→        if err != nil {
   106→                conn.Disconnect()
   107→                return fmt.Errorf("failed to create API channel: %w", err)
   108→        }
   109→
   110→        // Set reply timeout to 5 seconds
   111→        ch.SetReplyTimeout(5 * time.Second)
   112→
   113→        c.conn = conn
   114→        c.ch = ch
   115→        c.connected = true
   116→        log.Printf("[vpp] Connected to VPP binary API at %s", VPPAPISock)
   117→        return nil
   118→}

   120→func (c *VPPLiveClient) Disconnect() {
   121→        c.mu.Lock()
   122→        defer c.mu.Unlock()
   123→        if c.ch != nil {
   124→                c.ch.Close()
   125→        }
   126→        if c.conn != nil {
   127→                c.conn.Disconnect()
   128→        }
   129→        c.connected = false
   130→        log.Println("[vpp] Disconnected from VPP binary API")
   131→}

   141→// GetInterfaceList returns all VPP interfaces via sw_interface_dump (binary API)
   142→func (c *VPPLiveClient) GetInterfaceList() ([]VPPInterface, error) {
   143→        if !c.IsConnected() {
   144→                return nil, fmt.Errorf("VPP not connected")
   145→        }
   146→
   147→        // sw_interface_dump is a multi-reply request
   148→        req := &interfaces.SwInterfaceDump{
   149→                SwIfIndex:       interface_types.InterfaceIndex(0xFFFFFFFF), // all interfaces
   150→                NameFilterValid: false,
   151→        }
   152→
   153→        multiCtx := c.ch.SendMultiRequest(req)
   154→
   155→        var ifaces []VPPInterface
   156→        for {
   157→                details := &interfaces.SwInterfaceDetails{}
   158→                lastReply, err := multiCtx.ReceiveReply(details)
   159→                if err != nil {
   160→                        return ifaces, fmt.Errorf("sw_interface_dump failed: %w", err)
   161→                }
   162→                if lastReply {
   163→                        break
   164→                }
   165→
   166→                // Build VPPInterface from details
   167→                iface := VPPInterface{
   168→                        Index:      uint32(details.SwIfIndex),
   169→                        Name:       string(details.InterfaceName[:]), // fixed-size array → string
   170→                        State:      "down",
   171→                        MacAddress: fmt.Sprintf("%02x:%02x:%02x:%02x:%02x:%02x",
   172→                                details.L2Address[0], details.L2Address[1], details.L2Address[2],
   173→                                details.L2Address[3], details.L2Address[4], details.L2Address[5]),
   174→                }
   175→                if details.Flags&interface_types.IF_STATUS_API_FLAG_ADMIN_UP != 0 {
   176→                        iface.State = "up"
   177→                }
   178→                ifaces = append(ifaces, iface)
   179→        }
   180→
   181→        log.Printf("[vpp] GetInterfaceList: %d interfaces", len(ifaces))
   182→        return ifaces, nil
   183→}

   185→// SetInterfaceState brings a VPP interface up or down via sw_interface_set_flags

   197→                SwIfIndex: interface_types.InterfaceIndex(swIfIndex),

   201→
   202→        err := c.ch.SendRequest(req).ReceiveReply(reply)
   203→        if err != nil {
   204→                return fmt.Errorf("sw_interface_set_flags failed: %w", err)
   205→        }

   207→        stateStr := "down"
   208→        if up {
   209→                stateStr = "up"
   210→        }
   211→        log.Printf("[vpp] Set interface %d state %s (via binary API)", swIfIndex, stateStr)
   212→        return nil

   293→// CreateACL creates an ACL via acl_add_replace. Returns aclIndex.
   294→// rules: list of ACLRule (SrcIP, DstIP, SrcPort, DstPort, Proto, Action).
   295→func (c *VPPLiveClient) CreateACL(name string, rules []ACLRule) (uint32, error) {
   296→        if !c.IsConnected() {
   297→                return 0, fmt.Errorf("VPP not connected")
   298→        }
   299→
   300→        // Convert ACLRule → acl_types.ACLRule
   301→        binRules := make([]acl_types.ACLRule, 0, len(rules))
   302→        for i, r := range rules {
   303→                binRule, err := convertACLRule(r)
   304→                if err != nil {
   305→                        return 0, fmt.Errorf("rule %d invalid: %w", i, err)
   306→                }
   307→                binRules = append(binRules, binRule)
   308→        }
   309→
   310→        // acl_add_replace with acl_index=0xFFFFFFFF → create new ACL
   311→        req := &acl.ACLAddReplace{
   312→                ACLIndex: 0xFFFFFFFF, // 0xFFFFFFFF = create new (auto-assign index)
   313→                Tag:      name,
   314→                R:        binRules,
   315→        }
   316→        reply := &acl.ACLAddReplaceReply{}
   317→
   318→        if err := c.ch.SendRequest(req).ReceiveReply(reply); err != nil {
   319→                return 0, fmt.Errorf("acl_add_replace failed: %w", err)
   320→        }
   321→        if reply.Retval != 0 {
   322→                return 0, fmt.Errorf("acl_add_replace returned non-zero retval=%d", reply.Retval)
   323→        }
   324→
   325→        log.Printf("[vpp] Created ACL %s (aclIndex=%d, rules=%d) via binary API", name, reply.ACLIndex, len(rules))
   326→        return reply.ACLIndex, nil
   327→}

   329→// convertACLRule converts an HTTP API ACLRule to a GoVPP acl_types.ACLRule.
   330→// Action: "permit" or "deny" (default "deny"). Ports can be "80" or "8080-9000".
   331→func convertACLRule(r ACLRule) (acl_types.ACLRule, error) {
   332→        // Action
   333→        var action acl_types.ACLAction
   334→        switch strings.ToLower(r.Action) {
   335→        case "permit", "allow":
   336→                action = acl_types.ACL_ACTION_API_PERMIT
   337→        case "permit_reflect":
   338→                action = acl_types.ACL_ACTION_API_PERMIT_REFLECT
   339→        case "deny", "drop", "block", "":
   340→                action = acl_types.ACL_ACTION_API_DENY
   341→        default:
   342→                return acl_types.ACLRule{}, fmt.Errorf("invalid action '%s'", r.Action)
   343→        }
   344→
   345→        // Src prefix
   346→        srcPrefix, err := parsePrefixOrDefault(r.SrcIP, "0.0.0.0/0")
   347→        if err != nil {
   348→                return acl_types.ACLRule{}, fmt.Errorf("invalid src IP: %w", err)
   349→        }
   350→        // Dst prefix
   351→        dstPrefix, err := parsePrefixOrDefault(r.DstIP, "0.0.0.0/0")
   352→        if err != nil {
   353→                return acl_types.ACLRule{}, fmt.Errorf("invalid dst IP: %w", err)
   354→        }
   355→
   356→        // Proto
   357→        var proto ip_types.IPProto = ip_types.IP_API_PROTO_RESERVED // 255 = any
   358→        switch strings.ToLower(r.Proto) {
   359→        case "tcp":
   360→                proto = ip_types.IP_API_PROTO_TCP
   361→        case "udp":
   362→                proto = ip_types.IP_API_PROTO_UDP
   363→        case "icmp":
   364→                proto = ip_types.IP_API_PROTO_ICMP
   365→        case "", "any", "ip":
   366→                // leave as RESERVED (any)
   367→        }
   368→
   369→        // Ports (parse "80" → first=last=80; "8080-9000" → first=8080,last=9000)
   370→        srcFirst, srcLast, _ := parsePortRange(r.SrcPort)
   371→        dstFirst, dstLast, _ := parsePortRange(r.DstPort)
   372→
   373→        return acl_types.ACLRule{
   374→                IsPermit:               action,
   375→                SrcPrefix:              srcPrefix,
   376→                DstPrefix:              dstPrefix,
   377→                Proto:                  proto,
   378→                SrcportOrIcmptypeFirst: srcFirst,
   379→                SrcportOrIcmptypeLast:  srcLast,
   380→                DstportOrIcmpcodeFirst: dstFirst,
   381→                DstportOrIcmpcodeLast:  dstLast,
   382→                TCPFlagsMask:           0,
   383→                TCPFlagsValue:          0,
   384→        }, nil
   385→}
   386→
   387→// parsePrefixOrDefault parses "10.0.0.0/24" or returns default if input is empty.
   388→func parsePrefixOrDefault(ipStr, defaultStr string) (ip_types.Prefix, error) {
   389→        if ipStr == "" {
   390→                ipStr = defaultStr
   391→        }
   392→        return ip_types.ParsePrefix(ipStr)
   393→}
   394→
   395→// parsePortRange parses "80" → (80, 80) or "8080-9000" → (8080, 9000).
   396→// Returns (0, 0, false) if input is empty.
   397→func parsePortRange(portStr string) (first, last uint16, ok bool) {
   398→        portStr = strings.TrimSpace(portStr)
   399→        if portStr == "" {
   400→                return 0, 0, false
   401→        }
   402→        if strings.Contains(portStr, "-") {
   403→                parts := strings.SplitN(portStr, "-", 2)
   404→                f, err1 := strconv.Atoi(strings.TrimSpace(parts[0]))
   405→                l, err2 := strconv.Atoi(strings.TrimSpace(parts[1]))
   406→                if err1 != nil || err2 != nil || f < 0 || f > 65535 || l < 0 || l > 65535 || f > l {
   407→                        return 0, 0, false
   408→                }
   409→                return uint16(f), uint16(l), true
   410→        }
   411→        p, err := strconv.Atoi(portStr)
   412→        if err != nil || p < 0 || p > 65535 {
   413→                return 0, 0, false
   414→        }
   415→        return uint16(p), uint16(p), true
   416→}

   443→// CreatePolicer creates a VPP policer via policer_add_del.
   444→// cirBps/eirBps are in bits/sec; we convert to kbps (rate_type=KBPS).
   445→// Returns policerIndex.
   446→//
   447→// NOTE on 1R2C validation: VPP rejects 1R2C policers with Eb > 0
   448→// ("Policer parameter validation failed -- 1R2C / Unable to compute hw param").
   449→// For 1R2C: Eir=0, Eb=0 (no excess bucket). Use 1R3C_RFC_2697 if you need Eb > 0.
   450→func (c *VPPLiveClient) CreatePolicer(name string, cirBps, eirBps uint64) (uint32, error) {
   451→        if !c.IsConnected() {
   452→                return 0, fmt.Errorf("VPP not connected")
   453→        }
   454→
   455→        // Convert bits/sec → kbps (rate_type = SSE2_QOS_RATE_API_KBPS)
   456→        cirKbps := uint32(cirBps / 1000)
   457→        if cirKbps == 0 {
   458→                cirKbps = 1 // avoid zero rate
   459→        }
   460→
   461→        // For 1R2C: only committed bucket matters. Cb = 1 second of cir in bytes.
   462→        // (Cb is in bytes for KBPS rate type per VPP convention)
   463→        cb := uint64(cirKbps) * 1000 // 1s burst in bytes
   464→        if cb == 0 {
   465→                cb = 4000 // fallback: 4KB burst
   466→        }
   467→
   468→        req := &policer.PolicerAddDel{
   469→                IsAdd:      true,
   470→                Name:       name,
   471→                Cir:        cirKbps,
   472→                Eir:        0, // 1R2C: no excess rate
   473→                Cb:         cb,
   474→                Eb:         0, // 1R2C: no excess burst (VPP validation requires this)
   475→                RateType:   policer_types.SSE2_QOS_RATE_API_KBPS,
   476→                RoundType:  policer_types.SSE2_QOS_ROUND_API_TO_CLOSEST,
   477→                Type:       policer_types.SSE2_QOS_POLICER_TYPE_API_1R2C,
   478→                ColorAware: false,
   479→                ConformAction: policer_types.Sse2QosAction{
   480→                        Type: policer_types.SSE2_QOS_ACTION_API_TRANSMIT,
   481→                },
   482→                ExceedAction: policer_types.Sse2QosAction{
   483→                        Type: policer_types.SSE2_QOS_ACTION_API_DROP,
   484→                },
   485→                ViolateAction: policer_types.Sse2QosAction{
   486→                        Type: policer_types.SSE2_QOS_ACTION_API_DROP,
   487→                },
   488→        }
   489→        reply := &policer.PolicerAddDelReply{}
   490→
   491→        if err := c.ch.SendRequest(req).ReceiveReply(reply); err != nil {
   492→                return 0, fmt.Errorf("policer_add_del failed: %w", err)
   493→        }
   494→        if reply.PolicerIndex == 0xFFFFFFFF {
   495→                return 0, fmt.Errorf("policer_add_del returned invalid index 0xFFFFFFFF (likely failure)")
   496→        }
   497→        log.Printf("[vpp] Created policer %s (index=%d, cir=%d kbps) via binapi",
   498→                name, reply.PolicerIndex, cirKbps)
   499→        return reply.PolicerIndex, nil
   500→}

   502→// ApplyPolicerToInterface binds a policer to an interface. VPP uses classify tables
   503→// for this: create a classify table, add a session that fires the policer, then
   504→// bind the table to the interface via policer_classify_set_interface.
   505→//
   506→// Best-effort: VPP's full policer-on-interface requires ClassifyAddDelTable +
   507→// ClassifyAddDelSession + PolicerClassifySetInterface. Here we create a per-interface
   508→// classify table and bind it. The session must reference the policer by node index,
   509→// which differs per VPP build — for now we only do the table + interface bind and
   510→// log if the policer is not actually attached.
   511→func (c *VPPLiveClient) ApplyPolicerToInterface(swIfIndex uint32, policerIndex uint32) error {
   512→        if !c.IsConnected() {
   513→                return fmt.Errorf("VPP not connected")
   514→        }
   515→
   516→        // Step 1: create a classify table for IP4 input
   517→        tableReq := &classify.ClassifyAddDelTable{
   518→                IsAdd:         true,
   519→                TableIndex:    0xFFFFFFFF, // auto-assign
   520→                Nbuckets:      2,
   521→                MemorySize:    1048576, // 1MB
   522→                SkipNVectors:  0,
   523→                MatchNVectors: 1, // match on first 16 bytes (IP4 src+dst+proto+ports)
   524→                NextTableIndex: 0xFFFFFFFF,
   525→                MissNextIndex: 0xFFFFFFFF,
   526→                MaskLen:        16,
   527→                Mask: []byte{
   528→                        0xFF, 0xFF, 0xFF, 0xFF, // IP src
   529→                        0xFF, 0xFF, 0xFF, 0xFF, // IP dst
   530→                        0xFF, 0x00, 0x00, 0x00, // proto + 3 bytes pad
   531→                        0xFF, 0xFF, 0xFF, 0xFF, // ports
   532→                },
   533→        }
   534→        tableReply := &classify.ClassifyAddDelTableReply{}
   535→        if err := c.ch.SendRequest(tableReq).ReceiveReply(tableReply); err != nil {
   536→                return fmt.Errorf("classify_add_del_table failed: %w", err)
   537→        }
   538→        if tableReply.Retval != 0 {
   539→                return fmt.Errorf("classify_add_del_table retval=%d", tableReply.Retval)
   540→        }
   541→        tableIndex := tableReply.NewTableIndex
   542→
   543→        // Step 2: bind the classify table to the interface as the IP4 policer-classify table
   544→        bindReq := &classify.PolicerClassifySetInterface{
   545→                SwIfIndex:     interface_types.InterfaceIndex(swIfIndex),
   546→                IP4TableIndex: tableIndex,
   547→                IP6TableIndex: 0xFFFFFFFF, // not used
   548→                L2TableIndex:  0xFFFFFFFF, // not used
   549→                IsAdd:         true,
   550→        }
   551→        bindReply := &classify.PolicerClassifySetInterfaceReply{}
   552→        if err := c.ch.SendRequest(bindReq).ReceiveReply(bindReply); err != nil {
   553→                return fmt.Errorf("policer_classify_set_interface failed: %w", err)
   554→        }
   555→        if bindReply.Retval != 0 {
   556→                return fmt.Errorf("policer_classify_set_interface retval=%d", bindReply.Retval)
   557→        }
   558→
   559→        log.Printf("[vpp] Applied policer %d to interface %d via classify table %d (best-effort) via binapi",
   560→                policerIndex, swIfIndex, tableIndex)
   561→        // NOTE: this is a best-effort binding. Full per-subscriber policer attachment
   562→        // requires a ClassifyAddDelSession with HitNextIndex pointing to the policer
   563→        // graph node, which is environment-specific. The current implementation sets
   564→        // up the table+interface but does NOT inject a matching session.
   565→        return nil
   566→}

   813→// ChangeSubscriberBandwidth updates the policer for a subscriber.
   814→// Best-effort: deletes the existing policer (by name) and creates a new one
   815→// with the new rates. The policer name convention is "pol-<subscriberIP>".
   816→//
   817→// NOTE: This does NOT re-bind to the interface — see ApplyPolicerToInterface for
   818→// the limitation of the current binding model.
   819→func (c *VPPLiveClient) ChangeSubscriberBandwidth(subscriberIP string, newDownloadKbps, newUploadKbps uint64) error {
   820→        if !c.IsConnected() {
   821→                return fmt.Errorf("VPP not connected")
   822→        }
   823→
   824→        policerName := fmt.Sprintf("pol-%s", subscriberIP)
   825→        policerName = sanitizePolicerName(policerName)
   826→
   827→        // Look up existing entry (if any) so we can preserve external IP for re-binding
   828→        var existing *subscriberPolicerEntry
   829→        if v, ok := c.subscriberPolicers.Load(subscriberIP); ok {
   830→                existing = v.(*subscriberPolicerEntry)
   831→                // Best-effort delete the old policer (ignore error — may not exist)
   832→                _ = c.DeletePolicer(existing.PolicerName)
   833→        }
   834→
   835→        // Convert kbps → bps (CreatePolicer expects bps)
   836→        dlBps := newDownloadKbps * 1000
   837→        ulBps := newUploadKbps * 1000
   838→        // Use the larger of upload/download as CIR (VPP policers are single-direction per flow).
   839→        // For symmetric policers, EIR = the other direction.
   840→        cirBps := dlBps
   841→        eirBps := ulBps
   842→        if ulBps > dlBps {
   843→                cirBps = ulBps
   844→                eirBps = dlBps
   845→        }
   846→
   847→        policerIdx, err := c.CreatePolicer(policerName, cirBps, eirBps)
   848→        if err != nil {
   849→                return fmt.Errorf("CreatePolicer for %s failed: %w", policerName, err)
   850→        }
   851→
   852→        // Store in-memory state
   853→        entry := &subscriberPolicerEntry{
   854→                SubscriberIP:  subscriberIP,
   855→                PolicerName:   policerName,
   856→                PolicerIndex:  policerIdx,
   857→                DownloadKbps:  newDownloadKbps,
   858→                UploadKbps:    newUploadKbps,
   859→                CreatedAt:     time.Now(),
   860→        }
   861→        if existing != nil {
   862→                entry.ExternalIP = existing.ExternalIP
   863→        }
   864→        c.subscriberPolicers.Store(subscriberIP, entry)
   865→
   866→        log.Printf("[vpp] ChangeSubscriberBandwidth(ip=%s, dl=%d kbps, ul=%d kbps) → policerIndex=%d via binapi",
   867→                subscriberIP, newDownloadKbps, newUploadKbps, policerIdx)
   868→        return nil
   869→}

   871→// DisconnectSubscriber tears down all dataplane state for a subscriber:
   872→// deletes the policer and the static NAT mapping.
   873→func (c *VPPLiveClient) DisconnectSubscriber(subscriberIP string) error {
   874→        if !c.IsConnected() {
   875→                return fmt.Errorf("VPP not connected")
   876→        }
   877→
   878→        var errs []string
   879→
   880→        // 1. Delete the policer (if we tracked one)
   881→        if v, ok := c.subscriberPolicers.Load(subscriberIP); ok {
   882→                entry := v.(*subscriberPolicerEntry)
   883→                if err := c.DeletePolicer(entry.PolicerName); err != nil {
   884→                        errs = append(errs, fmt.Sprintf("delete_policer(%s): %v", entry.PolicerName, err))
   885→                }
   886→                // 2. Delete the static NAT mapping (if we tracked an external IP)
   887→                if entry.ExternalIP != "" {
   888→                        if err := c.DeleteStaticNat(subscriberIP, entry.ExternalIP); err != nil {
   889→                                errs = append(errs, fmt.Sprintf("delete_static_nat(%s->%s): %v", subscriberIP, entry.ExternalIP, err))
   890→                        }
   891→                }
   892→                c.subscriberPolicers.Delete(subscriberIP)
   893→        }
   894→
   895→        if len(errs) > 0 {
   896→                return fmt.Errorf("DisconnectSubscriber partial failure: %s", strings.Join(errs, "; "))
   897→        }
   898→        log.Printf("[vpp] Disconnected subscriber %s (policer + NAT removed) via binapi", subscriberIP)
   899→        return nil
   900→}

   902→// sanitizePolicerName ensures the name is ≤64 chars and contains only safe bytes
   903→// (VPP policer name field is a fixed 64-byte string).
   904→func sanitizePolicerName(name string) string {
   905→        // Replace dots/dashes with underscores for safety
   906→        out := strings.Map(func(r rune) rune {
   907→                if r >= 'a' && r <= 'z' {
   908→                        return r
   909→                }
   910→                if r >= 'A' && r <= 'Z' {
   911→                        return r
   912→                }
   913→                if r >= '0' && r <= '9' {
   914→                        return r
   915→                }
   916→                return '_'
   917→        }, name)
   918→        if len(out) > 60 {
   919→                out = out[:60]
   920→        }
   921→        return out
   922→}
