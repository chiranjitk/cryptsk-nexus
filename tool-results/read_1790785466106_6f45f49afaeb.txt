    40→    40→// parseIPWithPrefix parses "10.10.10.1/24" into (net.IP, prefixLen, error).
    41→    41→// Used for fallback handling when ip_types.ParseAddressWithPrefix is not viable.
    42→    42→func parseIPWithPrefix(ipWithPrefix string) (net.IP, uint8, error) {
    43→    43→        parts := strings.Split(ipWithPrefix, "/")
    44→    44→        if len(parts) != 2 {
    45→    45→                return nil, 0, fmt.Errorf("invalid IP/prefix: %s", ipWithPrefix)
    46→    46→        }
    47→    47→        ipAddr, err := parseIP(parts[0])
    48→    48→        if err != nil {
    49→    49→                return nil, 0, err
    50→    50→        }
    51→    51→        prefixLen, err := strconv.Atoi(parts[1])
    52→    52→        if err != nil {
    53→    53→                return nil, 0, fmt.Errorf("invalid prefix: %s", parts[1])
    54→    54→        }
    55→    55→        return ipAddr, uint8(prefixLen), nil
    56→    56→}
    57→    57→
    58→    58→// parseIP parses a single IPv4/IPv6 address string.
    59→    59→func parseIP(ipStr string) (net.IP, error) {
    60→    60→        ipAddr := net.ParseIP(ipStr)
    61→    61→        if ipAddr == nil {
    62→    62→                return nil, fmt.Errorf("invalid IP: %s", ipStr)
    63→    63→        }
    64→    64→        return ipAddr, nil
    65→    65→}

    95→    95→func (c *VPPLiveClient) Connect() error {
    96→    96→        c.mu.Lock()
    97→    97→        defer c.mu.Unlock()
    98→    98→
    99→    99→        conn, err := govpp.Connect(VPPAPISock)
   100→   100→        if err != nil {
   101→   101→                return fmt.Errorf("failed to connect to VPP at %s: %w", VPPAPISock, err)
   102→   102→        }
   103→   103→
   104→   104→        ch, err := conn.NewAPIChannel()
   105→   105→        if err != nil {
   106→   106→                conn.Disconnect()
   107→   107→                return fmt.Errorf("failed to create API channel: %w", err)
   108→   108→        }
   109→   109→
   110→   110→        // Set reply timeout to 5 seconds
   111→   111→        ch.SetReplyTimeout(5 * time.Second)
   112→   112→
   113→   113→        c.conn = conn
   114→   114→        c.ch = ch
   115→   115→        c.connected = true
   116→   116→        log.Printf("[vpp] Connected to VPP binary API at %s", VPPAPISock)
   117→   117→        return nil
   118→   118→}

   141→   141→// GetInterfaceList returns all VPP interfaces via sw_interface_dump (binary API)
   142→   142→func (c *VPPLiveClient) GetInterfaceList() ([]VPPInterface, error) {
   143→   143→        if !c.IsConnected() {
   144→   144→                return nil, fmt.Errorf("VPP not connected")
   145→   145→        }
   146→   146→
   147→   147→        // sw_interface_dump is a multi-reply request
   148→   148→        req := &interfaces.SwInterfaceDump{
   149→   149→                SwIfIndex:       interface_types.InterfaceIndex(0xFFFFFFFF), // all interfaces
   150→   150→                NameFilterValid: false,
   151→   151→        }
   152→   152→
   153→   153→        multiCtx := c.ch.SendMultiRequest(req)
   154→   154→
   155→   155→        var ifaces []VPPInterface
   156→   156→        for {
   157→   157→                details := &interfaces.SwInterfaceDetails{}
   158→   158→                lastReply, err := multiCtx.ReceiveReply(details)
   159→   159→                if err != nil {
   160→   160→                        return ifaces, fmt.Errorf("sw_interface_dump failed: %w", err)
   161→   161→                }
   162→   162→                if lastReply {
   163→   163→                        break
   164→   164→                }
   165→   165→
   166→   166→                // Build VPPInterface from details
   167→   167→                iface := VPPInterface{
   168→   168→                        Index:      uint32(details.SwIfIndex),
   169→   169→                        Name:       string(details.InterfaceName[:]), // fixed-size array → string
   170→   170→                        State:      "down",
   171→   171→                        MacAddress: fmt.Sprintf("%02x:%02x:%02x:%02x:%02x:%02x",
   172→   172→                                details.L2Address[0], details.L2Address[1], details.L2Address[2],
   173→   173→                                details.L2Address[3], details.L2Address[4], details.L2Address[5]),
   174→   174→                }
   175→   175→                if details.Flags&interface_types.IF_STATUS_API_FLAG_ADMIN_UP != 0 {
   176→   176→                        iface.State = "up"
   177→   177→                }
   178→   178→                ifaces = append(ifaces, iface)
   179→   179→        }
   180→   180→
   181→   181→        log.Printf("[vpp] GetInterfaceList: %d interfaces", len(ifaces))
   182→   182→        return ifaces, nil
   183→   183→}

   293→   293→// CreateACL creates an ACL via acl_add_replace. Returns aclIndex.
   294→   294→// rules: list of ACLRule (SrcIP, DstIP, SrcPort, DstPort, Proto, Action).
   295→   295→func (c *VPPLiveClient) CreateACL(name string, rules []ACLRule) (uint32, error) {
   296→   296→        if !c.IsConnected() {
   297→   297→                return 0, fmt.Errorf("VPP not connected")
   298→   298→        }
   299→   299→
   300→   300→        // Convert ACLRule → acl_types.ACLRule
   301→   301→        binRules := make([]acl_types.ACLRule, 0, len(rules))
   302→   302→        for i, r := range rules {
   303→   303→                binRule, err := convertACLRule(r)
   304→   304→                if err != nil {
   305→   305→                        return 0, fmt.Errorf("rule %d invalid: %w", i, err)
   306→   306→                }
   307→   307→                binRules = append(binRules, binRule)
   308→   308→        }
   309→   309→
   310→   310→        // acl_add_replace with acl_index=0xFFFFFFFF → create new ACL
   311→   311→        req := &acl.ACLAddReplace{
   312→   312→                ACLIndex: 0xFFFFFFFF, // 0xFFFFFFFF = create new (auto-assign index)
   313→   313→                Tag:      name,
   314→   314→                R:        binRules,
   315→   315→        }
   316→   316→        reply := &acl.ACLAddReplaceReply{}
   317→   317→
   318→   318→        if err := c.ch.SendRequest(req).ReceiveReply(reply); err != nil {
   319→   319→                return 0, fmt.Errorf("acl_add_replace failed: %w", err)
   320→   320→        }
   321→   321→        if reply.Retval != 0 {
   322→   322→                return 0, fmt.Errorf("acl_add_replace returned non-zero retval=%d", reply.Retval)
   323→   323→        }
   324→   324→
   325→   325→        log.Printf("[vpp] Created ACL %s (aclIndex=%d, rules=%d) via binary API", name, reply.ACLIndex, len(rules))
   326→   326→        return reply.ACLIndex, nil
   327→   327→}

   329→   329→// convertACLRule converts an HTTP API ACLRule to a GoVPP acl_types.ACLRule.
   330→   330→// Action: "permit" or "deny" (default "deny"). Ports can be "80" or "8080-9000".
   331→   331→func convertACLRule(r ACLRule) (acl_types.ACLRule, error) {
   332→   332→        // Action
   333→   333→        var action acl_types.ACLAction
   334→   334→        switch strings.ToLower(r.Action) {
   335→   335→        case "permit", "allow":
   336→   336→                action = acl_types.ACL_ACTION_API_PERMIT
   337→   337→        case "permit_reflect":
   338→   338→                action = acl_types.ACL_ACTION_API_PERMIT_REFLECT
   339→   339→        case "deny", "drop", "block", "":
   340→   340→                action = acl_types.ACL_ACTION_API_DENY
   341→   341→        default:
   342→   342→                return acl_types.ACLRule{}, fmt.Errorf("invalid action '%s'", r.Action)
   343→   343→        }
   344→   344→
   345→   345→        // Src prefix
   346→   346→        srcPrefix, err := parsePrefixOrDefault(r.SrcIP, "0.0.0.0/0")
   347→   347→        if err != nil {
   348→   348→                return acl_types.ACLRule{}, fmt.Errorf("invalid src IP: %w", err)
   349→   349→        }
   350→   350→        // Dst prefix
   351→   351→        dstPrefix, err := parsePrefixOrDefault(r.DstIP, "0.0.0.0/0")
   352→   352→        if err != nil {
   353→   353→                return acl_types.ACLRule{}, fmt.Errorf("invalid dst IP: %w", err)
   354→   354→        }
   355→   355→
   356→   356→        // Proto
   357→   357→        var proto ip_types.IPProto = ip_types.IP_API_PROTO_RESERVED // 255 = any
   358→   358→        switch strings.ToLower(r.Proto) {
   359→   359→        case "tcp":
   360→   360→                proto = ip_types.IP_API_PROTO_TCP
   361→   361→        case "udp":
   362→   362→                proto = ip_types.IP_API_PROTO_UDP
   363→   363→        case "icmp":
   364→   364→                proto = ip_types.IP_API_PROTO_ICMP
   365→   365→        case "", "any", "ip":
   366→   366→                // leave as RESERVED (any)
   367→   367→        }
   368→   368→
   369→   369→        // Ports (parse "80" → first=last=80; "8080-9000" → first=8080,last=9000)
   370→   370→        srcFirst, srcLast, _ := parsePortRange(r.SrcPort)
   371→   371→        dstFirst, dstLast, _ := parsePortRange(r.DstPort)
   372→   372→
   373→   373→        return acl_types.ACLRule{
   374→   374→                IsPermit:               action,
   375→   375→                SrcPrefix:              srcPrefix,
   376→   376→                DstPrefix:              dstPrefix,
   377→   377→                Proto:                  proto,
   378→   378→                SrcportOrIcmptypeFirst: srcFirst,
   379→   379→                SrcportOrIcmptypeLast:  srcLast,
   380→   380→                DstportOrIcmpcodeFirst: dstFirst,
   381→   381→                DstportOrIcmpcodeLast:  dstLast,
   382→   382→                TCPFlagsMask:           0,
   383→   383→                TCPFlagsValue:          0,
   384→   384→        }, nil
   385→   385→}

   387→   387→// parsePrefixOrDefault parses "10.0.0.0/24" or returns default if input is empty.
   388→   388→func parsePrefixOrDefault(ipStr, defaultStr string) (ip_types.Prefix, error) {
   389→   389→        if ipStr == "" {
   390→   390→                ipStr = defaultStr
   391→   391→        }
   392→   392→        return ip_types.ParsePrefix(ipStr)
   393→   393→}

   395→   395→// parsePortRange parses "80" → (80, 80) or "8080-9000" → (8080, 9000).
   396→   396→// Returns (0, 0, false) if input is empty.
   397→   397→func parsePortRange(portStr string) (first, last uint16, ok bool) {
   398→   398→        portStr = strings.TrimSpace(portStr)
   399→   399→        if portStr == "" {
   400→   400→                return 0, 0, false
   401→   401→        }
   402→   402→        if strings.Contains(portStr, "-") {
   403→   403→                parts := strings.SplitN(portStr, "-", 2)
   404→   404→                f, err1 := strconv.Atoi(strings.TrimSpace(parts[0]))
   405→   405→                l, err2 := strconv.Atoi(strings.TrimSpace(parts[1]))
   406→   406→                if err1 != nil || err2 != nil || f < 0 || f > 65535 || l < 0 || l > 65535 || f > l {
   407→   407→                        return 0, 0, false
   408→   408→                }
   409→   409→                return uint16(f), uint16(l), true
   410→   410→        }
   411→   411→        p, err := strconv.Atoi(portStr)
   412→   412→        if err != nil || p < 0 || p > 65535 {
   413→   413→                return 0, 0, false
   414→   414→        }
   415→   415→        return uint16(p), uint16(p), true
   416→   416→}

   443→   443→// CreatePolicer creates a VPP policer via policer_add_del.
   444→   444→// cirBps/eirBps are in bits/sec; we convert to kbps (rate_type=KBPS).
   445→   445→// Returns policerIndex.
   446→   446→//
   447→   447→// NOTE on 1R2C validation: VPP rejects 1R2C policers with Eb > 0
   448→   448→// ("Policer parameter validation failed -- 1R2C / Unable to compute hw param").
   449→   449→// For 1R2C: Eir=0, Eb=0 (no excess bucket). Use 1R3C_RFC_2697 if you need Eb > 0.
   450→   450→func (c *VPPLiveClient) CreatePolicer(name string, cirBps, eirBps uint64) (uint32, error) {
   451→   451→        if !c.IsConnected() {
   452→   452→                return 0, fmt.Errorf("VPP not connected")
   453→   453→        }
   454→   454→
   455→   455→        // Convert bits/sec → kbps (rate_type = SSE2_QOS_RATE_API_KBPS)
   456→   456→        cirKbps := uint32(cirBps / 1000)
   457→   457→        if cirKbps == 0 {
   458→   458→                cirKbps = 1 // avoid zero rate
   459→   459→        }
   460→   460→
   461→   461→        // For 1R2C: only committed bucket matters. Cb = 1 second of cir in bytes.
   462→   462→        // (Cb is in bytes for KBPS rate type per VPP convention)
   463→   463→        cb := uint64(cirKbps) * 1000 // 1s burst in bytes
   464→   464→        if cb == 0 {
   465→   465→                cb = 4000 // fallback: 4KB burst
   466→   466→        }
   467→   467→
   468→   468→        req := &policer.PolicerAddDel{
   469→   469→                IsAdd:      true,
   470→   470→                Name:       name,
   471→   471→                Cir:        cirKbps,
   472→   472→                Eir:        0, // 1R2C: no excess rate
   473→   473→                Cb:         cb,
   474→   474→                Eb:         0, // 1R2C: no excess burst (VPP validation requires this)
   475→   475→                RateType:   policer_types.SSE2_QOS_RATE_API_KBPS,
   476→   476→                RoundType:  policer_types.SSE2_QOS_ROUND_API_TO_CLOSEST,
   477→   477→                Type:       policer_types.SSE2_QOS_POLICER_TYPE_API_1R2C,
   478→   478→                ColorAware: false,
   479→   479→                ConformAction: policer_types.Sse2QosAction{
   480→   480→                        Type: policer_types.SSE2_QOS_ACTION_API_TRANSMIT,
   481→   481→                },
   482→   482→                ExceedAction: policer_types.Sse2QosAction{
   483→   483→                        Type: policer_types.SSE2_QOS_ACTION_API_DROP,
   484→   484→                },
   485→   485→                ViolateAction: policer_types.Sse2QosAction{
   486→   486→                        Type: policer_types.SSE2_QOS_ACTION_API_DROP,
   487→   487→                },
   488→   488→        }
   489→   489→        reply := &policer.PolicerAddDelReply{}
   490→   490→
   491→   491→        if err := c.ch.SendRequest(req).ReceiveReply(reply); err != nil {
   492→   492→                return 0, fmt.Errorf("policer_add_del failed: %w", err)
   493→   493→        }
   494→   494→        if reply.PolicerIndex == 0xFFFFFFFF {
   495→   495→                return 0, fmt.Errorf("policer_add_del returned invalid index 0xFFFFFFFF (likely failure)")
   496→   496→        }
   497→   497→        log.Printf("[vpp] Created policer %s (index=%d, cir=%d kbps) via binapi",
   498→   498→                name, reply.PolicerIndex, cirKbps)
   499→   499→        return reply.PolicerIndex, nil
   500→   500→}

   502→   502→// ApplyPolicerToInterface binds a policer to an interface so that every
   503→   503→// packet on that interface is policed. VPP uses classify tables for this:
   504→   504→//   1. Create a classify table whose Mask is all-zero (matches everything
   505→   505→//      in the masked-key space → every packet produces the same key, 0).
   506→   506→//   2. Add a ClassifyAddDelSession with HitNextIndex = policerIndex. When
   507→   507→//      a packet matches (which it always will, because the masked key is 0
   508→   508→//      and the session Match is 0), VPP redirects to the policer_classify
   509→   509→//      graph node and applies the policer at that index.
   510→   510→//   3. Bind the classify table to the interface via policer_classify_set_interface
   511→   511→//      so the IP4 input feature invokes the table for every IP4 packet.
   512→   512→//
   513→   513→// If any step fails the function returns an error — callers should treat it
   514→   514→// as non-fatal (the policer still exists in VPP's pool, it just isn't
   515→   515→// attached to a traffic flow).
   516→   516→func (c *VPPLiveClient) ApplyPolicerToInterface(swIfIndex uint32, policerIndex uint32) error {
   517→   517→        if !c.IsConnected() {
   518→   518→                return fmt.Errorf("VPP not connected")
   519→   519→        }
   520→   520→
   521→   521→        // Step 1: create a wildcard classify table for IP4 input.
   522→   522→        // Mask = all zeros (16 bytes) → masked-key is 0 for every packet →
   523→   523→        // any session with Match=zeros hits every packet.
   524→   524→        // MatchNVectors=1 → examine first 16 bytes (IP4 src + dst + proto + ports).
   525→   525→        tableReq := &classify.ClassifyAddDelTable{
   526→   526→                IsAdd:         true,
   527→   527→                TableIndex:    0xFFFFFFFF, // auto-assign
   528→   528→                Nbuckets:      2,
   529→   529→                MemorySize:    1048576, // 1MB
   530→   530→                SkipNVectors:  0,
   531→   531→                MatchNVectors: 1, // match on first 16 bytes (IP4 src+dst+proto+ports)
   532→   532→                NextTableIndex: 0xFFFFFFFF,
   533→   533→                MissNextIndex:  0xFFFFFFFF,
   534→   534→                MaskLen:        16,
   535→   535→                Mask: []byte{
   536→   536→                        0x00, 0x00, 0x00, 0x00,
   537→   537→                        0x00, 0x00, 0x00, 0x00,
   538→   538→                        0x00, 0x00, 0x00, 0x00,
   539→   539→                        0x00, 0x00, 0x00, 0x00,
   540→   540→                },
   541→   541→        }
   542→   542→        tableReply := &classify.ClassifyAddDelTableReply{}
   543→   543→        if err := c.ch.SendRequest(tableReq).ReceiveReply(tableReply); err != nil {
   544→   544→                return fmt.Errorf("classify_add_del_table failed: %w", err)
   545→   545→        }
   546→   546→        if tableReply.Retval != 0 {
   547→   547→                return fmt.Errorf("classify_add_del_table retval=%d", tableReply.Retval)
   548→   548→        }
   549→   549→        tableIndex := tableReply.NewTableIndex
   550→   550→
   551→   551→        // Step 2: add a wildcard classify session that hits the policer.
   552→   552→        // HitNextIndex = policerIndex: VPP's policer_classify graph node uses
   553→   553→        // this to look up the actual policer in the pool and apply it.
   554→   554→        // Match = all zeros → matches every packet given the all-zero table mask.
   555→   555→        // Action = SET_METADATA (lowest-impact action; the metadata field is
   556→   556→        // unused here — the policer itself is the real action).
   557→   557→        sessionReq := &classify.ClassifyAddDelSession{
   558→   558→                IsAdd:        true,
   559→   559→                TableIndex:   tableIndex,
   560→   560→                HitNextIndex: policerIndex, // <-- the key wiring to the policer
   561→   561→                OpaqueIndex:  0xFFFFFFFF,  // ~0 = unused
   562→   562→                Advance:      0,
   563→   563→                Action:       classify.CLASSIFY_API_ACTION_SET_METADATA,
   564→   564→                Metadata:     0,
   565→   565→                MatchLen:     16,
   566→   566→                Match: []byte{
   567→   567→                        0x00, 0x00, 0x00, 0x00,
   568→   568→                        0x00, 0x00, 0x00, 0x00,
   569→   569→                        0x00, 0x00, 0x00, 0x00,
   570→   570→                        0x00, 0x00, 0x00, 0x00,
   571→   571→                },
   572→   572→        }
   573→   573→        sessionReply := &classify.ClassifyAddDelSessionReply{}
   574→   574→        if err := c.ch.SendRequest(sessionReq).ReceiveReply(sessionReply); err != nil {
   575→   575→                return fmt.Errorf("classify_add_del_session failed: %w", err)
   576→   576→        }
   577→   577→        if sessionReply.Retval != 0 {
   578→   578→                return fmt.Errorf("classify_add_del_session retval=%d", sessionReply.Retval)
   579→   579→        }
   580→   580→
   581→   581→        // Step 3: bind the classify table to the interface as the IP4 policer-classify
   582→   582→        // table. This makes the IP4 input feature invoke our table for every IP4
   583→   583→        // packet on this interface.
   584→   584→        bindReq := &classify.PolicerClassifySetInterface{
   585→   585→                SwIfIndex:     interface_types.InterfaceIndex(swIfIndex),
   586→   586→                IP4TableIndex: tableIndex,
   587→   587→                IP6TableIndex: 0xFFFFFFFF, // not used
   588→   588→                L2TableIndex:  0xFFFFFFFF, // not used
   589→   589→                IsAdd:         true,
   590→   590→        }
   591→   591→        bindReply := &classify.PolicerClassifySetInterfaceReply{}
   592→   592→        if err := c.ch.SendRequest(bindReq).ReceiveReply(bindReply); err != nil {
   593→   593→                return fmt.Errorf("policer_classify_set_interface failed: %w", err)
   594→   594→        }
   595→   595→        if bindReply.Retval != 0 {
   596→   596→                return fmt.Errorf("policer_classify_set_interface retval=%d", bindReply.Retval)
   597→   597→        }
   598→   598→
   599→   599→        log.Printf("[vpp] Applied policer %d to interface %d via classify table %d + wildcard session (hit_next=%d) via binapi",
   600→   600→                policerIndex, swIfIndex, tableIndex, policerIndex)
   601→   601→        return nil
   602→   602→}

   640→   640→// AddStaticNat creates a 1:1 static NAT mapping (internalIP → externalIP).
   641→   641→// Tries nat44_add_del_static_mapping_v2 first; if that message is unsupported
   642→   642→// on the VPP build (error contains "Unsupported" or "-126"), falls back to
   643→   643→// the V1 nat44_add_del_static_mapping message (same fields minus V2-only
   644→   644→// MatchPool/PoolIPAddress).
   645→   645→func (c *VPPLiveClient) AddStaticNat(internalIP, externalIP string) error {
   646→   646→        if !c.IsConnected() {
   647→   647→                return fmt.Errorf("VPP not connected")
   648→   648→        }
   649→   649→        localIP, err := ip_types.ParseIP4Address(internalIP)
   650→   650→        if err != nil {
   651→   651→                return fmt.Errorf("invalid internal IP '%s': %w", internalIP, err)
   652→   652→        }
   653→   653→        extIP, err := ip_types.ParseIP4Address(externalIP)
   654→   654→        if err != nil {
   655→   655→                return fmt.Errorf("invalid external IP '%s': %w", externalIP, err)
   656→   656→        }
   657→   657→        tag := fmt.Sprintf("static-%s->%s", internalIP, externalIP)
   658→   658→
   659→   659→        // --- Step 1: try V2 message (nat44_add_del_static_mapping_v2) ---
   660→   660→        reqV2 := &nat44_ed.Nat44AddDelStaticMappingV2{
   661→   661→                IsAdd:             true,
   662→   662→                MatchPool:         false,
   663→   663→                Flags:             nat_types.NAT_IS_STATIC, // 1:1 static mapping
   664→   664→                PoolIPAddress:     ip_types.IP4Address{0, 0, 0, 0}, // unused for static
   665→   665→                LocalIPAddress:    localIP,
   666→   666→                ExternalIPAddress: extIP,
   667→   667→                Protocol:          0, // 0 = all protocols (identity mapping)
   668→   668→                LocalPort:         0,
   669→   669→                ExternalPort:      0,
   670→   670→                ExternalSwIfIndex: interface_types.InterfaceIndex(0xFFFFFFFF), // ~0 = use specific external IP
   671→   671→                VrfID:             0,
   672→   672→                Tag:               tag,
   673→   673→        }
   674→   674→        replyV2 := &nat44_ed.Nat44AddDelStaticMappingV2Reply{}
   675→   675→        if errV2 := c.ch.SendRequest(reqV2).ReceiveReply(replyV2); errV2 == nil {
   676→   676→                if replyV2.Retval == 0 {
   677→   677→                        log.Printf("[vpp] Added static NAT V2 %s -> %s via binary API", internalIP, externalIP)
   678→   678→                        return nil
   679→   679→                }
   680→   680→                // V2 message accepted but retval != 0 — fall through to V1
   681→   681→                log.Printf("[vpp] AddStaticNat V2 retval=%d, trying V1...", replyV2.Retval)
   682→   682→        } else {
   683→   683→                // V2 message error — fall back to V1 only if it's the "Unsupported (-126)" case
   684→   684→                errMsg := errV2.Error()
   685→   685→                if strings.Contains(errMsg, "Unsupported") || strings.Contains(errMsg, "-126") {
   686→   686→                        log.Printf("[vpp] AddStaticNat V2 unsupported on this VPP build, falling back to V1: %v", errV2)
   687→   687→                } else {
   688→   688→                        return fmt.Errorf("nat44_add_del_static_mapping_v2 failed: %w", errV2)
   689→   689→                }
   690→   690→        }
   691→   691→
   692→   692→        // --- Step 2: fall back to V1 message (nat44_add_del_static_mapping) ---
   693→   693→        // V1 struct has the same LocalIPAddress/ExternalIPAddress/Flags/Protocol/
   694→   694→        // LocalPort/ExternalPort/ExternalSwIfIndex/VrfID/Tag fields as V2 — it just
   695→   695→        // lacks V2's MatchPool/PoolIPAddress pair (no pool-assigned static mapping
   696→   696→        // support, which we don't use anyway).
   697→   697→        reqV1 := &nat44_ed.Nat44AddDelStaticMapping{
   698→   698→                IsAdd:             true,
   699→   699→                Flags:             nat_types.NAT_IS_STATIC,
   700→   700→                LocalIPAddress:    localIP,
   701→   701→                ExternalIPAddress: extIP,
   702→   702→                Protocol:          0, // identity mapping (all protocols)
   703→   703→                LocalPort:         0,
   704→   704→                ExternalPort:      0,
   705→   705→                ExternalSwIfIndex: interface_types.InterfaceIndex(0xFFFFFFFF),
   706→   706→                VrfID:             0,
   707→   707→                Tag:               tag,
   708→   708→        }
   709→   709→        replyV1 := &nat44_ed.Nat44AddDelStaticMappingReply{}
   710→   710→        if errV1 := c.ch.SendRequest(reqV1).ReceiveReply(replyV1); errV1 != nil {
   711→   711→                return fmt.Errorf("nat44_add_del_static_mapping V1 also failed: %w", errV1)
   712→   712→        }
   713→   713→        if replyV1.Retval != 0 {
   714→   714→                return fmt.Errorf("nat44_add_del_static_mapping V1 retval=%d", replyV1.Retval)
   715→   715→        }
   716→   716→        log.Printf("[vpp] Added static NAT V1 %s -> %s via binary API (V2 fallback)", internalIP, externalIP)
   717→   717→        return nil
   718→   718→}

   919→   919→// ChangeSubscriberBandwidth updates the policer for a subscriber.
   920→   920→// Best-effort: deletes the existing policer (by name) and creates a new one
   921→   921→// with the new rates. The policer name convention is "pol-<subscriberIP>".
   922→   922→//
   923→   923→// NOTE: This does NOT re-bind to the interface — see ApplyPolicerToInterface for
   924→   924→// the limitation of the current binding model.
   925→   925→func (c *VPPLiveClient) ChangeSubscriberBandwidth(subscriberIP string, newDownloadKbps, newUploadKbps uint64) error {
   926→   926→        if !c.IsConnected() {
   927→   927→                return fmt.Errorf("VPP not connected")
   928→   928→        }
   929→   929→
   930→   930→        policerName := fmt.Sprintf("pol-%s", subscriberIP)
   931→   931→        policerName = sanitizePolicerName(policerName)
   932→   932→
   933→   933→        // Look up existing entry (if any) so we can preserve external IP for re-binding
   934→   934→        var existing *subscriberPolicerEntry
   935→   935→        if v, ok := c.subscriberPolicers.Load(subscriberIP); ok {
   936→   936→                existing = v.(*subscriberPolicerEntry)
   937→   937→                // Best-effort delete the old policer (ignore error — may not exist)
   938→   938→                _ = c.DeletePolicer(existing.PolicerName)
   939→   939→        }
   940→   940→
   941→   941→        // Convert kbps → bps (CreatePolicer expects bps)
   942→   942→        dlBps := newDownloadKbps * 1000
   943→   943→        ulBps := newUploadKbps * 1000
   944→   944→        // Use the larger of upload/download as CIR (VPP policers are single-direction per flow).
   945→   945→        // For symmetric policers, EIR = the other direction.
   946→   946→        cirBps := dlBps
   947→   947→        eirBps := ulBps
   948→   948→        if ulBps > dlBps {
   949→   949→                cirBps = ulBps
   950→   950→                eirBps = dlBps
   951→   951→        }
   952→   952→
   953→   953→        policerIdx, err := c.CreatePolicer(policerName, cirBps, eirBps)
   954→   954→        if err != nil {
   955→   955→                return fmt.Errorf("CreatePolicer for %s failed: %w", policerName, err)
   956→   956→        }
   957→   957→
   958→   958→        // Store in-memory state
   959→   959→        entry := &subscriberPolicerEntry{
   960→   960→                SubscriberIP:  subscriberIP,
   961→   961→                PolicerName:   policerName,
   962→   962→                PolicerIndex:  policerIdx,
   963→   963→                DownloadKbps:  newDownloadKbps,
   964→   964→                UploadKbps:    newUploadKbps,
   965→   965→                CreatedAt:     time.Now(),
   966→   966→        }
   967→   967→        if existing != nil {
   968→   968→                entry.ExternalIP = existing.ExternalIP
   969→   969→        }
   970→   970→        c.subscriberPolicers.Store(subscriberIP, entry)
   971→   971→
   972→   972→        log.Printf("[vpp] ChangeSubscriberBandwidth(ip=%s, dl=%d kbps, ul=%d kbps) → policerIndex=%d via binapi",
   973→   973→                subscriberIP, newDownloadKbps, newUploadKbps, policerIdx)
   974→   974→        return nil
   975→   975→}

   977→   977→// DisconnectSubscriber tears down all dataplane state for a subscriber:
   978→   978→// deletes the policer and the static NAT mapping.
   979→   979→func (c *VPPLiveClient) DisconnectSubscriber(subscriberIP string) error {
   980→   980→        if !c.IsConnected() {
   981→   981→                return fmt.Errorf("VPP not connected")
   982→   982→        }
   983→   983→
   984→   984→        var errs []string
   985→   985→
   986→   986→        // 1. Delete the policer (if we tracked one)
   987→   987→        if v, ok := c.subscriberPolicers.Load(subscriberIP); ok {
   988→   988→                entry := v.(*subscriberPolicerEntry)
   989→   989→                if err := c.DeletePolicer(entry.PolicerName); err != nil {
   990→   990→                        errs = append(errs, fmt.Sprintf("delete_policer(%s): %v", entry.PolicerName, err))
   991→   991→                }
   992→   992→                // 2. Delete the static NAT mapping (if we tracked an external IP)
   993→   993→                if entry.ExternalIP != "" {
   994→   994→                        if err := c.DeleteStaticNat(subscriberIP, entry.ExternalIP); err != nil {
   995→   995→                                errs = append(errs, fmt.Sprintf("delete_static_nat(%s->%s): %v", subscriberIP, entry.ExternalIP, err))
   996→   996→                        }
   997→   997→                }
   998→   998→                c.subscriberPolicers.Delete(subscriberIP)
   999→   999→        }
  1000→  1000→
  1001→  1001→        if len(errs) > 0 {
  1002→  1002→                return fmt.Errorf("DisconnectSubscriber partial failure: %s", strings.Join(errs, "; "))
  1003→  1003→        }
  1004→  1004→        log.Printf("[vpp] Disconnected subscriber %s (policer + NAT removed) via binapi", subscriberIP)
  1005→  1005→        return nil
  1006→  1006→}

  1008→  1008→// sanitizePolicerName ensures the name is ≤64 chars and contains only safe bytes
  1009→  1009→// (VPP policer name field is a fixed 64-byte string).
  1010→  1010→func sanitizePolicerName(name string) string {
  1011→  1011→        // Replace dots/dashes with underscores for safety
  1012→  1012→        out := strings.Map(func(r rune) rune {
  1013→  1013→                if r >= 'a' && r <= 'z' {
  1014→  1014→                        return r
  1015→  1015→                }
  1016→  1016→                if r >= 'A' && r <= 'Z' {
  1017→  1017→                        return r
  1018→  1018→                }
  1019→  1019→                if r >= '0' && r <= '9' {
  1020→  1020→                        return r
  1021→  1021→                }
  1022→  1022→                return '_'
  1023→  1023→        }, name)
  1024→  1024→        if len(out) > 60 {
  1025→  1025→                out = out[:60]
  1026→  1026→        }
  1027→  1027→        return out
  1028→  1028→}
