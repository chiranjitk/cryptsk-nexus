    17→// ─── Serialization helper for raw SQL rows (bigint → number) ────────────────
    18→
    19→function serializeRows(rows: Record<string, unknown>[]): Record<string, unknown>[] {
    20→  return rows.map((row) => {
    21→    const serialized: Record<string, unknown> = {};
    22→    for (const [key, value] of Object.entries(row)) {
    23→      serialized[key] = typeof value === "bigint" ? Number(value) : value;
    24→    }
    25→    return serialized;
    26→  });
    27→}

    60→// ─── GET /api/aaa/groups ─────────────────────────────────────────────────────
    61→
    62→export async function GET(req: NextRequest) {
    63→  try {
    64→    await requireAuth(req);
    65→
    66→    const { searchParams } = new URL(req.url);
    67→    const search = searchParams.get("search")?.trim() || "";
    68→    const page = Math.max(1, parseInt(searchParams.get("page") || "1"));
    69→    const limit = Math.min(200, Math.max(1, parseInt(searchParams.get("limit") || "50")));
    70→    const offset = (page - 1) * limit;
    71→
    72→    // Build WHERE clause for search
    73→    const whereClause = search
    74→      ? `WHERE gname ILIKE $1`
    75→      : "";
    76→    const params: unknown[] = search ? [`%${search}%`] : [];
    77→
    78→    // Main query: distinct groupnames from both tables + enrichment
    79→    const [groupRows, totalResult] = await Promise.all([
    80→      db.$queryRawUnsafe<Record<string, unknown>[]>(
    81→        `SELECT * FROM (
    82→          SELECT DISTINCT groupname AS gname FROM radgroupcheck
    83→          UNION
    84→          SELECT DISTINCT groupname AS gname FROM radgroupreply
    85→        ) AS distinct_groups
    86→        ${whereClause}
    87→        ORDER BY gname ASC
    88→        LIMIT ${limit} OFFSET ${offset}`,
    89→        ...params
    90→      ),
    91→      db.$queryRawUnsafe<{ count: string }[]>(
    92→        `SELECT CAST(COUNT(*) AS text) AS count FROM (
    93→          SELECT DISTINCT groupname FROM radgroupcheck
    94→          UNION
    95→          SELECT DISTINCT groupname FROM radgroupreply
    96→        ) AS all_groups
    97→        ${whereClause}`,
    98→        ...params
    99→      ),
   100→    ]);
   101→
   102→    const totalGroups = Number(totalResult[0]?.count || 0);
   103→
   104→    if (groupRows.length === 0) {
   105→      return NextResponse.json({
   106→        success: true,
   107→        groups: [],
   108→        pagination: { page, limit, total: totalGroups, totalPages: 0 },
   109→        stats: { totalGroups: 0, totalUsersAcrossGroups: 0 },
   110→      });
   111→    }
   112→
   113→    // Fetch check attrs, reply attrs, user counts, and RadiusGroup enrichment for all groups
   114→    const groupNames = groupRows.map((r) => r.gname as string);
   115→
   116→    // Check attributes
   117→    const checkRows = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   118→      `SELECT id, groupname, attribute, op, value
   119→       FROM radgroupcheck
   120→       WHERE groupname = ANY($1)
   121→       ORDER BY groupname, attribute`,
   122→      groupNames
   123→    );
   124→
   125→    // Reply attributes
   126→    const replyRows = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   127→      `SELECT id, groupname, attribute, op, value
   128→       FROM radgroupreply
   129→       WHERE groupname = ANY($1)
   130→       ORDER BY groupname, attribute`,
   131→      groupNames
   132→    );
   133→
   134→    // User counts from radusergroup
   135→    const userCountRows = await db.$queryRawUnsafe<{ groupname: string; user_count: string }[]>(
   136→      `SELECT groupname, CAST(COUNT(*) AS text) AS user_count
   137→       FROM radusergroup
   138→       WHERE groupname = ANY($1)
   139→       GROUP BY groupname`,
   140→      groupNames
   141→    );
   142→
   143→    // RadiusGroup Prisma enrichment
   144→    const enrichedGroups = await db.radiusGroup.findMany({
   145→      where: { name: { in: groupNames } },
   146→      select: {
   147→        id: true,
   148→        name: true,
   149→        description: true,
   150→        priority: true,
   151→        speedLimitDown: true,
   152→        speedLimitUp: true,
   153→        dataLimit: true,
   154→        sessionTimeout: true,
   155→        framedIpv6Pool: true,
   156→        delegatedIpv6PrefixPool: true,
   157→      },
   158→    });
   159→
   160→    // Build maps
   161→    const checkMap = new Map<string, Record<string, unknown>[]>();
   162→    for (const row of checkRows) {
   163→      const gn = row.groupname as string;
   164→      if (!checkMap.has(gn)) checkMap.set(gn, []);
   165→      checkMap.get(gn)!.push(row);
   166→    }
   167→
   168→    const replyMap = new Map<string, Record<string, unknown>[]>();
   169→    for (const row of replyRows) {
   170→      const gn = row.groupname as string;
   171→      if (!replyMap.has(gn)) replyMap.set(gn, []);
   172→      replyMap.get(gn)!.push(row);
   173→    }
   174→
   175→    const userCountMap = new Map<string, number>();
   176→    let totalUsersAcrossGroups = 0;
   177→    for (const row of userCountRows) {
   178→      const count = Number(row.user_count);
   179→      userCountMap.set(row.groupname, count);
   180→      totalUsersAcrossGroups += count;
   181→    }
   182→
   183→    const enrichedMap = new Map<string, Record<string, unknown>>();
   184→    for (const eg of enrichedGroups) {
   185→      enrichedMap.set(eg.name, eg as unknown as Record<string, unknown>);
   186→    }
   187→
   188→    // Assemble final groups array
   189→    const groups = groupRows.map((row) => {
   190→      const gname = row.gname as string;
   191→      const checks = serializeRows(checkMap.get(gname) || []);
   192→      const replies = serializeRows(replyMap.get(gname) || []);
   193→      const userCount = userCountMap.get(gname) || 0;
   194→      const enrichment = enrichedMap.get(gname) || null;
   195→
   196→      return {
   197→        groupname: gname,
   198→        checkAttrs: checks,
   199→        replyAttrs: replies,
   200→        userCount,
   201→        checkCount: checks.length,
   202→        replyCount: replies.length,
   203→        enrichment,
   204→        description: enrichment?.description || "",
   205→        priority: enrichment?.priority || 0,
   206→      };
   207→    });
   208→
   209→    const totalPages = Math.ceil(totalGroups / limit);
   210→
   211→    return NextResponse.json({
   212→      success: true,
   213→      groups,
   214→      pagination: { page, limit, total: totalGroups, totalPages },
   215→      stats: {
   216→        totalGroups,
   217→        totalUsersAcrossGroups,
   218→      },
   219→    });
   220→  } catch (error: unknown) {
   221→    if (error instanceof AuthError) {
   222→      return NextResponse.json({ success: false, error: error.message }, { status: error.statusCode });
   223→    }
   224→    console.error("[aaa/groups] GET error:", error);
   225→    const msg = error instanceof Error ? error.message : "Internal server error";
   226→    return NextResponse.json({ success: false, error: msg }, { status: 500 });
   227→  }
   228→}

   230→// ─── POST /api/aaa/groups ────────────────────────────────────────────────────
   231→
   232→interface CreateGroupBody {
   233→  groupname: string;
   234→  description?: string;
   235→  priority?: number;
   236→  speedLimitDown?: number;
   237→  speedLimitUp?: number;
   238→  dataLimit?: number;
   239→  sessionTimeout?: number;
   240→  checkAttrs?: { attribute: string; op?: string; value: string }[];
   241→  replyAttrs?: { attribute: string; op?: string; value: string }[];
   242→}
   243→
   244→export async function POST(req: NextRequest) {
   245→  try {
   246→    const userId = await requireAuth(req);
   247→    const body: CreateGroupBody = await req.json();
   248→
   249→    const { groupname, description, priority, speedLimitDown, speedLimitUp, dataLimit, sessionTimeout, checkAttrs, replyAttrs } = body;
   250→
   251→    // ── Validate groupname
   252→    if (!groupname?.trim()) {
   253→      return NextResponse.json({ success: false, error: "groupname is required" }, { status: 400 });
   254→    }
   255→
   256→    const trimmedName = groupname.trim();
   257→
   258→    if (trimmedName.length > 64) {
   259→      return NextResponse.json({ success: false, error: "groupname must not exceed 64 characters" }, { status: 400 });
   260→    }
   261→
   262→    // ── Validate no duplicate
   263→    const existing = await db.$queryRawUnsafe<{ exists: boolean }[]>(
   264→      `SELECT EXISTS(
   265→        SELECT 1 FROM radgroupcheck WHERE groupname = $1
   266→        UNION
   267→        SELECT 1 FROM radgroupreply WHERE groupname = $1
   268→CAST(      ) AS text)(text AS boolean) AS exists`,
   269→      trimmedName
   270→    );
   271→
   272→    if (existing[0]?.exists) {
   273→      return NextResponse.json({ success: false, error: `Group '${trimmedName}' already exists in FreeRADIUS tables` }, { status: 409 });
   274→    }
   275→
   276→    // Also check RadiusGroup Prisma model
   277→    const existingPrisma = await db.radiusGroup.findUnique({ where: { name: trimmedName } });
   278→    if (existingPrisma) {
   279→      return NextResponse.json({ success: false, error: `Group '${trimmedName}' already exists in RadiusGroup model` }, { status: 409 });
   280→    }
   281→
   282→    // ── Validate and insert check attributes
   283→    if (checkAttrs && checkAttrs.length > 0) {
   284→      for (const attr of checkAttrs) {
   285→        if (!attr.attribute?.trim()) {
   286→          return NextResponse.json({ success: false, error: "Each check attribute must have a non-empty 'attribute'" }, { status: 400 });
   287→        }
   288→        if (!attr.value?.trim()) {
   289→          return NextResponse.json({ success: false, error: `Check attribute '${attr.attribute}' must have a non-empty 'value'` }, { status: 400 });
   290→        }
   291→        const op = attr.op || "==";
   292→        if (!VALID_CHECK_OPS.includes(op)) {
   293→          return NextResponse.json({ success: false, error: `Invalid operator '${op}' for check attribute '${attr.attribute}'. Valid: ${VALID_CHECK_OPS.join(", ")}` }, { status: 400 });
   294→        }
   295→        if (attr.attribute.length > 64 || attr.value.length > 253) {
   296→          return NextResponse.json({ success: false, error: `Attribute '${attr.attribute}' exceeds field length limits (attribute: 64, value: 253)` }, { status: 400 });
   297→        }
   298→      }
   299→
   300→      // Batch insert check attributes
   301→      for (const attr of checkAttrs) {
   302→        const op = attr.op || "==";
   303→        await db.$queryRawUnsafe(
   304→          `INSERT INTO radgroupcheck (groupname, attribute, op, value) VALUES ($1, $2, $3, $4)`,
   305→          trimmedName, attr.attribute.trim(), op, attr.value.trim()
   306→        );
   307→      }
   308→    }
   309→
   310→    // ── Validate and insert reply attributes
   311→    if (replyAttrs && replyAttrs.length > 0) {
   312→      for (const attr of replyAttrs) {
   313→        if (!attr.attribute?.trim()) {
   314→          return NextResponse.json({ success: false, error: "Each reply attribute must have a non-empty 'attribute'" }, { status: 400 });
   315→        }
   316→        if (!attr.value?.trim()) {
   317→          return NextResponse.json({ success: false, error: `Reply attribute '${attr.attribute}' must have a non-empty 'value'` }, { status: 400 });
   318→        }
   319→        const op = attr.op || "=";
   320→        if (!VALID_REPLY_OPS.includes(op)) {
   321→          return NextResponse.json({ success: false, error: `Invalid operator '${op}' for reply attribute '${attr.attribute}'. Valid: ${VALID_REPLY_OPS.join(", ")}` }, { status: 400 });
   322→        }
   323→        if (attr.attribute.length > 64 || attr.value.length > 253) {
   324→          return NextResponse.json({ success: false, error: `Attribute '${attr.attribute}' exceeds field length limits (attribute: 64, value: 253)` }, { status: 400 });
   325→        }
   326→      }
   327→
   328→      // Batch insert reply attributes
   329→      for (const attr of replyAttrs) {
   330→        const op = attr.op || "=";
   331→        await db.$queryRawUnsafe(
   332→          `INSERT INTO radgroupreply (groupname, attribute, op, value) VALUES ($1, $2, $3, $4)`,
   333→          trimmedName, attr.attribute.trim(), op, attr.value.trim()
   334→        );
   335→      }
   336→    }
   337→
   338→    // ── Create/update RadiusGroup Prisma model for enrichment
   339→    await db.radiusGroup.upsert({
   340→      where: { name: trimmedName },
   341→      create: {
   342→        name: trimmedName,
   343→        description: description || "",
   344→        priority: priority || 0,
   345→        speedLimitDown: speedLimitDown || 0,
   346→        speedLimitUp: speedLimitUp || 0,
   347→        dataLimit: dataLimit ?? null,
   348→        sessionTimeout: sessionTimeout ?? null,
   349→      },
   350→      update: {
   351→        description: description || "",
   352→        priority: priority || 0,
   353→        speedLimitDown: speedLimitDown || 0,
   354→        speedLimitUp: speedLimitUp || 0,
   355→        dataLimit: dataLimit ?? null,
   356→        sessionTimeout: sessionTimeout ?? null,
   357→      },
   358→    });
   359→
   360→    // ── Fetch the created data back for response
   361→    const [checks, replies] = await Promise.all([
   362→      db.$queryRawUnsafe<Record<string, unknown>[]>(
   363→        `SELECT id, groupname, attribute, op, value FROM radgroupcheck WHERE groupname = $1 ORDER BY attribute`,
   364→        trimmedName
   365→      ),
   366→      db.$queryRawUnsafe<Record<string, unknown>[]>(
   367→        `SELECT id, groupname, attribute, op, value FROM radgroupreply WHERE groupname = $1 ORDER BY attribute`,
   368→        trimmedName
   369→      ),
   370→    ]);
   371→
   372→    const radiusGroup = await db.radiusGroup.findUnique({ where: { name: trimmedName } });
   373→
   374→    return NextResponse.json({
   375→      success: true,
   376→      RadiusGroup: {
   377→        groupname: trimmedName,
   378→        checkAttrs: serializeRows(checks),
   379→        replyAttrs: serializeRows(replies),
   380→        checkCount: checks.length,
   381→        replyCount: replies.length,
   382→        userCount: 0,
   383→        enrichment: radiusGroup,
   384→      },
   385→    }, { status: 201 });
   386→  } catch (error: unknown) {
   387→    if (error instanceof AuthError) {
   388→      return NextResponse.json({ success: false, error: error.message }, { status: error.statusCode });
   389→    }
   390→    console.error("[aaa/groups] POST error:", error);
   391→    const msg = error instanceof Error ? error.message : "Internal server error";
   392→    return NextResponse.json({ success: false, error: msg }, { status: 500 });
   393→  }
   394→}

   396→// ─── PUT /api/aaa/groups ─────────────────────────────────────────────────────

   398→interface UpdateGroupAction {
   399→  groupname: string;
   400→  action:
   401→    | "add-check"
   402→    | "remove-check"
   403→    | "add-reply"
   404→    | "remove-reply"
   405→    | "update-check"
   406→    | "update-reply"
   407→    | "rename"
   408→    | "update-enrichment"
   409→    | "replace-checks"
   410→    | "replace-replies";
   411→  // For add-check / add-reply
   412→  attribute?: string;
   413→  op?: string;
   414→  value?: string;
   415→  // For remove-check / remove-reply / update-check / update-reply — by row id
   416→  id?: number | string;
   417→  // For update operations — new values
   418→  newAttribute?: string;
   419→  newOp?: string;
   420→  newValue?: string;
   421→  // For rename
   422→  newGroupname?: string;
   423→  // For update-enrichment
   424→  description?: string;
   425→  priority?: number;
   426→  speedLimitDown?: number;
   427→  speedLimitUp?: number;
   428→  dataLimit?: number | null;
   429→  sessionTimeout?: number | null;
   430→  framedIpv6Pool?: string;
   431→  delegatedIpv6PrefixPool?: string;
   432→  // For replace-checks / replace-replies — full arrays
   433→  checkAttrs?: { attribute: string; op?: string; value: string }[];
   434→  replyAttrs?: { attribute: string; op?: string; value: string }[];
   435→}
   436→
   437→export async function PUT(req: NextRequest) {
   438→  try {
   439→    await requireAuth(req);
   440→    const body: UpdateGroupAction = await req.json();
   441→
   442→    const { groupname, action } = body;
   443→
   444→    if (!groupname?.trim()) {
   445→      return NextResponse.json({ success: false, error: "groupname is required" }, { status: 400 });
   446→    }
   447→
   448→    if (!action) {
   449→      return NextResponse.json({ success: false, error: "action is required. Valid: add-check, remove-check, add-reply, remove-reply, update-check, update-reply, rename, update-enrichment, replace-checks, replace-replies" }, { status: 400 });
   450→    }
   451→
   452→    const trimmedName = groupname.trim();
   453→
   454→    // ── Verify group exists in FreeRADIUS tables
   455→    const exists = await db.$queryRawUnsafe<{ exists: boolean }[]>(
   456→      `SELECT EXISTS(
   457→        SELECT 1 FROM radgroupcheck WHERE groupname = $1
   458→        UNION
   459→        SELECT 1 FROM radgroupreply WHERE groupname = $1
   460→CAST(      ) AS text)(text AS boolean) AS exists`,
   461→      trimmedName
   462→    );
   463→
   464→    if (!exists[0]?.exists) {
   465→      return NextResponse.json({ success: false, error: `Group '${trimmedName}' not found` }, { status: 404 });
   466→    }
   467→
   468→    // ── Handle each action ──
   469→
   470→    switch (action) {
   471→      // ── ADD CHECK ATTRIBUTE ────────────────────────────────────────────
   472→      case "add-check": {
   473→        if (!body.attribute?.trim()) {
   474→          return NextResponse.json({ success: false, error: "attribute is required for add-check" }, { status: 400 });
   475→        }
   476→        if (!body.value?.trim() && body.value !== "0") {
   477→          return NextResponse.json({ success: false, error: "value is required for add-check" }, { status: 400 });
   478→        }
   479→        const op = body.op || "==";
   480→        if (!VALID_CHECK_OPS.includes(op)) {
   481→          return NextResponse.json({ success: false, error: `Invalid check operator '${op}'` }, { status: 400 });
   482→        }
   483→
   484→        // Check for duplicate attribute within the same group
   485→        const dupCheck = await db.$queryRawUnsafe<{ count: string }[]>(
   486→          `SELECT CAST(COUNT(*) AS text) AS count FROM radgroupcheck WHERE groupname = $1 AND attribute = $2`,
   487→          trimmedName, body.attribute.trim()
   488→        );
   489→        if (Number(dupCheck[0]?.count) > 0) {
   490→          return NextResponse.json({ success: false, error: `Check attribute '${body.attribute}' already exists for group '${trimmedName}'` }, { status: 409 });
   491→        }
   492→
   493→        const result = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   494→          `INSERT INTO radgroupcheck (groupname, attribute, op, value) VALUES ($1, $2, $3, $4) RETURNING id, groupname, attribute, op, value`,
   495→          trimmedName, body.attribute.trim(), op, body.value.trim()
   496→        );
   497→
   498→        return NextResponse.json({ success: true, action: "add-check", attribute: serializeRows(result)[0] });
   499→      }
   500→
   501→      // ── REMOVE CHECK ATTRIBUTE ────────────────────────────────────────
   502→      case "remove-check": {
   503→        if (!body.id) {
   504→          return NextResponse.json({ success: false, error: "id is required for remove-check" }, { status: 400 });
   505→        }
   506→        const numId = Number(body.id);
   507→
   508→        // Verify the row belongs to this group
   509→        const verify = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   510→          `SELECT id, attribute FROM radgroupcheck WHERE id = $1 AND groupname = $2`,
   511→          numId, trimmedName
   512→        );
   513→        if (verify.length === 0) {
   514→          return NextResponse.json({ success: false, error: `Check attribute id ${numId} not found for group '${trimmedName}'` }, { status: 404 });
   515→        }
   516→
   517→        await db.$queryRawUnsafe(`DELETE FROM radgroupcheck WHERE id = $1`, numId);
   518→
   519→        return NextResponse.json({ success: true, action: "remove-check", removed: { id: numId, attribute: verify[0].attribute } });
   520→      }
   521→
   522→      // ── ADD REPLY ATTRIBUTE ────────────────────────────────────────────
   523→      case "add-reply": {
   524→        if (!body.attribute?.trim()) {
   525→          return NextResponse.json({ success: false, error: "attribute is required for add-reply" }, { status: 400 });
   526→        }
   527→        if (!body.value?.trim() && body.value !== "0") {
   528→          return NextResponse.json({ success: false, error: "value is required for add-reply" }, { status: 400 });
   529→        }
   530→        const op = body.op || "=";
   531→        if (!VALID_REPLY_OPS.includes(op)) {
   532→          return NextResponse.json({ success: false, error: `Invalid reply operator '${op}'` }, { status: 400 });
   533→        }
   534→
   535→        // Check for duplicate
   536→        const dupReply = await db.$queryRawUnsafe<{ count: string }[]>(
   537→          `SELECT CAST(COUNT(*) AS text) AS count FROM radgroupreply WHERE groupname = $1 AND attribute = $2`,
   538→          trimmedName, body.attribute.trim()
   539→        );
   540→        if (Number(dupReply[0]?.count) > 0) {
   541→          return NextResponse.json({ success: false, error: `Reply attribute '${body.attribute}' already exists for group '${trimmedName}'` }, { status: 409 });
   542→        }
   543→
   544→        const result = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   545→          `INSERT INTO radgroupreply (groupname, attribute, op, value) VALUES ($1, $2, $3, $4) RETURNING id, groupname, attribute, op, value`,
   546→          trimmedName, body.attribute.trim(), op, body.value.trim()
   547→        );
   548→
   549→        return NextResponse.json({ success: true, action: "add-reply", attribute: serializeRows(result)[0] });
   550→      }
   551→
   552→      // ── REMOVE REPLY ATTRIBUTE ────────────────────────────────────────
   553→      case "remove-reply": {
   554→        if (!body.id) {
   555→          return NextResponse.json({ success: false, error: "id is required for remove-reply" }, { status: 400 });
   556→        }
   557→        const numId = Number(body.id);
   558→
   559→        const verify = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   560→          `SELECT id, attribute FROM radgroupreply WHERE id = $1 AND groupname = $2`,
   561→          numId, trimmedName
   562→        );
   563→        if (verify.length === 0) {
   564→          return NextResponse.json({ success: false, error: `Reply attribute id ${numId} not found for group '${trimmedName}'` }, { status: 404 });
   565→        }
   566→
   567→        await db.$queryRawUnsafe(`DELETE FROM radgroupreply WHERE id = $1`, numId);
   568→
   569→        return NextResponse.json({ success: true, action: "remove-reply", removed: { id: numId, attribute: verify[0].attribute } });
   570→      }
   571→
   572→      // ── UPDATE CHECK ATTRIBUTE ────────────────────────────────────────
   573→      case "update-check": {
   574→        if (!body.id) {
   575→          return NextResponse.json({ success: false, error: "id is required for update-check" }, { status: 400 });
   576→        }
   577→        const numId = Number(body.id);
   578→
   579→        const verify = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   580→          `SELECT id, groupname, attribute, op, value FROM radgroupcheck WHERE id = $1 AND groupname = $2`,
   581→          numId, trimmedName
   582→        );
   583→        if (verify.length === 0) {
   584→          return NextResponse.json({ success: false, error: `Check attribute id ${numId} not found for group '${trimmedName}'` }, { status: 404 });
   585→        }
   586→
   587→        const current = verify[0];
   588→        const newAttribute = body.newAttribute || (current.attribute as string);
   589→        const newOp = body.newOp || (current.op as string);
   590→        const newValue = body.newValue !== undefined ? body.newValue : (current.value as string);
   591→
   592→        if (!VALID_CHECK_OPS.includes(newOp)) {
   593→          return NextResponse.json({ success: false, error: `Invalid check operator '${newOp}'` }, { status: 400 });
   594→        }
   595→
   596→        // Check for duplicate attribute if attribute name is changing
   597→        if (newAttribute !== current.attribute) {
   598→          const dupCheck = await db.$queryRawUnsafe<{ count: string }[]>(
   599→            `SELECT CAST(COUNT(*) AS text) AS count FROM radgroupcheck WHERE groupname = $1 AND attribute = $2 AND id != $3`,
   600→            trimmedName, newAttribute, numId
   601→          );
   602→          if (Number(dupCheck[0]?.count) > 0) {
   603→            return NextResponse.json({ success: false, error: `Check attribute '${newAttribute}' already exists for group '${trimmedName}'` }, { status: 409 });
   604→          }
   605→        }
   606→
   607→        const result = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   608→          `UPDATE radgroupcheck SET attribute = $1, op = $2, value = $3 WHERE id = $4 RETURNING id, groupname, attribute, op, value`,
   609→          newAttribute, newOp, newValue, numId
   610→        );
   611→
   612→        return NextResponse.json({ success: true, action: "update-check", attribute: serializeRows(result)[0] });
   613→      }
   614→
   615→      // ── UPDATE REPLY ATTRIBUTE ────────────────────────────────────────
   616→      case "update-reply": {
   617→        if (!body.id) {
   618→          return NextResponse.json({ success: false, error: "id is required for update-reply" }, { status: 400 });
   619→        }
   620→        const numId = Number(body.id);
   621→
   622→        const verify = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   623→          `SELECT id, groupname, attribute, op, value FROM radgroupreply WHERE id = $1 AND groupname = $2`,
   624→          numId, trimmedName
   625→        );
   626→        if (verify.length === 0) {
   627→          return NextResponse.json({ success: false, error: `Reply attribute id ${numId} not found for group '${trimmedName}'` }, { status: 404 });
   628→        }
   629→
   630→        const current = verify[0];
   631→        const newAttribute = body.newAttribute || (current.attribute as string);
   632→        const newOp = body.newOp || (current.op as string);
   633→        const newValue = body.newValue !== undefined ? body.newValue : (current.value as string);
   634→
   635→        if (!VALID_REPLY_OPS.includes(newOp)) {
   636→          return NextResponse.json({ success: false, error: `Invalid reply operator '${newOp}'` }, { status: 400 });
   637→        }
   638→
   639→        if (newAttribute !== current.attribute) {
   640→          const dupReply = await db.$queryRawUnsafe<{ count: string }[]>(
   641→            `SELECT CAST(COUNT(*) AS text) AS count FROM radgroupreply WHERE groupname = $1 AND attribute = $2 AND id != $3`,
   642→            trimmedName, newAttribute, numId
   643→          );
   644→          if (Number(dupReply[0]?.count) > 0) {
   645→            return NextResponse.json({ success: false, error: `Reply attribute '${newAttribute}' already exists for group '${trimmedName}'` }, { status: 409 });
   646→          }
   647→        }
   648→
   649→        const result = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   650→          `UPDATE radgroupreply SET attribute = $1, op = $2, value = $3 WHERE id = $4 RETURNING id, groupname, attribute, op, value`,
   651→          newAttribute, newOp, newValue, numId
   652→        );
   653→
   654→        return NextResponse.json({ success: true, action: "update-reply", attribute: serializeRows(result)[0] });
   655→      }
   656→
   657→      // ── RENAME GROUP ──────────────────────────────────────────────────
   658→      case "rename": {
   659→        if (!body.newGroupname?.trim()) {
   660→          return NextResponse.json({ success: false, error: "newGroupname is required for rename" }, { status: 400 });
   661→        }
   662→        const newName = body.newGroupname.trim();
   663→
   664→        if (newName.length > 64) {
   665→          return NextResponse.json({ success: false, error: "newGroupname must not exceed 64 characters" }, { status: 400 });
   666→        }
   667→
   668→        // Check if new name already exists
   669→        const newNameExists = await db.$queryRawUnsafe<{ exists: boolean }[]>(
   670→          `SELECT EXISTS(
   671→            SELECT 1 FROM radgroupcheck WHERE groupname = $1
   672→            UNION
   673→            SELECT 1 FROM radgroupreply WHERE groupname = $1
   674→CAST(          ) AS text)(text AS boolean) AS exists`,
   675→          newName
   676→        );
   677→        if (newNameExists[0]?.exists) {
   678→          return NextResponse.json({ success: false, error: `Group '${newName}' already exists` }, { status: 409 });
   679→        }
   680→
   681→        // Check RadiusGroup
   682→        const rgExists = await db.radiusGroup.findUnique({ where: { name: newName } });
   683→        if (rgExists) {
   684→          return NextResponse.json({ success: false, error: `Group '${newName}' already exists in RadiusGroup model` }, { status: 409 });
   685→        }
   686→
   687→        // Rename in radgroupcheck
   688→        await db.$queryRawUnsafe(`UPDATE radgroupcheck SET groupname = $1 WHERE groupname = $2`, newName, trimmedName);
   689→
   690→        // Rename in radgroupreply
   691→        await db.$queryRawUnsafe(`UPDATE radgroupreply SET groupname = $1 WHERE groupname = $2`, newName, trimmedName);
   692→
   693→        // Rename in radusergroup
   694→        await db.$queryRawUnsafe(`UPDATE radusergroup SET groupname = $1 WHERE groupname = $2`, newName, trimmedName);
   695→
   696→        // Rename in RadiusGroup Prisma model
   697→        const existingRG = await db.radiusGroup.findUnique({ where: { name: trimmedName } });
   698→        if (existingRG) {
   699→          await db.radiusGroup.update({ where: { name: trimmedName }, data: { name: newName } });
   700→        }
   701→
   702→        return NextResponse.json({ success: true, action: "rename", oldGroupname: trimmedName, newGroupname: newName });
   703→      }
   704→
   705→      // ── UPDATE ENRICHMENT (RadiusGroup Prisma model) ──────────────────
   706→      case "update-enrichment": {
   707→        const updateData: Record<string, unknown> = {};
   708→        if (body.description !== undefined) updateData.description = body.description;
   709→        if (body.priority !== undefined) updateData.priority = Number(body.priority);
   710→        if (body.speedLimitDown !== undefined) updateData.speedLimitDown = Number(body.speedLimitDown);
   711→        if (body.speedLimitUp !== undefined) updateData.speedLimitUp = Number(body.speedLimitUp);
   712→        if (body.dataLimit !== undefined) updateData.dataLimit = body.dataLimit ?? null;
   713→        if (body.sessionTimeout !== undefined) updateData.sessionTimeout = body.sessionTimeout ?? null;
   714→        if (body.framedIpv6Pool !== undefined) updateData.framedIpv6Pool = body.framedIpv6Pool;
   715→        if (body.delegatedIpv6PrefixPool !== undefined) updateData.delegatedIpv6PrefixPool = body.delegatedIpv6PrefixPool;
   716→
   717→        // Upsert the RadiusGroup model
   718→        const enriched = await db.radiusGroup.upsert({
   719→          where: { name: trimmedName },
   720→          create: {
   721→            name: trimmedName,
   722→            description: (body.description as string) || "",
   723→            priority: Number(body.priority) || 0,
   724→            speedLimitDown: Number(body.speedLimitDown) || 0,
   725→            speedLimitUp: Number(body.speedLimitUp) || 0,
   726→            dataLimit: body.dataLimit ?? null,
   727→            sessionTimeout: body.sessionTimeout ?? null,
   728→            framedIpv6Pool: (body.framedIpv6Pool as string) || "",
   729→            delegatedIpv6PrefixPool: (body.delegatedIpv6PrefixPool as string) || "",
   730→          },
   731→          update: updateData,
   732→        });
   733→
   734→        return NextResponse.json({ success: true, action: "update-enrichment", enrichment: enriched });
   735→      }
   736→
   737→      // ── REPLACE ALL CHECK ATTRIBUTES ──────────────────────────────────
   738→      case "replace-checks": {
   739→        if (!Array.isArray(body.checkAttrs)) {
   740→          return NextResponse.json({ success: false, error: "checkAttrs array is required for replace-checks" }, { status: 400 });
   741→        }
   742→
   743→        // Validate all attrs before deleting
   744→        for (const attr of body.checkAttrs) {
   745→          if (!attr.attribute?.trim() || (!attr.value?.trim() && attr.value !== "0")) {
   746→            return NextResponse.json({ success: false, error: "Each check attribute must have non-empty attribute and value" }, { status: 400 });
   747→          }
   748→          const op = attr.op || "==";
   749→          if (!VALID_CHECK_OPS.includes(op)) {
   750→            return NextResponse.json({ success: false, error: `Invalid check operator '${op}'` }, { status: 400 });
   751→          }
   752→        }
   753→
   754→        // Delete all existing checks
   755→        await db.$queryRawUnsafe(`DELETE FROM radgroupcheck WHERE groupname = $1`, trimmedName);
   756→
   757→        // Insert new ones
   758→        for (const attr of body.checkAttrs) {
   759→          const op = attr.op || "==";
   760→          await db.$queryRawUnsafe(
   761→            `INSERT INTO radgroupcheck (groupname, attribute, op, value) VALUES ($1, $2, $3, $4)`,
   762→            trimmedName, attr.attribute.trim(), op, attr.value.trim()
   763→          );
   764→        }
   765→
   766→        const newChecks = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   767→          `SELECT id, groupname, attribute, op, value FROM radgroupcheck WHERE groupname = $1 ORDER BY attribute`,
   768→          trimmedName
   769→        );
   770→
   771→        return NextResponse.json({ success: true, action: "replace-checks", checkAttrs: serializeRows(newChecks), count: newChecks.length });
   772→      }
   773→
   774→      // ── REPLACE ALL REPLY ATTRIBUTES ──────────────────────────────────
   775→      case "replace-replies": {
   776→        if (!Array.isArray(body.replyAttrs)) {
   777→          return NextResponse.json({ success: false, error: "replyAttrs array is required for replace-replies" }, { status: 400 });
   778→        }
   779→
   780→        for (const attr of body.replyAttrs) {
   781→          if (!attr.attribute?.trim() || (!attr.value?.trim() && attr.value !== "0")) {
   782→            return NextResponse.json({ success: false, error: "Each reply attribute must have non-empty attribute and value" }, { status: 400 });
   783→          }
   784→          const op = attr.op || "=";
   785→          if (!VALID_REPLY_OPS.includes(op)) {
   786→            return NextResponse.json({ success: false, error: `Invalid reply operator '${op}'` }, { status: 400 });
   787→          }
   788→        }
   789→
   790→        await db.$queryRawUnsafe(`DELETE FROM radgroupreply WHERE groupname = $1`, trimmedName);
   791→
   792→        for (const attr of body.replyAttrs) {
   793→          const op = attr.op || "=";
   794→          await db.$queryRawUnsafe(
   795→            `INSERT INTO radgroupreply (groupname, attribute, op, value) VALUES ($1, $2, $3, $4)`,
   796→            trimmedName, attr.attribute.trim(), op, attr.value.trim()
   797→          );
   798→        }
   799→
   800→        const newReplies = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   801→          `SELECT id, groupname, attribute, op, value FROM radgroupreply WHERE groupname = $1 ORDER BY attribute`,
   802→          trimmedName
   803→        );
   804→
   805→        return NextResponse.json({ success: true, action: "replace-replies", replyAttrs: serializeRows(newReplies), count: newReplies.length });
   806→      }
   807→
   808→      default:
   809→        return NextResponse.json({
   810→          success: false,
   811→          error: `Unknown action '${action}'. Valid actions: add-check, remove-check, add-reply, remove-reply, update-check, update-reply, rename, update-enrichment, replace-checks, replace-replies`,
   812→        }, { status: 400 });
   813→    }
   814→  } catch (error: unknown) {
   815→    if (error instanceof AuthError) {
   816→      return NextResponse.json({ success: false, error: error.message }, { status: error.statusCode });
   817→    }
   818→    console.error("[aaa/groups] PUT error:", error);
   819→    const msg = error instanceof Error ? error.message : "Internal server error";
   820→    return NextResponse.json({ success: false, error: msg }, { status: 500 });
   821→  }
   822→}

   826→export async function DELETE(req: NextRequest) {
   827→  try {
   828→    await requireAuth(req);
   829→
   830→    const { searchParams } = new URL(req.url);
   831→    const groupname = searchParams.get("groupname")?.trim();
   832→
   833→    if (!groupname) {
   834→      return NextResponse.json({ success: false, error: "groupname query parameter is required" }, { status: 400 });
   835→    }
   836→
   837→    // Verify group exists
   838→    const exists = await db.$queryRawUnsafe<{ exists: boolean }[]>(
   839→      `SELECT EXISTS(
   840→        SELECT 1 FROM radgroupcheck WHERE groupname = $1
   841→        UNION
   842→        SELECT 1 FROM radgroupreply WHERE groupname = $1
   843→CAST(      ) AS text)(text AS boolean) AS exists`,
   844→      groupname
   845→    );
   846→
   847→    if (!exists[0]?.exists) {
   848→      return NextResponse.json({ success: false, error: `Group '${groupname}' not found` }, { status: 404 });
   849→    }
   850→
   851→    // Count users assigned to this group before deletion
   852→    const userCountResult = await db.$queryRawUnsafe<{ count: string }[]>(
   853→      `SELECT CAST(COUNT(*) AS text) AS count FROM radusergroup WHERE groupname = $1`,
   854→      groupname
   855→    );
   856→    const affectedUsers = Number(userCountResult[0]?.count || 0);
   857→
   858→    // Count check/reply attrs
   859→    const [checkCountResult, replyCountResult] = await Promise.all([
   860→      db.$queryRawUnsafe<{ count: string }[]>(
   861→        `SELECT CAST(COUNT(*) AS text) AS count FROM radgroupcheck WHERE groupname = $1`,
   862→        groupname
   863→      ),
   864→      db.$queryRawUnsafe<{ count: string }[]>(
   865→        `SELECT CAST(COUNT(*) AS text) AS count FROM radgroupreply WHERE groupname = $1`,
   866→        groupname
   867→      ),
   868→    ]);
   869→    const deletedChecks = Number(checkCountResult[0]?.count || 0);
   870→    const deletedReplies = Number(replyCountResult[0]?.count || 0);
   871→
   872→    // Execute all deletions in a logical sequence
   873→    // 1. Delete all radgroupcheck entries
   874→    await db.$queryRawUnsafe(`DELETE FROM radgroupcheck WHERE groupname = $1`, groupname);
   875→
   876→    // 2. Delete all radgroupreply entries
   877→    await db.$queryRawUnsafe(`DELETE FROM radgroupreply WHERE groupname = $1`, groupname);
   878→
   879→    // 3. Remove all radusergroup assignments for this group
   880→    await db.$queryRawUnsafe(`DELETE FROM radusergroup WHERE groupname = $1`, groupname);
   881→
   882→    // 4. Remove RadiusGroup Prisma model if exists
   883→    try {
   884→      await db.radiusGroup.deleteMany({ where: { name: groupname } });
   885→    } catch {
   886→      // RadiusGroup might not exist, that's fine
   887→    }
   888→
   889→    return NextResponse.json({
   890→      success: true,
   891→      deleted: {
   892→        groupname,
   893→        deletedChecks,
   894→        deletedReplies,
   895→        unassignedUsers: affectedUsers,
   896→      },
   897→    });
   898→  } catch (error: unknown) {
   899→    if (error instanceof AuthError) {
   900→      return NextResponse.json({ success: false, error: error.message }, { status: error.statusCode });
   901→    }
   902→    console.error("[aaa/groups] DELETE error:", error);
   903→    const msg = error instanceof Error ? error.message : "Internal server error";
   904→    return NextResponse.json({ success: false, error: msg }, { status: 500 });
   905→  }
   906→}
