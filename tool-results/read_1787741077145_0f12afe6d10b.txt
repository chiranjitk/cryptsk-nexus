    40→# ─── Step 0: Clean Previous Installation (rerunnable) ─────────────────
    41→clean_previous() {
    42→  info "Cleaning previous installation (if any)..."
    43→  
    44→  # Stop PM2 processes
    45→  npx pm2 delete all 2>/dev/null || true
    46→  
    47→  # Drop old database and user if they exist
    48→  if command -v psql &>/dev/null; then
    49→    sudo -u postgres psql -c "DROP DATABASE IF EXISTS ${DB_NAME};" 2>/dev/null || true
    50→    sudo -u postgres psql -c "DROP USER IF EXISTS ${DB_USER};" 2>/dev/null || true
    51→  fi
    52→  
    53→  # Remove old install dir
    54→  if [ -d "$INSTALL_DIR" ]; then
    55→    warn "Removing ${INSTALL_DIR}..."
    56→    rm -rf "$INSTALL_DIR"
    57→  fi
    58→  
    59→  # Remove old user
    60→  if id "$APP_USER" &>/dev/null; then
    61→    warn "Removing user ${APP_USER}..."
    62→    userdel -r "$APP_USER" 2>/dev/null || true
    63→  fi
    64→  
    65→  ok "Cleanup complete"
    66→}

   122→# ─── Step 2: Install Bun ────────────────────────────────────────────────
   123→install_bun() {
   124→  if command -v bun &>/dev/null; then
   125→    ok "Bun already installed: $(bun --version)"
   126→    return
   127→  fi
   128→  info "Installing Bun runtime..."
   129→  curl -fsSL https://bun.sh/install | bash -s "bun-v${BUN_VERSION}"
   130→  # Make available system-wide
   131→  export BUN_INSTALL="$HOME/.bun"
   132→  export PATH="$BUN_INSTALL/bin:$PATH"
   133→  
   134→  # Add to profile for all users
   135→  echo 'export BUN_INSTALL="$HOME/.bun"' > /etc/profile.d/bun.sh
   136→  echo 'export PATH="$BUN_INSTALL/bin:$PATH"' >> /etc/profile.d/bun.sh
   137→  chmod +x /etc/profile.d/bun.sh
   138→  
   139→  ok "Bun $(bun --version) installed"
   140→}

   193→# ─── Step 5: Configure PostgreSQL ───────────────────────────────────────
   194→configure_postgresql() {
   195→  info "Configuring PostgreSQL..."
   196→  
   197→  # Ensure PostgreSQL is running
   198→  systemctl enable postgresql
   199→  systemctl start postgresql
   200→  
   201→  # Wait for PostgreSQL to be ready
   202→  for i in $(seq 1 30); do
   203→    if sudo -u postgres pg_isready -q; then
   204→      break
   205→    fi
   206→    sleep 1
   207→  done
   208→  
   209→  sudo -u postgres pg_isready -q || fail "PostgreSQL failed to start"
   210→  
   211→  # Create user and database
   212→  info "Creating database user: ${DB_USER}..."
   213→  sudo -u postgres psql -c "CREATE USER ${DB_USER} WITH PASSWORD '${DB_PASS}' SUPERUSER;" 2>/dev/null || \
   214→    sudo -u postgres psql -c "ALTER USER ${DB_USER} WITH PASSWORD '${DB_PASS}' SUPERUSER;"
   215→  
   216→  info "Creating database: ${DB_NAME}..."
   217→  sudo -u postgres psql -c "CREATE DATABASE ${DB_NAME} OWNER ${DB_USER};" 2>/dev/null || \
   218→    sudo -u postgres psql -c "ALTER DATABASE ${DB_NAME} OWNER TO ${DB_USER};"
   219→  
   220→  # Install extensions
   221→  info "Installing PostgreSQL extensions..."
   222→  for ext in pgcrypto pg_stat_statements citext btree_gin btree_gist pg_trgm; do
   223→    sudo -u postgres psql -d "$DB_NAME" -c "CREATE EXTENSION IF NOT EXISTS $ext;" > /dev/null 2>&1
   224→  done
   225→  
   226→  # Configure pg_hba.conf for password auth
   227→  PG_HBA="/etc/postgresql/${PG_VERSION}/main/pg_hba.conf"
   228→  if [ -f "$PG_HBA" ]; then
   229→    info "Configuring pg_hba.conf for md5 authentication..."
   230→    # Backup original
   231→    cp "$PG_HBA" "${PG_HBA}.bak"
   232→    # Write new config
   233→    cat > "$PG_HBA" << 'HBAEOF'
   234→# PostgreSQL Client Authentication Configuration
   235→# TYPE  DATABASE        USER            ADDRESS                 METHOD
   236→local   all             all                                     md5
   237→host    all             all             127.0.0.1/32            md5
   238→host    all             all             ::1/128                 md5
   239→host    all             all             0.0.0.0/0               md5
   240→host    all             all             ::/0                    md5
   241→HBAEOF
   242→    
   243→    # Also allow trust for local unix socket (for sudo -u postgres)
   244→    # This is needed for some maintenance operations
   245→    sed -i '1i local   all             postgres                                peer' "$PG_HBA"
   246→    
   247→    systemctl reload postgresql
   248→  fi
   249→  
   250→  # Configure postgresql.conf
   251→  PG_CONF="/etc/postgresql/${PG_VERSION}/main/postgresql.conf"
   252→  if [ -f "$PG_CONF" ]; then
   253→    # Increase shared buffers and work mem for ISP workload
   254→    if ! rg -q 'shared_buffers.*256MB' "$PG_CONF"; then
   255→      echo "shared_buffers = 256MB" >> "$PG_CONF"
   256→    fi
   257→    if ! rg -q 'work_mem.*64MB' "$PG_CONF"; then
   258→      echo "work_mem = 64MB" >> "$PG_CONF"
   259→    fi
   260→    if ! rg -q 'effective_cache_size.*1GB' "$PG_CONF"; then
   261→      echo "effective_cache_size = 1GB" >> "$PG_CONF"
   262→    fi
   263→    systemctl reload postgresql
   264→  fi
   265→  
   266→  # Test connection
   267→  PGPASSWORD="$DB_PASS" psql -U "$DB_USER" -d "$DB_NAME" -h 127.0.0.1 -c "SELECT 1;" > /dev/null 2>&1 \
   268→    || fail "Database connection test failed"
   269→  
   270→  ok "PostgreSQL configured (${DB_NAME} on port ${DB_PORT})"
   271→}

   297→# ─── Step 8: Install Mini-Service Dependencies ───────────────────────────
   298→install_mini_service_deps() {
   299→  info "Installing mini-service dependencies..."
   300→  cd "$INSTALL_DIR"
   301→  
   302→  # Install shared deps first
   303→  if [ -d "mini-services/shared" ]; then
   304→    cd mini-services/shared
   305→    sudo -u ${APP_USER} npm install --production=false 2>/dev/null || true
   306→    cd "$INSTALL_DIR"
   307→  fi
   308→  
   309→  # Install deps for each mini-service
   310→  for svc_dir in mini-services/*/; do
   311→    [ -f "$svc_dir/package.json" ] || continue
   312→    svc_name=$(basename "$svc_dir")
   313→    [ "$svc_name" = "shared" ] && continue
   314→    
   315→    info "  Installing ${svc_name}..."
   316→    cd "$svc_dir"
   317→    # Generate Prisma client for services that use it
   318→    if [ -f "package.json" ] && rg -q '@prisma/client' package.json 2>/dev/null; then
   319→      sudo -u ${APP_USER} npm install --production=false 2>&1 | tail -2
   320→      # Generate Prisma client using the main schema
   321→      sudo -u ${APP_USER} npx prisma generate --schema="$INSTALL_DIR/prisma/schema.prisma" 2>&1 | tail -2
   322→    else
   323→      sudo -u ${APP_USER} npm install --production=false 2>&1 | tail -2
   324→    fi
   325→    cd "$INSTALL_DIR"
   326→  done
   327→  
   328→  ok "Mini-service dependencies installed"
   329→}
