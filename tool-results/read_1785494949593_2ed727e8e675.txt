   688→            },
   689→          },
   690→        },
   691→        take: 5000,
   692→      }),
   693→      db.subscriber.count({
   694→        where: { createdAt: { gte: startOfLastMonth, lte: endOfLastMonth } },
   695→      }),
   696→      db.installation.findMany({
   697→        where: {
   698→          status: "COMPLETED",

   701→        include: {

   704→              planId: true,

   713→    let installationCostThisMonth = 0;

   716→    }

   718→    let cac = 0;
   719→    if (newThisMonth > 0 && installationCostThisMonth > 0) {
   720→      cac = Math.round(installationCostThisMonth / newThisMonth);
   721→    } else if (newThisMonth > 0) {
   722→      cac = Math.round((revenueThisMonth * 0.15) / newThisMonth);
   723→    }

   725→    let prevInstallationCost = 0;
   726→    for (const inst of completedInstallationsPrevMonth) {

   734→      cacPrev = Math.round((revenueLastMonth * 0.15) / prevMonthNewSubs);

   815→      areaWiseRevenue: areaWiseRevenue.slice(0, 8),
   816→
   817→      // Lists
   818→      recentComplaints: recentComplaints.map((c) => ({
