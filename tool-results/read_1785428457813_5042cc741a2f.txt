   169→// ─── Helpers ──────────────────────────────────────────────────────
   170→function formatLeaseTime(seconds: number): string {
   171→  if (seconds >= 86400) return `${Math.floor(seconds / 86400)}d`;
   172→  if (seconds >= 3600) return `${Math.floor(seconds / 3600)}h`;
   173→  if (seconds >= 60) return `${Math.floor(seconds / 60)}m`;
   174→  return `${seconds}s`;
   175→}
   176→
   177→function formatUptime(seconds: number): string {
   178→  if (!seconds) return "N/A";
   179→  if (seconds < 60) return `${seconds}s`;
   180→  if (seconds < 3600) return `${Math.floor(seconds / 60)}m ${seconds % 60}s`;
   181→  if (seconds < 86400) return `${Math.floor(seconds / 3600)}h ${Math.floor((seconds % 3600) / 60)}m`;
   182→  const d = Math.floor(seconds / 86400);
   183→  const h = Math.floor((seconds % 86400) / 3600);
   184→  return `${d}d ${h}h`;
   185→}
   186→
   187→function formatDateTime(dateStr: string): string {
   188→  if (!dateStr) return "—";
   189→  try {
   190→    return new Date(dateStr).toLocaleString();
   191→  } catch {
   192→    return dateStr;
   193→  }
   194→}
   195→
   196→function calcFreeIPs(subnet: string, netmask: string, rangeStart: string, rangeEnd: string, activeLeases = 0): number {
   197→  try {
   198→    const ipToNum = (ip: string) => {
   199→      const parts = ip.split(".").map(Number);
   200→      return ((parts[0] << 24) | (parts[1] << 16) | (parts[2] << 8) | parts[3]) >>> 0;
   201→    };
   202→    const start = ipToNum(rangeStart);
   203→    const end = ipToNum(rangeEnd);
   204→    const total = end >= start ? end - start + 1 : 0;
   205→    return Math.max(0, total - activeLeases);
   206→  } catch {
   207→    return 0;
   208→  }
   209→}
