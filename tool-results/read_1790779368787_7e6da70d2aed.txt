     5→// This is the Go binary API client that talks to VPP via
     6→// the GoVPP binary API (NEVER vppctl shell — ADR-008).
     7→//

     9→//   - Get interface list from VPP (binary API)

    14→//   - Health check VPP

    16→// Build:
    17→//   cd gateway/vpp/govpp-adapter
    18→//   go mod tidy
    19→//   go build -o cryptsk-govpp-adapter
    20→//   ./cryptsk-govpp-adapter

   137→func main() {
   138→        // Try to connect to VPP via binary API
   139→        if err := vppClient.Connect(); err != nil {
   140→                log.Printf("[warn] VPP not connected: %v", err)
   141→                log.Printf("[info] GoVPP adapter will retry connection every 10s")
   142→        }
   143→
   144→        // Start connection retry loop
   145→        go func() {
   146→                for {
   147→                        time.Sleep(10 * time.Second)
   148→                        if !vppClient.IsConnected() {
   149→                                if err := vppClient.Connect(); err == nil {
   150→                                        log.Printf("[vpp] Reconnected to VPP binary API")
   151→                                }
   152→                        }
   153→                }
   154→        }()

   189→// setCORS sets Access-Control-Allow-Origin: * on all responses so the
   190→// OSS/BSS front-end can call this adapter directly.
   191→func setCORS(w http.ResponseWriter) {
   192→        w.Header().Set("Access-Control-Allow-Origin", "*")
   193→        w.Header().Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
   194→        w.Header().Set("Access-Control-Allow-Headers", "Content-Type, Authorization")
   195→}
   196→
   197→// handleOptions returns true if the request was an OPTIONS preflight that
   198→// has already been answered (caller should return immediately).
   199→func handleOptions(w http.ResponseWriter, r *http.Request) bool {
   200→        if r.Method == http.MethodOptions {
   201→                setCORS(w)
   202→                w.WriteHeader(http.StatusNoContent)
   203→                return true
   204→        }
   205→        return false
   206→}
   207→
   208→// writeJSON writes the given value as JSON with CORS headers.
   209→func writeJSON(w http.ResponseWriter, v interface{}) {
   210→        setCORS(w)
   211→        w.Header().Set("Content-Type", "application/json")
   212→        json.NewEncoder(w).Encode(v)
   213→}
   214→
   215→// writeError writes a JSON error envelope with CORS headers.
   216→func writeError(w http.ResponseWriter, code int, msg string) {
   217→        setCORS(w)
   218→        w.Header().Set("Content-Type", "application/json")
   219→        w.WriteHeader(code)
   220→        json.NewEncoder(w).Encode(map[string]interface{}{
   221→                "success": false,
   222→                "error":   msg,
   223→        })
   224→}

   287→        if handleOptions(w, r) {
   288→                return
   289→        }
   290→        if r.Method != "POST" {
   291→                writeError(w, http.StatusMethodNotAllowed, "Method not allowed")
   292→                return
   293→        }
   294→        var req struct {
   295→                SwIfIndex uint32 `json:"swIfIndex"`
   296→                Up        bool   `json:"up"`
   297→        }
   298→        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
   299→                writeError(w, http.StatusBadRequest, err.Error())
   300→                return
   301→        }
   302→        if err := vppClient.SetInterfaceState(req.SwIfIndex, req.Up); err != nil {
   303→                writeJSON(w, map[string]interface{}{"success": false, "error": err.Error()})
   304→                return
   305→        }
   306→        writeJSON(w, map[string]interface{}{
   307→                "success":   true,
   308→                "swIfIndex": req.SwIfIndex,
   309→                "state":     map[bool]string{true: "up", false: "down"}[req.Up],
   310→        })

   340→func applyHandler(w http.ResponseWriter, r *http.Request) {
   341→        if handleOptions(w, r) {
   342→                return
   343→        }
   344→        if r.Method != "POST" {
   345→                writeError(w, http.StatusMethodNotAllowed, "Method not allowed")
   346→                return
   347→        }
   348→        var obj DataplaneObject
   349→        if err := json.NewDecoder(r.Body).Decode(&obj); err != nil {
   350→                writeError(w, http.StatusBadRequest, err.Error())
   351→                return
   352→        }
   353→
   354→        // Route to appropriate VPP operation based on type
   355→        switch obj.Type {
   356→        case "nat":
   357→                if internal, ok := obj.Config["internalIP"].(string); ok {
   358→                        if external, ok := obj.Config["externalIP"].(string); ok {
   359→                                if err := vppClient.AddStaticNat(internal, external); err != nil {
   360→                                        writeJSON(w, map[string]interface{}{"success": false, "error": err.Error()})
   361→                                        return
   362→                                }
   363→                        }
   364→                }
   365→        case "acl":
   366→                rulesJSON, _ := json.Marshal(obj.Config["rules"])
   367→                var rules []ACLRule
   368→                json.Unmarshal(rulesJSON, &rules)
   369→                aclIdx, err := vppClient.CreateACL(obj.Subscriber, rules)
   370→                if err != nil {
   371→                        writeJSON(w, map[string]interface{}{"success": false, "error": err.Error()})
   372→                        return
   373→                }
   374→                _ = aclIdx
   375→        case "qos":
   376→                if cir, ok := obj.Config["cirBps"].(float64); ok {
   377→                        policerIdx, err := vppClient.CreatePolicer(obj.Subscriber, uint64(cir), 0)
   378→                        if err != nil {
   379→                                writeJSON(w, map[string]interface{}{"success": false, "error": err.Error()})
   380→                                return
   381→                        }
   382→                        _ = policerIdx
   383→                }
   384→        case "disconnect":
   385→                if err := vppClient.DisconnectSubscriber(obj.IP); err != nil {
   386→                        writeJSON(w, map[string]interface{}{"success": false, "error": err.Error()})
   387→                        return
   388→                }
   389→        case "bandwidth":
   390→                if down, ok := obj.Config["downloadKbps"].(float64); ok {
   391→                        if up, ok := obj.Config["uploadKbps"].(float64); ok {
   392→                                if err := vppClient.ChangeSubscriberBandwidth(obj.IP, uint64(down), uint64(up)); err != nil {
   393→                                        writeJSON(w, map[string]interface{}{"success": false, "error": err.Error()})
   394→                                        return
   395→                                }
   396→                        }
   397→                }
   398→        }
   399→
   400→        log.Printf("[govpp] Applied %s for subscriber %s (IP: %s)", obj.Type, obj.Subscriber, obj.IP)
   401→        writeJSON(w, map[string]interface{}{
   402→                "success": true,
   403→                "message": fmt.Sprintf("Dataplane object %s applied for %s via binary API", obj.Type, obj.Subscriber),
   404→        })
   405→}

   420→// subscriberProgramHandler — POST /subscriber/program
   421→//
   422→// Programs a full subscriber dataplane state on VPP:
   423→//   1. CreatePolicer (with speedDownKbps/speedUpKbps)
   424→//   2. AddStaticNat (framedIp → externalIp if provided)
   425→//   3. CreateACL (if aclProfileId or rules array provided)
   426→//
   427→// Body: { sessionId, subscriberId, username, framedIp, mac, nasIp,
   428→//         vlanId?, vrf?, policyId?, aclProfileId?, qosProfileId?,
   429→//         natProfileId?, ipPool?, speedDownKbps, speedUpKbps,
   430→//         timeoutSec?, circuitId?, remoteId?, pppoeSessionId?,
   431→//         dhcpClientId?, externalIp?, aclRules? }
   432→func subscriberProgramHandler(w http.ResponseWriter, r *http.Request) {
   433→        if handleOptions(w, r) {
   434→                return
   435→        }
   436→        if r.Method != "POST" {
   437→                writeError(w, http.StatusMethodNotAllowed, "Method not allowed")
   438→                return
   439→        }
   440→        var req struct {
   441→                SessionID       string `json:"sessionId"`
   442→                SubscriberID    string `json:"subscriberId"`
   443→                Username        string `json:"username"`
   444→                FramedIP        string `json:"framedIp"`
   445→                MAC             string `json:"mac"`
   446→                NasIP           string `json:"nasIp"`
   447→                VlanID          string `json:"vlanId"`
   448→                VRF             string `json:"vrf"`
   449→                PolicyID        string `json:"policyId"`
   450→                ACLProfileID    string `json:"aclProfileId"`
   451→                QoSProfileID    string `json:"qosProfileId"`
   452→                NATProfileID    string `json:"natProfileId"`
   453→                IPPool          string `json:"ipPool"`
   454→                SpeedDownKbps   uint64 `json:"speedDownKbps"`
   455→                SpeedUpKbps     uint64 `json:"speedUpKbps"`
   456→                TimeoutSec      int    `json:"timeoutSec"`
   457→                CircuitID       string `json:"circuitId"`
   458→                RemoteID        string `json:"remoteId"`
   459→                PPPoESessionID  string `json:"pppoeSessionId"`
   460→                DHCPClientID    string `json:"dhcpClientId"`
   461→                ExternalIP      string `json:"externalIp"`
   462→                ACLRules        []ACLRule `json:"aclRules"`
   463→        }
   464→        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
   465→                writeError(w, http.StatusBadRequest, err.Error())
   466→                return
   467→        }
   468→        if req.SessionID == "" {
   469→                // Auto-generate a session ID if none provided
   470→                req.SessionID = uuid.New().String()
   471→        }
   472→        if req.FramedIP == "" {
   473→                writeError(w, http.StatusBadRequest, "framedIp is required")
   474→                return
   475→        }
   476→
   477→        policy := &SessionPolicy{
   478→                SessionID:      req.SessionID,
   479→                SubscriberID:   req.SubscriberID,
   480→                Username:      req.Username,
   481→                FramedIP:       req.FramedIP,
   482→                MAC:            req.MAC,
   483→                NasIP:          req.NasIP,
   484→                VlanID:         req.VlanID,
   485→                VRF:            req.VRF,
   486→                PolicyID:       req.PolicyID,
   487→                ACLProfileID:   req.ACLProfileID,
   488→                QoSProfileID:   req.QoSProfileID,
   489→                NATProfileID:   req.NATProfileID,
   490→                IPPool:         req.IPPool,
   491→                SpeedDownKbps:  req.SpeedDownKbps,
   492→                SpeedUpKbps:    req.SpeedUpKbps,
   493→                TimeoutSec:     req.TimeoutSec,
   494→                CircuitID:      req.CircuitID,
   495→                RemoteID:       req.RemoteID,
   496→                PPPoESessionID: req.PPPoESessionID,
   497→                DHCPClientID:   req.DHCPClientID,
   498→                ExternalIP:     req.ExternalIP,
   499→                CreatedAt:      time.Now(),
   500→                ProgrammedAt:   time.Now(),
   501→        }
   502→
   503→        stepErrors := []string{}
   504→
   505→        // Step 1: Create the policer for the subscriber (bandwidth shaping)
   506→        // Convert kbps → bps (CreatePolicer expects bps)
   507→        if req.SpeedDownKbps > 0 || req.SpeedUpKbps > 0 {
   508→                cirBps := req.SpeedDownKbps * 1000
   509→                eirBps := req.SpeedUpKbps * 1000
   510→                policerName := sanitizePolicerName(fmt.Sprintf("pol-%s", req.FramedIP))
   511→                policerIdx, err := vppClient.CreatePolicer(policerName, cirBps, eirBps)
   512→                if err != nil {
   513→                        stepErrors = append(stepErrors, fmt.Sprintf("CreatePolicer: %v", err))
   514→                        log.Printf("[govpp] /subscriber/program: CreatePolicer failed for %s: %v", req.FramedIP, err)
   515→                } else {
   516→                        policy.PolicerIndex = policerIdx
   517→                        // Track in the VPP client's in-memory subscriber map for CoA / disconnect
   518→                        vppClient.subscriberPolicers.Store(req.FramedIP, &subscriberPolicerEntry{
   519→                                SubscriberIP: req.FramedIP,
   520→                                PolicerName:  policerName,
   521→                                PolicerIndex: policerIdx,
   522→                                DownloadKbps: req.SpeedDownKbps,
   523→                                UploadKbps:   req.SpeedUpKbps,
   524→                                ExternalIP:   req.ExternalIP,
   525→                                CreatedAt:    time.Now(),
   526→                        })
   527→                }
   528→        }
   529→
   530→        // Step 2: Create static NAT mapping (if externalIp provided)
   531→        // NOTE: NAT failures are NON-FATAL — the policer is the primary
   532→        // bandwidth enforcement mechanism, and NAT plugin may not be
   533→        // enabled in all VPP builds. Log as a warning, don't append to stepErrors.
   534→        if req.ExternalIP != "" {
   535→                if err := vppClient.AddStaticNat(req.FramedIP, req.ExternalIP); err != nil {
   536→                        log.Printf("[govpp] /subscriber/program: AddStaticNat WARNING for %s→%s: %v (NAT may not be configured — session proceeds with policer only)",
   537→                                req.FramedIP, req.ExternalIP, err)
   538→                        policy.NatMappingExists = false
   539→                } else {
   540→                        policy.NatMappingExists = true
   541→                }
   542→        }
   543→
   544→        // Step 3: Create ACL (if rules provided)
   545→        if len(req.ACLRules) > 0 {
   546→                aclName := fmt.Sprintf("acl-%s", req.SessionID)
   547→                aclIdx, err := vppClient.CreateACL(aclName, req.ACLRules)
   548→                if err != nil {
   549→                        stepErrors = append(stepErrors, fmt.Sprintf("CreateACL: %v", err))
   550→                        log.Printf("[govpp] /subscriber/program: CreateACL failed for %s: %v", req.SessionID, err)
   551→                } else {
   552→                        policy.ACLIndex = aclIdx
   553→                }
   554→        }
   555→
   556→        // Store in-memory session policy
   557→        sessionPolicies.Store(req.SessionID, policy)
   558→
   559→        log.Printf("[govpp] /subscriber/program: session=%s subscriber=%s ip=%s policer=%d acl=%d nat=%v errors=%d",
   560→                req.SessionID, req.SubscriberID, req.FramedIP,
   561→                policy.PolicerIndex, policy.ACLIndex, policy.NatMappingExists, len(stepErrors))
   562→
   563→        writeJSON(w, map[string]interface{}{
   564→                "success":         len(stepErrors) == 0,
   565→                "sessionId":       req.SessionID,
   566→                "policerIndex":    policy.PolicerIndex,
   567→                "aclIndex":        policy.ACLIndex,
   568→                "natMappingExists": policy.NatMappingExists,
   569→                "warnings":        stepErrors,
   570→                "programmedAt":    policy.ProgrammedAt,
   571→        })

   575→// Body: { sessionId }
   576→// Returns: { verified, checks: {policerExists, aclExists, natMappingExists} }
   577→func subscriberVerifyHandler(w http.ResponseWriter, r *http.Request) {
   578→        if handleOptions(w, r) {
   579→                return
   580→        }
   581→        if r.Method != "POST" {
   582→                writeError(w, http.StatusMethodNotAllowed, "Method not allowed")
   583→                return
   584→        }
   585→        var req struct {
   586→                SessionID string `json:"sessionId"`
   587→        }
   588→        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
   589→                writeError(w, http.StatusBadRequest, err.Error())
   590→                return
   591→        }
   592→        if req.SessionID == "" {
   593→                writeError(w, http.StatusBadRequest, "sessionId is required")
   594→                return
   595→        }
   596→
   597→        v, ok := sessionPolicies.Load(req.SessionID)
   598→        if !ok {
   599→                writeJSON(w, map[string]interface{}{
   600→                        "verified":  false,
   601→                        "sessionId": req.SessionID,
   602→                        "error":     "session not found in in-memory map",
   603→                })
   604→                return
   605→        }
   606→        policy := v.(*SessionPolicy)
   607→
   608→        // Best-effort live checks via VPP binapi (may fail if VPP not connected)
   609→        policerExists := policy.PolicerIndex != 0
   610→        aclExists := policy.ACLIndex != 0
   611→        natMappingExists := policy.NatMappingExists
   612→
   613→        // Try live verification via dump endpoints (best-effort — ignore errors)
   614→        if vppClient.IsConnected() {
   615→                // Check NAT pool for the external IP
   616→                if addrs, err := vppClient.ListNatAddresses(); err == nil {
   617→                        for _, a := range addrs {
   618→                                ipStr := ""
   619→                                // IP4Address is [4]uint8 — format as dotted quad
   620→                                if len(a.IPAddress) == 4 {
   621→                                        ipStr = fmt.Sprintf("%d.%d.%d.%d", a.IPAddress[0], a.IPAddress[1], a.IPAddress[2], a.IPAddress[3])
   622→                                }
   623→                                if ipStr == policy.ExternalIP {
   624→                                        natMappingExists = true
   625→                                        break
   626→                                }
   627→                        }
   628→                }
   629→                // Check policer list
   630→                if pols, err := vppClient.ListPolicers(); err == nil {
   631→                        for _, p := range pols {
   632→                                _ = p // we'd compare by index but PolicerDetails doesn't carry an index field
   633→                        }
   634→                }
   635→        }
   636→
   637→        verified := policerExists || aclExists || natMappingExists
   638→        policy.VerifiedAt = time.Now()
   639→
   640→        writeJSON(w, map[string]interface{}{
   641→                "verified":  verified,
   642→                "sessionId": req.SessionID,
   643→                "checks": map[string]bool{
   644→                        "policerExists":     policerExists,
   645→                        "aclExists":         aclExists,
   646→                        "natMappingExists":  natMappingExists,
   647→                },
   648→                "policy":   policy,
   649→                "checkedAt": policy.VerifiedAt,
   650→        })
   651→}

   655→func subscriberRemoveHandler(w http.ResponseWriter, r *http.Request) {

   657→                return
   658→        }

   660→                writeError(w, http.StatusMethodNotAllowed, "Method not allowed")

   662→        }

   664→                SessionID string `json:"sessionId"`

   667→                writeError(w, http.StatusBadRequest, err.Error())
   668→                return
   669→        }

   671→                writeError(w, http.StatusBadRequest, "sessionId is required")

   682→                return
   683→        }

   690→        if err := vppClient.DisconnectSubscriber(policy.FramedIP); err != nil {
   691→                errs = append(errs, fmt.Sprintf("DisconnectSubscriber: %v", err))

   699→
   700→        sessionPolicies.Delete(req.SessionID)
   701→
   702→        if len(errs) > 0 {
   703→                writeJSON(w, map[string]interface{}{
   704→                        "success":   false,
   705→                        "sessionId": req.SessionID,

   757→// coaHandler — POST /coa
   758→// Body: { sessionId, subscriberIP, downloadKbps, uploadKbps }.
   759→// Updates the policer for an in-flight session.
   760→func coaHandler(w http.ResponseWriter, r *http.Request) {
   761→        if handleOptions(w, r) {
   762→                return
   763→        }
   764→        if r.Method != "POST" {
   765→                writeError(w, http.StatusMethodNotAllowed, "Method not allowed")
   766→                return
   767→        }
   768→        var req struct {
   769→                SessionID     string `json:"sessionId"`
   770→                SubscriberIP  string `json:"subscriberIP"`
   771→                DownloadKbps  uint64 `json:"downloadKbps"`
   772→                UploadKbps    uint64 `json:"uploadKbps"`
   773→        }
   774→        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
   775→                writeError(w, http.StatusBadRequest, err.Error())
   776→                return
   777→        }
   778→
   779→        // Determine the subscriber IP: prefer the request's subscriberIP,
   780→        // otherwise look up from sessionPolicies.
   781→        subscriberIP := req.SubscriberIP
   782→        if subscriberIP == "" && req.SessionID != "" {
   783→                if v, ok := sessionPolicies.Load(req.SessionID); ok {
   784→                        subscriberIP = v.(*SessionPolicy).FramedIP
   785→                }
   786→        }
   787→        if subscriberIP == "" {
   788→                writeError(w, http.StatusBadRequest, "either subscriberIP or a valid sessionId is required")
   789→                return
   790→        }
   791→
   792→        if err := vppClient.ChangeSubscriberBandwidth(subscriberIP, req.DownloadKbps, req.UploadKbps); err != nil {
   793→                writeJSON(w, map[string]interface{}{"success": false, "error": err.Error()})
   794→                return
   795→        }
   796→
   797→        // Update the in-memory SessionPolicy with new rates (and possibly a new policer index)
   798→        if req.SessionID != "" {
   799→                if v, ok := sessionPolicies.Load(req.SessionID); ok {
   800→                        p := v.(*SessionPolicy)
   801→                        p.SpeedDownKbps = req.DownloadKbps
   802→                        p.SpeedUpKbps = req.UploadKbps
   803→                        // Refresh policer index from the VPP client's subscriber map
   804→                        if pv, ok2 := vppClient.subscriberPolicers.Load(subscriberIP); ok2 {
   805→                                p.PolicerIndex = pv.(*subscriberPolicerEntry).PolicerIndex
   806→                        }
   807→                }
   808→        }
   809→
   810→        log.Printf("[govpp] /coa: session=%s ip=%s down=%d up=%d → policer updated",
   811→                req.SessionID, subscriberIP, req.DownloadKbps, req.UploadKbps)
   812→        writeJSON(w, map[string]interface{}{
   813→                "success":      true,
   814→                "sessionId":    req.SessionID,
   815→                "subscriberIP": subscriberIP,
   816→                "downloadKbps": req.DownloadKbps,
   817→                "uploadKbps":   req.UploadKbps,
   818→                "changedAt":    time.Now(),
   819→        })
   820→}
