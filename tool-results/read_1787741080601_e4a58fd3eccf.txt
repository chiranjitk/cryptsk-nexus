   193→   193→# ─── Step 5: Configure PostgreSQL ───────────────────────────────────────
   194→   194→configure_postgresql() {
   195→   195→  info "Configuring PostgreSQL..."
   196→   196→  
   197→   197→  # Ensure PostgreSQL is running
   198→   198→  systemctl enable postgresql
   199→   199→  systemctl start postgresql
   200→   200→  
   201→   201→  # Wait for PostgreSQL to be ready
   202→   202→  for i in $(seq 1 30); do
   203→   203→    if sudo -u postgres pg_isready -q; then
   204→   204→      break
   205→   205→    fi
   206→   206→    sleep 1
   207→   207→  done
   208→   208→  
   209→   209→  sudo -u postgres pg_isready -q || fail "PostgreSQL failed to start"
   210→   210→  
   211→   211→  # Create user and database
   212→   212→  info "Creating database user: ${DB_USER}..."
   213→   213→  sudo -u postgres psql -c "CREATE USER ${DB_USER} WITH PASSWORD '${DB_PASS}' SUPERUSER;" 2>/dev/null || \
   214→   214→    sudo -u postgres psql -c "ALTER USER ${DB_USER} WITH PASSWORD '${DB_PASS}' SUPERUSER;"
   215→   215→  
   216→   216→  info "Creating database: ${DB_NAME}..."
   217→   217→  sudo -u postgres psql -c "CREATE DATABASE ${DB_NAME} OWNER ${DB_USER};" 2>/dev/null || \
   218→   218→    sudo -u postgres psql -c "ALTER DATABASE ${DB_NAME} OWNER TO ${DB_USER};"
   219→   219→  
   220→   220→  # Install extensions
   221→   221→  info "Installing PostgreSQL extensions..."
   222→   222→  for ext in pgcrypto pg_stat_statements citext btree_gin btree_gist pg_trgm; do
   223→   223→    sudo -u postgres psql -d "$DB_NAME" -c "CREATE EXTENSION IF NOT EXISTS $ext;" > /dev/null 2>&1
   224→   224→  done
   225→   225→  
   226→   226→  # Configure pg_hba.conf for password auth
   227→   227→  PG_HBA="/etc/postgresql/${PG_VERSION}/main/pg_hba.conf"
   228→   228→  if [ -f "$PG_HBA" ]; then
   229→   229→    info "Configuring pg_hba.conf for md5 authentication..."
   230→   230→    # Backup original
   231→   231→    cp "$PG_HBA" "${PG_HBA}.bak"
   232→   232→    # Write new config
   233→   233→    cat > "$PG_HBA" << 'HBAEOF'
   234→   234→# PostgreSQL Client Authentication Configuration
   235→   235→# TYPE  DATABASE        USER            ADDRESS                 METHOD
   236→   236→local   all             all                                     md5
   237→   237→host    all             all             127.0.0.1/32            md5
   238→   238→host    all             all             ::1/128                 md5
   239→   239→host    all             all             0.0.0.0/0               md5
   240→   240→host    all             all             ::/0                    md5
   241→   241→HBAEOF
   242→   242→    
   243→   243→    # Also allow trust for local unix socket (for sudo -u postgres)
   244→   244→    # This is needed for some maintenance operations
   245→   245→    sed -i '1i local   all             postgres                                peer' "$PG_HBA"
   246→   246→    
   247→   247→    systemctl reload postgresql
   248→   248→  fi
   249→   249→  
   250→   250→  # Configure postgresql.conf
   251→   251→  PG_CONF="/etc/postgresql/${PG_VERSION}/main/postgresql.conf"
   252→   252→  if [ -f "$PG_CONF" ]; then
   253→   253→    # Increase shared buffers and work mem for ISP workload
   254→   254→    if ! rg -q 'shared_buffers.*256MB' "$PG_CONF"; then
   255→   255→      echo "shared_buffers = 256MB" >> "$PG_CONF"
   256→   256→    fi
   257→   257→    if ! rg -q 'work_mem.*64MB' "$PG_CONF"; then
   258→   258→      echo "work_mem = 64MB" >> "$PG_CONF"
   259→   259→    fi
   260→   260→    if ! rg -q 'effective_cache_size.*1GB' "$PG_CONF"; then
   261→   261→      echo "effective_cache_size = 1GB" >> "$PG_CONF"
   262→   262→    fi
   263→   263→    systemctl reload postgresql
   264→   264→  fi
   265→   265→  
   266→   266→  # Test connection
   267→   267→  PGPASSWORD="$DB_PASS" psql -U "$DB_USER" -d "$DB_NAME" -h 127.0.0.1 -c "SELECT 1;" > /dev/null 2>&1 \
   268→   268→    || fail "Database connection test failed"
   269→   269→  
   270→   270→  ok "PostgreSQL configured (${DB_NAME} on port ${DB_PORT})"
   271→   271→}

   298→   298→install_mini_service_deps() {
   299→   299→  info "Installing mini-service dependencies..."
   300→   300→  cd "$INSTALL_DIR"
   301→   301→  
   302→   302→  # Install shared deps first
   303→   303→  if [ -d "mini-services/shared" ]; then
   304→   304→    cd mini-services/shared
   305→   305→    sudo -u ${APP_USER} npm install --production=false 2>/dev/null || true
   306→   306→    cd "$INSTALL_DIR"
   307→   307→  fi
   308→   308→  
   309→   309→  # Install deps for each mini-service
   310→   310→  for svc_dir in mini-services/*/; do
   311→   311→    [ -f "$svc_dir/package.json" ] || continue
   312→   312→    svc_name=$(basename "$svc_dir")
   313→   313→    [ "$svc_name" = "shared" ] && continue
   314→   314→    
   315→   315→    info "  Installing ${svc_name}..."
   316→   316→    cd "$svc_dir"
   317→   317→    # Generate Prisma client for services that use it
   318→   318→    if [ -f "package.json" ] && rg -q '@prisma/client' package.json 2>/dev/null; then
   319→   319→      sudo -u ${APP_USER} npm install --production=false 2>&1 | tail -2
   320→   320→      # Generate Prisma client using the main schema
   321→   321→      sudo -u ${APP_USER} npx prisma generate --schema="$INSTALL_DIR/prisma/schema.prisma" 2>&1 | tail -2
   322→   322→    else
   323→   323→      sudo -u ${APP_USER} npm install --production=false 2>&1 | tail -2
   324→   324→    fi
   325→   325→    cd "$INSTALL_DIR"
   326→   326→  done
   327→   327→  
   328→   328→  ok "Mini-service dependencies installed"
   329→   329→}
