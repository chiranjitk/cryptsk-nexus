   891→// Deterministic pseudo-random based on seed for consistent-but-varying data
   892→function seededRandom(seed: number): number {
   893→  const x = Math.sin(seed + 1) * 10000;
   894→  return x - Math.floor(x);
   895→}
   896→
   897→// Get time-based variation factor so data looks alive
   898→function timeVariation(base: number, seed: number, range: number = 0.15): number {
   899→  const now = Date.now();
   900→  const minuteBucket = Math.floor(now / 60000); // changes every minute
   901→  const variation = seededRandom(seed + minuteBucket) * range * 2 - range;
   902→  return Math.round(base * (1 + variation));
   903→}
