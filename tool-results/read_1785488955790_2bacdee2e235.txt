   639→  const abortControllerRef = useRef<AbortController | null>(null);
   640→
   641→  // Form state
   642→  const [email, setEmail] = useState("");
   643→  const [password, setPassword] = useState("");
   644→  const [showPassword, setShowPassword] = useState(false);
   645→  const [rememberMe, setRememberMe] = useState(false);
   646→  const [fieldErrors, setFieldErrors] = useState<{ email?: string; password?: string }>({});
   647→  const [shakeKey, setShakeKey] = useState(0);
   648→  const [networkRetry, setNetworkRetry] = useState(0);
   649→  const year = new Date().getFullYear().toString();
   650→
   651→  // Check auth on mount
   652→  useEffect(() => {
   653→    checkAuth();
   654→  }, [checkAuth]);
   655→
   656→  // Cleanup abort controller on unmount
   657→  useEffect(() => {
   658→    return () => {
   659→      abortControllerRef.current?.abort();
   660→    };
   661→  }, []);
   662→
   663→  // If already authenticated, skip showing login (parent handles redirect)
   664→  useEffect(() => {
   665→    if (isAuthenticated) {
   666→      const user = useAuthStore.getState().user;
   667→      if (user) {
   668→        setUser(user);
   669→      }
   670→    }
   671→  }, [isAuthenticated, setUser]);
   672→
   673→  // Clear global error when user types
   674→  const handleEmailChange = useCallback(
   675→    (e: React.ChangeEvent<HTMLInputElement>) => {
   676→      setEmail(e.target.value);
   677→      if (error) clearError();
   678→      if (fieldErrors.email) setFieldErrors((prev) => ({ ...prev, email: undefined }));
   679→      if (networkRetry > 0) setNetworkRetry(0);
   680→    },
   681→    [error, clearError, fieldErrors.email, networkRetry]
   682→  );
   683→
   684→  const handlePasswordChange = useCallback(
   685→    (e: React.ChangeEvent<HTMLInputElement>) => {
   686→      setPassword(e.target.value);
   687→      if (error) clearError();
   688→      if (fieldErrors.password) setFieldErrors((prev) => ({ ...prev, password: undefined }));
   689→      if (networkRetry > 0) setNetworkRetry(0);
   690→    },
   691→    [error, clearError, fieldErrors.password, networkRetry]
   692→  );
   693→
   694→  // ── Welcome TTS after login ──────────────────────────────────────
   695→  const playWelcomeTTS = useCallback((userName?: string) => {
   696→    try {
   697→      const name = userName || "there";
   698→      const text = `Welcome back, ${name}. How can I help you today?`;
   699→      fetch("/api/voice/speak", {
   700→        method: "POST",
   701→        headers: { "Content-Type": "application/json" },
   702→        body: JSON.stringify({ text }),
   703→      })
   704→        .then((res) => {
   705→          if (!res.ok) throw new Error("TTS failed");
   706→          return res.blob();
   707→        })
   708→        .then((blob) => {
   709→          const url = URL.createObjectURL(blob);
   710→          const audio = new Audio(url);
   711→          audio.play().catch(() => {});
   712→          audio.onended = () => URL.revokeObjectURL(url);
   713→          audio.onerror = () => URL.revokeObjectURL(url);
   714→        })
   715→        .catch(() => {
   716→          // TTS failure is non-critical — silent fallback
   717→        });
   718→    } catch {
   719→      // Ignore TTS errors during login
   720→    }
   721→  }, []);
   722→
   723→  // Form submission with timeout & retry
   724→  const handleSubmit = async (e: FormEvent<HTMLFormElement>) => {
   725→    e.preventDefault();
   726→
   727→    // Abort any previous in-flight request
   728→    abortControllerRef.current?.abort();
   729→    const controller = new AbortController();
   730→    abortControllerRef.current = controller;
   731→
   732→    // Client-side validation
   733→    const errors: { email?: string; password?: string } = {};
   734→    if (!email.trim()) {
   735→      errors.email = "Email is required";
   736→    } else if (!isValidEmail(email.trim())) {
   737→      errors.email = "Please enter a valid email address";
   738→    }
   739→    if (!password.trim()) {
   740→      errors.password = "Password is required";
   741→    } else if (password.length < 6) {
   742→      errors.password = "Password must be at least 6 characters";
   743→    }
   744→
   745→    if (Object.keys(errors).length > 0) {
   746→      setFieldErrors(errors);
   747→      setShakeKey((prev) => prev + 1);
   748→      return;
   749→    }
   750→
   751→    setFieldErrors({});
   752→
   753→    // Attempt login with built-in timeout (15s) and auto-retry

   755→    let attempt = 0;
   756→    let success = false;

   758→    while (attempt <= maxRetries && !success) {

   761→      if (attempt > 1) {

   763→      }

   765→      try {

   771→      if (!success) {

   776→          storeError?.includes("timeout") ||

   779→        if (!isNetworkError || attempt > maxRetries || controller.signal.aborted) {

   800→      welcomeEl.innerHTML = `<div style="display:flex;align-items:center;gap:8px;"><svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="#059669" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round"><path d="M22 11.08V12a10 10 0 1 1-5.93-9.14"/><polyline points="22 4 12 14.01 9 11.01"/></svg><div><div style="font-weight:600;font-size:14px;color:#111827;">Welcome back!</div><div style="font-size:13px;color:#4B5563;margin-top:2px;">Signed in as ${escapeHtml(user?.name || email)}</div></div></div>`;
