   721→            </CardHeader>
   722→            <CardContent>
   723→              {/* History Search */}
   724→              <div className="mb-4">
   725→                <div className="relative max-w-sm">
   726→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   727→                  <Input placeholder="Search by code, subscriber name, code, or phone..." value={historySearch} onChange={(e) => { setHistorySearch(e.target.value); setHistoryPage(1); }} className="pl-9" />

   730→
   731→              {historyLoading ? (

   733→              ) : historyItems.length > 0 ? (

   735→                  <Table><TableHeader><TableRow>
   736→                    <TableHead className="text-xs">Voucher Code</TableHead>
   737→                    <TableHead className="text-xs text-right">Amount</TableHead>
   738→                    <TableHead className="text-xs">Subscriber</TableHead>
   739→                    <TableHead className="text-xs">Phone</TableHead>
   740→                    <TableHead className="text-xs">Plan</TableHead>
   741→                    <TableHead className="text-xs">Redeemed At</TableHead>
   742→                  </TableRow></TableHeader>
   743→                  <TableBody>
   744→                    {historyItems.map((v) => (
   745→                      <TableRow key={v.id} className="hover:bg-muted/50 cursor-pointer" onClick={() => setDetailVoucher(v as unknown as VoucherItem)}>

   755→                      </TableRow>
   756→                    ))}
   757→                  </TableBody></Table>
   758→                </div>
   759→              ) : (
   760→                <div className="flex flex-col items-center justify-center py-16"><History className="h-12 w-12 text-muted-foreground/40 mb-3" /><p className="text-muted-foreground font-medium">No redemption history</p><p className="text-sm text-muted-foreground/70 mt-1">When vouchers are redeemed, the events will appear here</p></div>
   761→              )}
   762→
   763→              {/* History Pagination */}
   764→              {historyTotalPages > 1 && (
   765→                <div className="flex items-center justify-between border-t mt-4 pt-3">
   766→                  <p className="text-xs text-muted-foreground">{(historyPage - 1) * limit + 1}-{Math.min(historyPage * limit, historyTotal)} of {historyTotal}</p>
   767→                  <div className="flex items-center gap-1">
   768→                    <Button variant="outline" size="sm" disabled={historyPage <= 1} onClick={() => setHistoryPage(historyPage - 1)}><ChevronLeft className="h-4 w-4" /></Button>
   769→                    {Array.from({ length: Math.min(5, historyTotalPages) }, (_, i) => { let pn: number; if (historyTotalPages <= 5) pn = i + 1; else if (historyPage <= 3) pn = i + 1; else if (historyPage >= historyTotalPages - 2) pn = historyTotalPages - 4 + i; else pn = historyPage - 2 + i; return (<Button key={pn} variant={historyPage === pn ? "default" : "outline"} size="sm" className="h-7 w-8 text-xs" onClick={() => setHistoryPage(pn)}>{pn}</Button>); })}
   770→                    <Button variant="outline" size="sm" disabled={historyPage >= historyTotalPages} onClick={() => setHistoryPage(historyPage + 1)}><ChevronRight className="h-4 w-4" /></Button>
   771→                  </div>
   772→                </div>
   773→              )}
   774→            </CardContent>
   775→          </Card>
   776→        </TabsContent>
   777→      </Tabs>
   778→
   779→      {/* ═══ Generate Dialog ═══ */}
   780→      <Dialog open={showGenerate} onOpenChange={setShowGenerate}>
   781→        <DialogContent className="max-w-md">
   782→          <DialogHeader><DialogTitle>Generate Vouchers</DialogTitle><DialogDescription>Create new vouchers with custom denomination and validity</DialogDescription></DialogHeader>
   783→          <div className="space-y-4 py-2">
   784→            {templates.length > 0 && (
   785→              <div className="space-y-2">
   786→                <Label className="text-xs font-medium">Quick Fill from Template</Label>

   794→            <div className="grid grid-cols-2 gap-3">

   797→            </div>
   798→            <div className="grid grid-cols-2 gap-3">
   799→              <div className="space-y-1"><Label className="text-xs font-medium">Validity (days)</Label><Input type="number" min="1" value={genForm.validityDays} onChange={(e) => setGenForm({ ...genForm, validityDays: e.target.value })} placeholder="30" /></div>
   800→              <div className="space-y-1"><Label className="text-xs font-medium">Plan (optional)</Label>
   801→                <Select value={genForm.planId} onValueChange={(v) => setGenForm({ ...genForm, planId: v })}><SelectTrigger><SelectValue placeholder="Any plan" /></SelectTrigger><SelectContent><SelectItem value="__all__">Any plan</SelectItem>{plans.map((p) => <SelectItem key={p.id} value={p.id}>{p.name}</SelectItem>)}</SelectContent></Select>
   802→              </div>
   803→            </div>
   804→          </div>
   805→          <DialogFooter>
   806→            <Button variant="outline" onClick={() => setShowGenerate(false)}>Cancel</Button>
   807→            <Button className="bg-red-600 hover:bg-red-700 text-white" onClick={handleGenerate} disabled={generateMutation.isPending}>
   808→              {generateMutation.isPending && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   809→              Generate {genForm.count} Voucher(s)
   810→            </Button>
   811→          </DialogFooter>
   812→        </DialogContent>
   813→      </Dialog>
   814→
   815→      {/* ═══ Import CSV Dialog ═══ */}
   816→      <Dialog open={showImport} onOpenChange={handleImportClose}>
   817→        <DialogContent className="max-w-md">
   818→          <DialogHeader><DialogTitle><Download className="h-5 w-5 mr-2 inline" />Import Vouchers from CSV</DialogTitle><DialogDescription>Upload a CSV file to bulk-create vouchers. Max 500 vouchers per import.</DialogDescription></DialogHeader>
   819→          <div className="space-y-4 py-2">
   820→            <div className="border-2 border-dashed rounded-lg p-6 text-center">
   821→              <Download className="h-8 w-8 mx-auto mb-2 text-muted-foreground" />
   822→              {importFile ? (
   823→                <div className="space-y-1">
   824→                  <p className="text-sm font-medium">{importFile.name}</p>
   825→                  <p className="text-xs text-muted-foreground">{(importFile.size / 1024).toFixed(1)} KB</p>
   826→                  <Button variant="ghost" size="sm" className="text-xs text-red-600 mt-1" onClick={() => { setImportFile(null); setImportResult(null); }}>Remove</Button>
   827→                </div>
   828→              ) : (
   829→                <Label htmlFor="csv-upload" className="cursor-pointer">
   830→                  <Input id="csv-upload" type="file" accept=".csv" className="hidden" onChange={(e) => { const f = e.target.files?.[0]; if (f) setImportFile(f); }} />
   831→                  <div className="flex flex-col items-center gap-1">
   832→                    <p className="text-sm text-muted-foreground">Click to select CSV file</p>
   833→                    <p className="text-[10px] text-muted-foreground">Max 5MB • CSV format</p>
   834→                  </div>
   835→                </Label>
   836→              )}
   837→            </div>
   838→
   839→            {/* Import Result */}
   840→            {importResult && (
   841→              <div className={`rounded-lg p-3 text-xs space-y-1 ${importResult.errors.length > 0 ? "bg-amber-50 border border-amber-200" : "bg-green-50 border border-green-200"}`}>
   842→                <p className="font-semibold">{importResult.created} voucher(s) created{importResult.skipped > 0 ? `, ${importResult.skipped} skipped` : ""}</p>
   843→                {importResult.errors.length > 0 && (
   844→                  <div className="mt-1.5 space-y-0.5">
   845→                    <p className="font-medium text-amber-700">Errors ({importResult.errors.length}):</p>
   846→                    {importResult.errors.slice(0, 5).map((err, i) => <p key={i} className="text-amber-600">{err}</p>)}
   847→                    {importResult.errors.length > 5 && <p className="text-amber-500">...and {importResult.errors.length - 5} more</p>}
   848→                  </div>
   849→                )}
   850→              </div>
   851→            )}
   852→
   853→            <div className="bg-muted/50 rounded-lg p-3 text-xs space-y-1">
   854→              <p className="font-semibold">CSV Format Requirements:</p>
   855→              <p>Required columns: <code className="bg-muted px-1 rounded">denomination</code> (or amount/price/value)</p>
   856→              <p>Optional: <code className="bg-muted px-1 rounded">plan</code>, <code className="bg-muted px-1 rounded">validity_days</code>, <code className="bg-muted px-1 rounded">count</code>, <code className="bg-muted px-1 rounded">prefix</code></p>
   857→              <p className="text-muted-foreground mt-1">Example: denomination,plan,validity_days,count</p>
   858→            </div>
   859→          </div>
   860→          <DialogFooter>
   861→            <Button variant="outline" onClick={handleImportClose}>Close</Button>
   862→            <Button className="bg-red-600 hover:bg-red-700 text-white" onClick={handleImport} disabled={!importFile || importMutation.isPending}>
   863→              {importMutation.isPending && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   864→              Import Vouchers
   865→            </Button>
   866→          </DialogFooter>
   867→        </DialogContent>
   868→      </Dialog>
   869→
   870→      {/* ═══ Detail Dialog ═══ */}
   871→      <VoucherDetailDialog
   872→        voucher={detailVoucher}
   873→        onClose={() => setDetailVoucher(null)}
   874→        onPrint={(v) => printVoucherCards([v])}
   875→        onCancel={(id) => cancelMutation.mutate(id)}
   876→      />
   877→    </div>
   878→  );
