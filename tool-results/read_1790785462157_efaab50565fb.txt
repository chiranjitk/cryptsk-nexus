     5→// This is the Go binary API client that talks to VPP via
     6→// the GoVPP binary API (NEVER vppctl shell — ADR-008).
     7→//
     8→// It provides a REST API for the OSS/BSS plane to:
     9→//   - Get interface list from VPP (binary API)

    11→//   - Set interface IP address (binary API)

    14→//   - Health check VPP

    16→// Build:
    17→//   cd gateway/vpp/govpp-adapter
    18→//   go mod tidy
    19→//   go build -o cryptsk-govpp-adapter
    20→//   ./cryptsk-govpp-adapter

   189→// setCORS sets Access-Control-Allow-Origin: * on all responses so the
   190→// OSS/BSS front-end can call this adapter directly.
   191→func setCORS(w http.ResponseWriter) {
   192→        w.Header().Set("Access-Control-Allow-Origin", "*")
   193→        w.Header().Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
   194→        w.Header().Set("Access-Control-Allow-Headers", "Content-Type, Authorization")
   195→}
   196→
   197→// handleOptions returns true if the request was an OPTIONS preflight that
   198→// has already been answered (caller should return immediately).
   199→func handleOptions(w http.ResponseWriter, r *http.Request) bool {
   200→        if r.Method == http.MethodOptions {
   201→                setCORS(w)
   202→                w.WriteHeader(http.StatusNoContent)
   203→                return true
   204→        }
   205→        return false
   206→}
   207→
   208→// writeJSON writes the given value as JSON with CORS headers.
   209→func writeJSON(w http.ResponseWriter, v interface{}) {
   210→        setCORS(w)
   211→        w.Header().Set("Content-Type", "application/json")
   212→        json.NewEncoder(w).Encode(v)
   213→}
   214→
   215→// writeError writes a JSON error envelope with CORS headers.
   216→func writeError(w http.ResponseWriter, code int, msg string) {
   217→        setCORS(w)
   218→        w.Header().Set("Content-Type", "application/json")
   219→        w.WriteHeader(code)
   220→        json.NewEncoder(w).Encode(map[string]interface{}{
   221→                "success": false,
   222→                "error":   msg,
   223→        })
   224→}

   302→        if err := vppClient.SetInterfaceState(req.SwIfIndex, req.Up); err != nil {
   303→                writeJSON(w, map[string]interface{}{"success": false, "error": err.Error()})
   304→                return
   305→        }

   307→                "success":   true,
   308→                "swIfIndex": req.SwIfIndex,
   309→                "state":     map[bool]string{true: "up", false: "down"}[req.Up],

   340→func applyHandler(w http.ResponseWriter, r *http.Request) {
   341→        if handleOptions(w, r) {
   342→                return
   343→        }
   344→        if r.Method != "POST" {
   345→                writeError(w, http.StatusMethodNotAllowed, "Method not allowed")
   346→                return
   347→        }
   348→        var obj DataplaneObject
   349→        if err := json.NewDecoder(r.Body).Decode(&obj); err != nil {
   350→                writeError(w, http.StatusBadRequest, err.Error())
   351→                return
   352→        }
   353→
   354→        // Route to appropriate VPP operation based on type
   355→        switch obj.Type {
   356→        case "nat":
   357→                if internal, ok := obj.Config["internalIP"].(string); ok {
   358→                        if external, ok := obj.Config["externalIP"].(string); ok {
   359→                                if err := vppClient.AddStaticNat(internal, external); err != nil {
   360→                                        writeJSON(w, map[string]interface{}{"success": false, "error": err.Error()})
   361→                                        return
   362→                                }
   363→                        }
   364→                }
   365→        case "acl":
   366→                rulesJSON, _ := json.Marshal(obj.Config["rules"])
   367→                var rules []ACLRule
   368→                json.Unmarshal(rulesJSON, &rules)
   369→                aclIdx, err := vppClient.CreateACL(obj.Subscriber, rules)
   370→                if err != nil {
   371→                        writeJSON(w, map[string]interface{}{"success": false, "error": err.Error()})
   372→                        return
   373→                }
   374→                _ = aclIdx
   375→        case "qos":
   376→                if cir, ok := obj.Config["cirBps"].(float64); ok {
   377→                        policerIdx, err := vppClient.CreatePolicer(obj.Subscriber, uint64(cir), 0)
   378→                        if err != nil {
   379→                                writeJSON(w, map[string]interface{}{"success": false, "error": err.Error()})
   380→                                return
   381→                        }
   382→                        _ = policerIdx
   383→                }
   384→        case "disconnect":
   385→                if err := vppClient.DisconnectSubscriber(obj.IP); err != nil {
   386→                        writeJSON(w, map[string]interface{}{"success": false, "error": err.Error()})
   387→                        return
   388→                }
   389→        case "bandwidth":
   390→                if down, ok := obj.Config["downloadKbps"].(float64); ok {
   391→                        if up, ok := obj.Config["uploadKbps"].(float64); ok {
   392→                                if err := vppClient.ChangeSubscriberBandwidth(obj.IP, uint64(down), uint64(up)); err != nil {
   393→                                        writeJSON(w, map[string]interface{}{"success": false, "error": err.Error()})
   394→                                        return
   395→                                }
   396→                        }
   397→                }
   398→        }
   399→
   400→        log.Printf("[govpp] Applied %s for subscriber %s (IP: %s)", obj.Type, obj.Subscriber, obj.IP)
   401→        writeJSON(w, map[string]interface{}{
   402→                "success": true,
   403→                "message": fmt.Sprintf("Dataplane object %s applied for %s via binary API", obj.Type, obj.Subscriber),
   404→        })
   405→}

   420→// subscriberProgramHandler — POST /subscriber/program
   421→//
   422→// Programs a full subscriber dataplane state on VPP:
   423→//   1. CreatePolicer (with speedDownKbps/speedUpKbps)
   424→//   2. AddStaticNat (framedIp → externalIp if provided)
   425→//   3. CreateACL (if aclProfileId or rules array provided)
   426→//
   427→// Body: { sessionId, subscriberId, username, framedIp, mac, nasIp,
   428→//         vlanId?, vrf?, policyId?, aclProfileId?, qosProfileId?,
   429→//         natProfileId?, ipPool?, speedDownKbps, speedUpKbps,
   430→//         timeoutSec?, circuitId?, remoteId?, pppoeSessionId?,
   431→//         dhcpClientId?, externalIp?, aclRules? }
   432→func subscriberProgramHandler(w http.ResponseWriter, r *http.Request) {
   433→        if handleOptions(w, r) {
   434→                return
   435→        }
   436→        if r.Method != "POST" {
   437→                writeError(w, http.StatusMethodNotAllowed, "Method not allowed")
   438→                return
   439→        }
   440→        var req struct {
   441→                SessionID       string `json:"sessionId"`
   442→                SubscriberID    string `json:"subscriberId"`
   443→                Username        string `json:"username"`
   444→                FramedIP        string `json:"framedIp"`
   445→                MAC             string `json:"mac"`
   446→                NasIP           string `json:"nasIp"`
   447→                VlanID          string `json:"vlanId"`
   448→                VRF             string `json:"vrf"`
   449→                PolicyID        string `json:"policyId"`
   450→                ACLProfileID    string `json:"aclProfileId"`
   451→                QoSProfileID    string `json:"qosProfileId"`
   452→                NATProfileID    string `json:"natProfileId"`
   453→                IPPool          string `json:"ipPool"`
   454→                SpeedDownKbps   uint64 `json:"speedDownKbps"`
   455→                SpeedUpKbps     uint64 `json:"speedUpKbps"`
   456→                TimeoutSec      int    `json:"timeoutSec"`
   457→                CircuitID       string `json:"circuitId"`
   458→                RemoteID        string `json:"remoteId"`
   459→                PPPoESessionID  string `json:"pppoeSessionId"`
   460→                DHCPClientID    string `json:"dhcpClientId"`
   461→                ExternalIP      string `json:"externalIp"`
   462→                ACLRules        []ACLRule `json:"aclRules"`
   463→                // SwIfIndex is the VPP interface index to apply the policer to.
   464→                // Defaults to 2 (tap0 in the standard CRYPTSK VPP config) when unset.
   465→                SwIfIndex       uint32 `json:"swIfIndex"`
   466→        }
   467→        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
   468→                writeError(w, http.StatusBadRequest, err.Error())
   469→                return
   470→        }
   471→        if req.SessionID == "" {
   472→                // Auto-generate a session ID if none provided
   473→                req.SessionID = uuid.New().String()
   474→        }
   475→        if req.FramedIP == "" {
   476→                writeError(w, http.StatusBadRequest, "framedIp is required")
   477→                return
   478→        }
   479→        // Default swIfIndex to 2 (tap0 — the subscriber-facing interface in the
   480→        // standard CRYPTSK VPP config) when the caller doesn't specify one.
   481→        if req.SwIfIndex == 0 {
   482→                req.SwIfIndex = 2
   483→        }
   484→
   485→        policy := &SessionPolicy{
   486→                SessionID:      req.SessionID,
   487→                SubscriberID:   req.SubscriberID,
   488→                Username:      req.Username,
   489→                FramedIP:       req.FramedIP,
   490→                MAC:            req.MAC,
   491→                NasIP:          req.NasIP,
   492→                VlanID:         req.VlanID,
   493→                VRF:            req.VRF,
   494→                PolicyID:       req.PolicyID,
   495→                ACLProfileID:   req.ACLProfileID,
   496→                QoSProfileID:   req.QoSProfileID,
   497→                NATProfileID:   req.NATProfileID,
   498→                IPPool:         req.IPPool,
   499→                SpeedDownKbps:  req.SpeedDownKbps,
   500→                SpeedUpKbps:    req.SpeedUpKbps,
   501→                TimeoutSec:     req.TimeoutSec,
   502→                CircuitID:      req.CircuitID,
   503→                RemoteID:       req.RemoteID,
   504→                PPPoESessionID: req.PPPoESessionID,
   505→                DHCPClientID:   req.DHCPClientID,
   506→                ExternalIP:     req.ExternalIP,
   507→                CreatedAt:      time.Now(),
   508→                ProgrammedAt:   time.Now(),
   509→        }
   510→
   511→        stepErrors := []string{}
   512→
   513→        // Step 1: Create the policer for the subscriber (bandwidth shaping)
   514→        // Convert kbps → bps (CreatePolicer expects bps)
   515→        if req.SpeedDownKbps > 0 || req.SpeedUpKbps > 0 {
   516→                cirBps := req.SpeedDownKbps * 1000
   517→                eirBps := req.SpeedUpKbps * 1000
   518→                policerName := sanitizePolicerName(fmt.Sprintf("pol-%s", req.FramedIP))
   519→                policerIdx, err := vppClient.CreatePolicer(policerName, cirBps, eirBps)
   520→                if err != nil {
   521→                        stepErrors = append(stepErrors, fmt.Sprintf("CreatePolicer: %v", err))
   522→                        log.Printf("[govpp] /subscriber/program: CreatePolicer failed for %s: %v", req.FramedIP, err)
   523→                } else {
   524→                        policy.PolicerIndex = policerIdx
   525→                        // Track in the VPP client's in-memory subscriber map for CoA / disconnect
   526→                        vppClient.subscriberPolicers.Store(req.FramedIP, &subscriberPolicerEntry{
   527→                                SubscriberIP: req.FramedIP,
   528→                                PolicerName:  policerName,
   529→                                PolicerIndex: policerIdx,
   530→                                DownloadKbps: req.SpeedDownKbps,
   531→                                UploadKbps:   req.SpeedUpKbps,
   532→                                ExternalIP:   req.ExternalIP,
   533→                                CreatedAt:    time.Now(),
   534→                        })
   535→                        // Step 1b: attach the policer to the subscriber-facing interface.
   536→                        // Non-fatal: if this fails, the policer still exists in VPP's pool
   537→                        // but is not bound to a traffic flow. Logged as a warning so the
   538→                        // program call still succeeds and the session-engine continues.
   539→                        if err := vppClient.ApplyPolicerToInterface(req.SwIfIndex, policerIdx); err != nil {
   540→                                log.Printf("[govpp] /subscriber/program: ApplyPolicerToInterface WARNING swIfIndex=%d policer=%d: %v (policer exists but not bound to traffic flow)",
   541→                                        req.SwIfIndex, policerIdx, err)
   542→                        }
   543→                }
   544→        }
   545→
   546→        // Step 2: Create static NAT mapping (if externalIp provided)
   547→        // NOTE: NAT failures are NON-FATAL — the policer is the primary
   548→        // bandwidth enforcement mechanism, and NAT plugin may not be
   549→        // enabled in all VPP builds. Log as a warning, don't append to stepErrors.
   550→        if req.ExternalIP != "" {
   551→                if err := vppClient.AddStaticNat(req.FramedIP, req.ExternalIP); err != nil {
   552→                        log.Printf("[govpp] /subscriber/program: AddStaticNat WARNING for %s→%s: %v (NAT may not be configured — session proceeds with policer only)",
   553→                                req.FramedIP, req.ExternalIP, err)
   554→                        policy.NatMappingExists = false
   555→                } else {
   556→                        policy.NatMappingExists = true
   557→                }
   558→        }
   559→
   560→        // Step 3: Create ACL (if rules provided)
   561→        if len(req.ACLRules) > 0 {
   562→                aclName := fmt.Sprintf("acl-%s", req.SessionID)
   563→                aclIdx, err := vppClient.CreateACL(aclName, req.ACLRules)
   564→                if err != nil {
   565→                        stepErrors = append(stepErrors, fmt.Sprintf("CreateACL: %v", err))
   566→                        log.Printf("[govpp] /subscriber/program: CreateACL failed for %s: %v", req.SessionID, err)
   567→                } else {
   568→                        policy.ACLIndex = aclIdx
   569→                }
   570→        }
   571→
   572→        // Store in-memory session policy
   573→        sessionPolicies.Store(req.SessionID, policy)
   574→
   575→        log.Printf("[govpp] /subscriber/program: session=%s subscriber=%s ip=%s policer=%d acl=%d nat=%v errors=%d",
   576→                req.SessionID, req.SubscriberID, req.FramedIP,
   577→                policy.PolicerIndex, policy.ACLIndex, policy.NatMappingExists, len(stepErrors))
   578→
   579→        writeJSON(w, map[string]interface{}{
   580→                "success":         len(stepErrors) == 0,
   581→                "sessionId":       req.SessionID,
   582→                "policerIndex":    policy.PolicerIndex,
   583→                "aclIndex":        policy.ACLIndex,
   584→                "natMappingExists": policy.NatMappingExists,
   585→                "warnings":        stepErrors,
   586→                "programmedAt":    policy.ProgrammedAt,
   587→        })
   588→}

   590→// subscriberVerifyHandler — POST /subscriber/verify
   591→// Body: { sessionId }
   592→// Returns: { verified, checks: {policerExists, aclExists, natMappingExists} }
   593→func subscriberVerifyHandler(w http.ResponseWriter, r *http.Request) {
   594→        if handleOptions(w, r) {
   595→                return
   596→        }
   597→        if r.Method != "POST" {
   598→                writeError(w, http.StatusMethodNotAllowed, "Method not allowed")
   599→                return
   600→        }
   601→        var req struct {
   602→                SessionID string `json:"sessionId"`
   603→        }
   604→        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
   605→                writeError(w, http.StatusBadRequest, err.Error())
   606→                return
   607→        }
   608→        if req.SessionID == "" {
   609→                writeError(w, http.StatusBadRequest, "sessionId is required")
   610→                return
   611→        }
   612→
   613→        v, ok := sessionPolicies.Load(req.SessionID)
   614→        if !ok {
   615→                writeJSON(w, map[string]interface{}{
   616→                        "verified":  false,
   617→                        "sessionId": req.SessionID,
   618→                        "error":     "session not found in in-memory map",
   619→                })
   620→                return
   621→        }
   622→        policy := v.(*SessionPolicy)
   623→
   624→        // Best-effort live checks via VPP binapi (may fail if VPP not connected)
   625→        policerExists := policy.PolicerIndex != 0
   626→        aclExists := policy.ACLIndex != 0
   627→        natMappingExists := policy.NatMappingExists
   628→
   629→        // Try live verification via dump endpoints (best-effort — ignore errors)
   630→        if vppClient.IsConnected() {
   631→                // Check NAT pool for the external IP
   632→                if addrs, err := vppClient.ListNatAddresses(); err == nil {
   633→                        for _, a := range addrs {
   634→                                ipStr := ""
   635→                                // IP4Address is [4]uint8 — format as dotted quad
   636→                                if len(a.IPAddress) == 4 {
   637→                                        ipStr = fmt.Sprintf("%d.%d.%d.%d", a.IPAddress[0], a.IPAddress[1], a.IPAddress[2], a.IPAddress[3])
   638→                                }
   639→                                if ipStr == policy.ExternalIP {
   640→                                        natMappingExists = true
   641→                                        break
   642→                                }
   643→                        }
   644→                }
   645→                // Check policer list
   646→                if pols, err := vppClient.ListPolicers(); err == nil {
   647→                        for _, p := range pols {
   648→                                _ = p // we'd compare by index but PolicerDetails doesn't carry an index field
   649→                        }
   650→                }
   651→        }
   652→
   653→        verified := policerExists || aclExists || natMappingExists
   654→        policy.VerifiedAt = time.Now()
   655→
   656→        writeJSON(w, map[string]interface{}{
   657→                "verified":  verified,
   658→                "sessionId": req.SessionID,
   659→                "checks": map[string]bool{
   660→                        "policerExists":     policerExists,
   661→                        "aclExists":         aclExists,
   662→                        "natMappingExists":  natMappingExists,
   663→                },
   664→                "policy":   policy,
   665→                "checkedAt": policy.VerifiedAt,
   666→        })
   667→}

   670→// Body: { sessionId }. Removes policer + NAT + ACL from VPP.
   671→func subscriberRemoveHandler(w http.ResponseWriter, r *http.Request) {
   672→        if handleOptions(w, r) {
   673→                return
   674→        }
   675→        if r.Method != "POST" {
   676→                writeError(w, http.StatusMethodNotAllowed, "Method not allowed")
   677→                return
   678→        }
   679→        var req struct {
   680→                SessionID string `json:"sessionId"`
   681→        }
   682→        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
   683→                writeError(w, http.StatusBadRequest, err.Error())
   684→                return
   685→        }
   686→        if req.SessionID == "" {
   687→                writeError(w, http.StatusBadRequest, "sessionId is required")

   689→        }
   690→

   698→                return
   699→        }
   700→        policy := v.(*SessionPolicy)
   701→
   702→        var errs []string
   703→
   704→        // 1. DisconnectSubscriber (deletes policer + static NAT, uses VPP client's
   705→        //    in-memory subscriberPolicers map).
   706→        if err := vppClient.DisconnectSubscriber(policy.FramedIP); err != nil {
   707→                errs = append(errs, fmt.Sprintf("DisconnectSubscriber: %v", err))

   709→

   714→        }
   715→
   716→        sessionPolicies.Delete(req.SessionID)
   717→
   718→        if len(errs) > 0 {
   719→                writeJSON(w, map[string]interface{}{
   720→                        "success":   false,
   721→                        "sessionId": req.SessionID,
   722→                        "errors":    errs,

   773→// coaHandler — POST /coa
   774→// Body: { sessionId, subscriberIP, downloadKbps, uploadKbps }.
   775→// Updates the policer for an in-flight session.
   776→func coaHandler(w http.ResponseWriter, r *http.Request) {
   777→        if handleOptions(w, r) {
   778→                return
   779→        }
   780→        if r.Method != "POST" {
   781→                writeError(w, http.StatusMethodNotAllowed, "Method not allowed")
   782→                return
   783→        }
   784→        var req struct {
   785→                SessionID     string `json:"sessionId"`
   786→                SubscriberIP  string `json:"subscriberIP"`
   787→                DownloadKbps  uint64 `json:"downloadKbps"`
   788→                UploadKbps    uint64 `json:"uploadKbps"`
   789→        }
   790→        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
   791→                writeError(w, http.StatusBadRequest, err.Error())
   792→                return
   793→        }
   794→
   795→        // Determine the subscriber IP: prefer the request's subscriberIP,
   796→        // otherwise look up from sessionPolicies.
   797→        subscriberIP := req.SubscriberIP
   798→        if subscriberIP == "" && req.SessionID != "" {
   799→                if v, ok := sessionPolicies.Load(req.SessionID); ok {
   800→                        subscriberIP = v.(*SessionPolicy).FramedIP
   801→                }
   802→        }
   803→        if subscriberIP == "" {
   804→                writeError(w, http.StatusBadRequest, "either subscriberIP or a valid sessionId is required")
   805→                return
   806→        }
   807→
   808→        if err := vppClient.ChangeSubscriberBandwidth(subscriberIP, req.DownloadKbps, req.UploadKbps); err != nil {
   809→                writeJSON(w, map[string]interface{}{"success": false, "error": err.Error()})
   810→                return
   811→        }
   812→
   813→        // Update the in-memory SessionPolicy with new rates (and possibly a new policer index)
   814→        if req.SessionID != "" {
   815→                if v, ok := sessionPolicies.Load(req.SessionID); ok {
   816→                        p := v.(*SessionPolicy)
   817→                        p.SpeedDownKbps = req.DownloadKbps
   818→                        p.SpeedUpKbps = req.UploadKbps
   819→                        // Refresh policer index from the VPP client's subscriber map
   820→                        if pv, ok2 := vppClient.subscriberPolicers.Load(subscriberIP); ok2 {
   821→                                p.PolicerIndex = pv.(*subscriberPolicerEntry).PolicerIndex
   822→                        }
   823→                }
   824→        }
   825→
   826→        log.Printf("[govpp] /coa: session=%s ip=%s down=%d up=%d → policer updated",
   827→                req.SessionID, subscriberIP, req.DownloadKbps, req.UploadKbps)
   828→        writeJSON(w, map[string]interface{}{
   829→                "success":      true,
   830→                "sessionId":    req.SessionID,
   831→                "subscriberIP": subscriberIP,
   832→                "downloadKbps": req.DownloadKbps,
   833→                "uploadKbps":   req.UploadKbps,
   834→                "changedAt":    time.Now(),
   835→        })
   836→}
