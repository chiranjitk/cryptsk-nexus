    40→def nv(d, *keys, default=""):
    41→    """Navigate nested dict/list with keys"""
    42→    v = d
    43→    for k in keys:
    44→        if isinstance(v, dict):
    45→            v = v.get(k, default)
    46→        elif isinstance(v, list) and isinstance(k, int):
    47→            v = v[k] if k < len(v) else default
    48→        else:
    49→            return default
    50→    return v if v is not None else default
