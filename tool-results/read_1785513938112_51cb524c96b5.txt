    48→function rec(id, desc, expected, actual, evidence = '') {
    49→  const pass = String(actual) === String(expected);
    50→  let ev = '';
    51→  if (typeof evidence === 'string') {
    52→    ev = evidence.length > 500 ? evidence.substring(0, 500) + '...' : evidence;
    53→  } else if (Array.isArray(evidence)) {
    54→    ev = JSON.stringify(evidence);
    55→  } else if (evidence && typeof evidence === 'object') {
    56→    ev = JSON.stringify(evidence);
    57→    if (ev.length > 500) ev = ev.substring(0, 500) + '...';
    58→  }
    59→  RESULTS.push({ id, desc, expected, actual, pass, evidence: ev });
    60→  const icon = pass ? 'PASS' : 'FAIL';
    61→  console.log(`  [${icon}] ${id}: ${desc} | Expected: ${expected} | Got: ${actual}`);
    62→}
