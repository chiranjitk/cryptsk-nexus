   212→// ─── Relative time helper ───────────────────────────────
   213→function relativeTime(dateStr: string): string {
   214→  const now = Date.now();
   215→  const then = new Date(dateStr).getTime();
   216→  const diffMs = now - then;
   217→  const seconds = Math.floor(diffMs / 1000);
   218→  if (seconds < 60) return "just now";
   219→  const minutes = Math.floor(seconds / 60);
   220→  if (minutes < 60) return `${minutes}m ago`;
   221→  const hours = Math.floor(minutes / 60);
   222→  if (hours < 24) return `${hours}h ago`;
   223→  const days = Math.floor(hours / 24);
   224→  if (days < 30) return `${days}d ago`;
   225→  const months = Math.floor(days / 30);
   226→  if (months < 12) return `${months}mo ago`;
   227→  return `${Math.floor(months / 12)}y ago`;
   228→}

   230→function activityDotColor(dateStr: string, status: string): string {
   231→  if (status === "SUSPENDED" || status === "DISCONNECTED") return "bg-red-400";
   232→  if (!dateStr) return "bg-gray-400";
   233→  const diffH = (Date.now() - new Date(dateStr).getTime()) / (1000 * 60 * 60);
   234→  if (diffH < 24) return "bg-green-500";
   235→  if (diffH < 72) return "bg-yellow-400";
   236→  return "bg-red-400";
   237→}

   245→async function uploadFile(file: File, type: "profile" | "kyc"): Promise<string> {
   246→  const formData = new FormData();
   247→  formData.append("file", file);
   248→  formData.append("type", type);
   249→  const resp = await fetch("/api/upload", { method: "POST", body: formData });
   250→  if (!resp.ok) {
   251→    const err = await resp.json().catch(() => ({}));
   252→    throw new Error(err.error || "Upload failed");
   253→  }
   254→  const data = await resp.json();
   255→  return data.path; // store relative path in DB
   256→}

   258→// ─── Bulk action mutation ───────────────────────────────
   259→function useBulkStatusMutation() {
   260→  const queryClient = useQueryClient();
   261→  return useMutation({
   262→    mutationFn: ({ ids, status }: { ids: string[]; status: string }) =>
   263→      apiFetch("/api/subscribers/bulk", { method: "POST", body: JSON.stringify({ action: "change-status", subscriberIds: ids, status }) }),
   264→    onSuccess: (d: any, vars) => {
   265→      if (d.error) { toast.error(d.error); return; }
   266→      const labels: Record<string, string> = { ACTIVE: "activated", SUSPENDED: "suspended" };
   267→      toast.success(`${vars.ids.length} subscriber${vars.ids.length > 1 ? "s" : ""} ${labels[vars.status] || "updated"}`);
   268→      queryClient.invalidateQueries({ queryKey: ["subscribers"] });
   269→    },
   270→    onError: () => toast.error("Bulk action failed"),
   271→  });
   272→}
