   161→ * Close a session by setting it to CLOSED
   162→ */
   163→async function closeSession(
   164→  sessionId: string,
   165→  cause: string,
   166→  reason: string,
   167→  terminatedBy: string
   168→) {
   169→  const session = await db.nasSession.findUnique({ where: { sessionId } });
   170→  if (!session) return null;
   171→  if (session.status === "CLOSED") return session;
   172→
   173→  const now = new Date();
   174→  const sessionTimeSec = Math.floor((now.getTime() - session.startTime.getTime()) / 1000);
   175→
   176→  const closed = await db.nasSession.update({
   177→    where: { id: session.id },
   178→    data: {
   179→      status: "CLOSED",
   180→      stopTime: now,
   181→      terminateCause: cause,
   182→      disconnectReason: reason,
   183→      terminatedBy,
   184→      sessionTimeSec,
   185→    },
   186→  });
   187→
   188→  // Update subscriber last auth info
   189→  await db.subscriber.update({
   190→    where: { id: session.subscriberId },
   191→    data: {
   192→      lastAuthAt: now,
   193→      lastAuthResult: cause,
   194→    },
   195→  });
   196→
   197→  return closed;
   198→}

   253→// ─── Auto-Enforcement Cron (every 30s) ─────────────────────
   254→// Automatically enforces data/time/idle limits on active sessions
   255→// like 24Online / Antlabs built-in enforcement
   256→
   257→setInterval(async () => {
   258→  try {
   259→    const nasConfig = await db.nasConfig.findUnique({ where: { id: "builtin" } });
   260→    if (!nasConfig) return;
   261→
   262→    const enforceData = nasConfig.enforceDataLimit ?? true;
   263→    const enforceTime = nasConfig.enforceTimeLimit ?? true;
   264→    const enforceIdle = nasConfig.enforceIdleTimeout ?? true;
   265→
   266→    // Skip if all enforcement is disabled
   267→    if (!enforceData && !enforceTime && !enforceIdle) return;
   268→
   269→    const activeSessions = await db.nasSession.findMany({
   270→      where: { status: "ACTIVE" },
   271→    });
   272→
   273→    if (activeSessions.length === 0) return;
   274→
   275→    const now = Date.now();
   276→    let disconnectedCount = 0;
   277→
   278→    for (const session of activeSessions) {
   279→      const sessionAgeSec = Math.floor((now - session.startTime.getTime()) / 1000);
   280→      const totalBytes = Number(session.inputOctets) + Number(session.outputOctets);
   281→      const totalMb = totalBytes / (1024 * 1024);
   282→      const idleSec = Math.floor((now - session.lastActivity.getTime()) / 1000);
   283→
   284→      let shouldDisconnect = false;
   285→      let reason = "";
   286→      let eventType = "";
   287→
   288→      // Check data limit
   289→      if (enforceData && session.dataLimitMb && totalMb >= session.dataLimitMb) {
   290→        shouldDisconnect = true;
   291→        reason = `Data limit reached: ${totalMb.toFixed(1)}MB / ${session.dataLimitMb}MB`;
   292→        eventType = "DATA_LIMIT_REACHED";
   293→      }
   294→
   295→      // Check session timeout
   296→      if (!shouldDisconnect && enforceTime && session.sessionTimeoutSec && sessionAgeSec >= session.sessionTimeoutSec) {
   297→        shouldDisconnect = true;
   298→        reason = `Session timeout: ${formatDurationCron(sessionAgeSec)} / ${formatDurationCron(session.sessionTimeoutSec)}`;
   299→        eventType = "TIME_LIMIT_REACHED";
   300→      }
   301→
   302→      // Check idle timeout
   303→      if (!shouldDisconnect && enforceIdle && session.idleTimeoutSec && idleSec >= session.idleTimeoutSec) {
   304→        shouldDisconnect = true;
   305→        reason = `Idle timeout: ${formatDurationCron(idleSec)} / ${formatDurationCron(session.idleTimeoutSec)}`;
   306→        eventType = "IDLE_TIMEOUT";
   307→      }
   308→
   309→      if (shouldDisconnect) {
   310→        const cause = eventType === "DATA_LIMIT_REACHED" ? "Data-Limit"
   311→          : eventType === "TIME_LIMIT_REACHED" ? "Session-Timeout"
   312→          : "Idle-Timeout";
   313→
   314→        try {
   315→          await closeSession(session.sessionId, cause, reason, "cron");
   316→
   317→          await logEvent({
   318→            nasSessionId: session.id,
   319→            sessionId: session.sessionId,
   320→            subscriberId: session.subscriberId,
   321→            username: session.username,
   322→            eventType,
   323→            context: { reason, sessionAgeSec, totalMb, idleSec },
   324→            source: "cron",
   325→          });
   326→
   327→          broadcastWs("session_stop", {
   328→            sessionId: session.sessionId,
   329→            username: session.username,
   330→            cause,
   331→          });
   332→
   333→          disconnectedCount++;
   334→        } catch (err) {
   335→          logger.error("Auto-enforce failed for session", { sessionId: session.sessionId, error: String(err) });
   336→        }
   337→      }
   338→    }
   339→
   340→    if (disconnectedCount > 0) {
   341→      logger.info("Auto-enforcement completed", { checked: activeSessions.length, disconnected: disconnectedCount });
   342→    }
   343→  } catch (err) {
   344→    logger.error("Auto-enforcement error", { error: String(err) });
   345→  }
   346→}, 30_000);

   348→function formatDurationCron(seconds: number): string {
   349→  if (seconds < 60) return `${seconds}s`;
   350→  if (seconds < 3600) return `${Math.floor(seconds / 60)}m`;
   351→  const h = Math.floor(seconds / 3600);
   352→  const m = Math.floor((seconds % 3600) / 60);
   353→  return `${h}h ${m}m`;
   354→}

   738→    }

   744→        { subscriber: { phone: { contains: search } } },
   745→        { subscriber: { code: { contains: search, mode: "insensitive" } } },
   746→      ];
   747→    }
   748→
   749→    const [sessions, total] = await Promise.all([
   750→      db.nasSession.findMany({
   751→        where,
   752→        include: {
   753→          subscriber: {
   754→            select: {
   755→              id: true,
   756→              name: true,
   757→              phone: true,
   758→              code: true,
