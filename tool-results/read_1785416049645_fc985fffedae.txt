   263→// ─── Recent Pages helpers ──────────────────────────────────────
   264→
   265→const RECENT_PAGES_KEY = "cryptsk-recent-pages";
   266→const MAX_RECENT_PAGES = 5;
   267→
   268→interface RecentPageEntry {
   269→  label: string;
   270→  section: string;
   271→  timestamp: number;
   272→}
   273→
   274→function getRecentPages(): RecentPageEntry[] {
   275→  if (typeof window === "undefined") return [];
   276→  try {
   277→    const stored = localStorage.getItem(RECENT_PAGES_KEY);
   278→    if (!stored) return [];
   279→    return JSON.parse(stored) as RecentPageEntry[];
   280→  } catch {
   281→    return [];
   282→  }
   283→}
   284→
   285→function addRecentPage(label: string, section: string) {
   286→  try {
   287→    const existing = getRecentPages();
   288→    // Remove duplicate if exists
   289→    const filtered = existing.filter(
   290→      (p) => !(p.label === label && p.section === section)
   291→    );
   292→    // Add to front
   293→    filtered.unshift({ label, section, timestamp: Date.now() });
   294→    // Keep only max entries
   295→    const trimmed = filtered.slice(0, MAX_RECENT_PAGES);
   296→    localStorage.setItem(RECENT_PAGES_KEY, JSON.stringify(trimmed));
   297→  } catch {
   298→    // ignore localStorage errors
   299→  }
   300→}

   305→// ─── Main Command Palette Component ────────────────────────────
   306→
   307→export function CommandPalette() {
   308→  const {
   309→    commandPaletteOpen,
   310→    setCommandPaletteOpen,
   311→    toggleCommandPalette,
   312→    currentPage,
   313→    currentSection,
   314→    setCurrentPage,
   315→  } = useAppStore();
   316→  const mounted = useSyncExternalStore(
   317→    emptySubscribe,
   318→    () => true,
   319→    () => false
   320→  );
   321→  const recentPages: RecentPageEntry[] = mounted ? getRecentPages() : [];
   322→
   323→  // Subscriber search state
   324→  const [query, setQuery] = useState("");
   325→  const [subscriberResults, setSubscriberResults] = useState<SubscriberResult[]>([]);
   326→  const [subscriberTotal, setSubscriberTotal] = useState(0);
   327→  const [subscriberLoading, setSubscriberLoading] = useState(false);
   328→  const [subscriberError, setSubscriberError] = useState(false);
   329→  const debounceRef = useRef<NodeJS.Timeout | null>(null);
   330→
   331→  const hasQuery = query.trim().length >= 2;
   332→  const showSubscriberResults = hasQuery && (subscriberResults.length > 0 || subscriberLoading || subscriberError);
   333→
   334→  // Keyboard shortcut: Ctrl+K / Cmd+K
   335→  useEffect(() => {
   336→    function handleKeyDown(e: KeyboardEvent) {
   337→      if ((e.metaKey || e.ctrlKey) && e.key === "k") {
   338→        e.preventDefault();
   339→        e.stopPropagation();
   340→        toggleCommandPalette();
   341→      }
   342→    }
   343→    document.addEventListener("keydown", handleKeyDown, true);
   344→    return () => document.removeEventListener("keydown", handleKeyDown, true);
   345→  }, [toggleCommandPalette]);
   346→
   347→  // Reset state when dialog closes
   348→  const handleOpenChange = useCallback(
   349→    (nextOpen: boolean) => {
   350→      setCommandPaletteOpen(nextOpen);
   351→      if (!nextOpen) {
   352→        // Reset subscriber search state when closing
   353→        setQuery("");
   354→        setSubscriberResults([]);
   355→        setSubscriberTotal(0);
   356→        setSubscriberLoading(false);
   357→        setSubscriberError(false);
   358→      }
   359→    },
   360→    [setCommandPaletteOpen]
   361→  );
   362→
   363→  // Debounced subscriber search (300ms)
   364→  useEffect(() => {
   365→    if (debounceRef.current) clearTimeout(debounceRef.current);
   366→
   367→    if (query.trim().length < 2) {
   368→      setSubscriberResults([]);
   369→      setSubscriberTotal(0);
   370→      setSubscriberLoading(false);
   371→      setSubscriberError(false);
   372→      return;
   373→    }
   374→
   375→    setSubscriberLoading(true);
   376→    setSubscriberError(false);
   377→
   378→    debounceRef.current = setTimeout(async () => {
   379→      try {
   380→        const res = await fetch(
   381→          `/api/subscribers?search=${encodeURIComponent(query.trim())}&limit=5`,
   382→          { credentials: "include" }
   383→        );
   384→        if (!res.ok) throw new Error("Search failed");
   385→        const data = await res.json();
   386→        setSubscriberResults(data.subscribers ?? []);
   387→        setSubscriberTotal(data.total ?? 0);
   388→      } catch {
   389→        setSubscriberError(true);
   390→        setSubscriberResults([]);
   391→      } finally {
   392→        setSubscriberLoading(false);
   393→      }
   394→    }, 300);
   395→
   396→    return () => {
   397→      if (debounceRef.current) clearTimeout(debounceRef.current);
   398→    };
   399→  }, [query]);
   400→
   401→  // Navigate to a page and record it in recent pages
   402→  const handleNavigate = useCallback(
   403→    (pageLabel: string, section: string) => {
   404→      setCurrentPage(pageLabel, section);
   405→      addRecentPage(pageLabel, section);
   406→      setCommandPaletteOpen(false);
   407→    },
   408→    [setCurrentPage, setCommandPaletteOpen]
   409→  );
   410→
   411→  // Handle subscriber selection → navigate to subscribers page with specific ID
   412→  const handleSubscriberSelect = useCallback(
   413→    (subscriber: SubscriberResult) => {
   414→      setCommandPaletteOpen(false);
   415→      // Navigate to subscribers page with the specific subscriber
   416→      setCurrentPage("Subscribers", "MAIN");
   417→      window.location.hash = `subscribers?id=${subscriber.id}`;
   418→      toast.success(`Viewing ${subscriber.name}`);
   419→    },
   420→    [setCommandPaletteOpen, setCurrentPage]
   421→  );
   422→
   423→  // Handle quick action
   424→  const handleQuickAction = useCallback(
   425→    (action: QuickAction) => {
   426→      setCurrentPage(action.targetPage, action.section);
   427→      addRecentPage(action.targetPage, action.section);
   428→      setCommandPaletteOpen(false);
   429→    },
   430→    [setCurrentPage, setCommandPaletteOpen]
   431→  );
   432→
   433→  // Build the value string for cmdk fuzzy matching
   434→  const pageValue = useCallback(
   435→    (page: PageEntry) => {
   436→      const kw = page.keywords?.join(" ") ?? "";
   437→      return `${page.label} ${page.section} ${kw}`.toLowerCase();
   438→    },
   439→    []
   440→  );
   441→
   442→  // Resolve icon from page label
   443→  const resolvePageIcon = useCallback((label: string, section: string): LucideIcon => {
   444→    const sectionPages = PAGE_REGISTRY[section];
   445→    if (sectionPages) {
   446→      const found = sectionPages.find((p) => p.label === label);
   447→      if (found) return found.icon;
   448→    }
   449→    return FileText;
   450→  }, []);
   451→
   452→  return (
   453→    <Dialog open={commandPaletteOpen} onOpenChange={handleOpenChange}>
   454→      <DialogContent
   455→        className="overflow-hidden p-0 sm:max-w-xl"
   456→        showCloseButton={false}
   457→        a11yTitle="Command Palette"
   458→      >
   459→        <Command
   460→          shouldFilter={true}
   461→          className="[&_[cmdk-group-heading]]:text-muted-foreground **:data-[slot=command-input-wrapper]:h-12 [&_[cmdk-group-heading]]:px-2 [&_[cmdk-group-heading]]:font-medium [&_[cmdk-group]]:px-2 [&_[cmdk-group]:not([hidden])_~[cmdk-group]]:pt-0 [&_[cmdk-input-wrapper]_svg]:h-5 [&_[cmdk-input-wrapper]_svg]:w-5 [&_[cmdk-input]]:h-12 [&_[cmdk-item]]:px-2 [&_[cmdk-item]]:py-3 [&_[cmdk-item]_svg]:h-5 [&_[cmdk-item]_svg]:w-5"
   462→        >
   463→          <CommandInput
   464→            placeholder="Search subscribers, pages, actions..."
   465→            value={query}
   466→            onValueChange={setQuery}
   467→          />
   468→
   469→          {/* ── Subscriber Search Results (shown when query has 2+ chars) ── */}
   470→          {showSubscriberResults && (
   471→            <div className="border-b border-border/50">
   472→              {/* Section heading */}
   473→              <div className="flex items-center justify-between px-3 py-1.5">
   474→                <p className="text-[11px] font-semibold text-muted-foreground uppercase tracking-wider">
   475→                  Search Subscribers
   476→                </p>
   477→                {subscriberTotal > 5 && !subscriberLoading && (
   478→                  <span className="text-[10px] text-muted-foreground/60">
   479→                    {subscriberTotal} found — showing top 5
   480→                  </span>
   481→                )}
   482→              </div>
   483→
   484→              {/* Loading state */}
   485→              {subscriberLoading && (
   486→                <div className="flex items-center gap-3 px-3 py-4 text-muted-foreground">
   487→                  <Loader2 className="h-4 w-4 animate-spin shrink-0" />
   488→                  <span className="text-sm">Searching subscribers...</span>
   489→                </div>
   490→              )}
   491→
   492→              {/* Error state */}
   493→              {subscriberError && !subscriberLoading && (
   494→                <div className="flex items-center gap-3 px-3 py-3 text-muted-foreground">
   495→                  <AlertCircle className="h-4 w-4 text-red-500 shrink-0" />
   496→                  <span className="text-sm">Search failed. Try again.</span>
   497→                </div>
   498→              )}
   499→
   500→              {/* Results */}
   501→              {!subscriberLoading && !subscriberError && subscriberResults.length > 0 && (
   502→                <div className="divide-y divide-border/30">
   503→                  {subscriberResults.map((subscriber) => {
   504→                    const StatusIcon =
   505→                      SUBSCRIBER_STATUS_ICONS[subscriber.status] || Wifi;
   506→                    const statusStyle =
   507→                      SUBSCRIBER_STATUS_STYLES[subscriber.status] ||
   508→                      SUBSCRIBER_STATUS_STYLES.DISCONNECTED;
   509→                    const statusLabel =
   510→                      SUBSCRIBER_STATUS_LABELS[subscriber.status] ||
   511→                      subscriber.status;
   512→
   513→                    return (
   514→                      <div
