    13→// ─── Global fetch interceptor ────────────────────────────────
    14→// Injects Bearer token into all /api/ requests for proxy/gateway
    15→// environments where HttpOnly cookies are stripped.
    16→if (typeof window !== "undefined") {
    17→  const _originalFetch = window.fetch;
    18→  window.fetch = async (input: RequestInfo | URL, init?: RequestInit): Promise<Response> => {
    19→    const url = typeof input === "string" ? input : input instanceof URL ? input.href : input.url;
    20→    if (url.includes("/api/") && !init?.headers?.["Authorization"]) {
    21→      try {
    22→        const token = localStorage.getItem("cryptsk_auth_token");
    23→        if (token) {
    24→          init = init || {};
    25→          init.headers = {
    26→            ...(typeof init.headers === "object" ? init.headers : {}),
    27→            Authorization: `Bearer ${token}`,
    28→          };
    29→        }
    30→      } catch { /* ignore */ }
    31→    }
    32→    return _originalFetch.call(window, input, init);
    33→  };
    34→}

    36→// ─── Error Boundary ─────────────────────────────────────────────
    37→class PageErrorBoundary extends Component<
    38→  { children: ReactNode; pageName: string },
    39→  { hasError: boolean; error: Error | null }
    40→> {
    41→  constructor(props: { children: ReactNode; pageName: string }) {
    42→    super(props);
    43→    this.state = { hasError: false, error: null };
    44→  }
    45→  static getDerivedStateFromError(error: Error) {
    46→    return { hasError: true, error };
    47→  }
    48→  componentDidCatch(error: Error, info: ErrorInfo) {
    49→    console.error(`[PageErrorBoundary] Error in ${this.props.pageName}:`, error, info);
    50→  }
    51→  handleReset = () => {
    52→    this.setState({ hasError: false, error: null });
    53→  };
    54→  render() {
    55→    if (this.state.hasError) {
    56→      return (
    57→        <div className="flex flex-col items-center justify-center h-96 gap-4 text-center px-4">
    58→          <div className="rounded-full bg-red-100 p-4">
    59→            <svg className="h-8 w-8 text-red-600" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2}>
    60→              <path strokeLinecap="round" strokeLinejoin="round" d="M12 9v2m0 4h.01M21 12a9 9 0 11-18 0 9 9 0 0118 0z" />
    61→            </svg>
    62→          </div>
    63→          <h2 className="text-lg font-semibold text-foreground">Something went wrong</h2>
    64→          <p className="text-sm text-muted-foreground max-w-md">
    65→            The page &quot;{this.props.pageName}&quot; encountered an error while rendering. Try refreshing or navigating to another page.
    66→          </p>
    67→          <p className="text-xs text-muted-foreground/70 font-mono max-w-lg">
    68→            {this.state.error?.message}
    69→          </p>
    70→          <div className="flex gap-2 mt-2">
    71→            <Button variant="outline" size="sm" onClick={this.handleReset}>
    72→              Try Again
    73→            </Button>
    74→            <Button variant="outline" size="sm" onClick={() => window.location.reload()}>
    75→              Refresh Page
    76→            </Button>
    77→          </div>
    78→        </div>
    79→      );
    80→    }
    81→    return this.props.children;
    82→  }
    83→}

   416→// ─── Root Page ─────────────────────────────────────────────────
   417→
   418→export default function PageRouter() {
   419→  const { isAuthenticated, isLoading, isLoggingIn } = useAuthStore();
   420→  const { currentPage, setCurrentPage } = useAppStore();
   421→  const searchParams = useSearchParams();
   422→
   423→  // Handle ?portal=selfcare query param to open Self-Care Portal directly (e.g. from new tab)
   424→  useEffect(() => {
   425→    const portal = searchParams.get("portal");
   426→    if (portal === "selfcare" && currentPage !== "SelfCare") {
   427→      setCurrentPage("SelfCare", "SELF-CARE");
   428→    }
   429→  }, [searchParams, currentPage, setCurrentPage]);
   430→
   431→  // Show loading spinner only during initial auth check, NOT during login
   432→  if (isLoading && !isAuthenticated && !isLoggingIn) {
   433→    return (
   434→      <div className="fixed inset-0 z-50 flex items-center justify-center bg-white">
   435→        <div className="flex flex-col items-center gap-4">
   436→          <Loader2 className="w-8 h-8 text-red-600 animate-spin" />
   437→          <p className="text-sm text-slate-500">Verifying session…</p>
   438→        </div>
   439→      </div>
   440→    );
   441→  }
   442→
   443→  // Self-Care Portal has its own auth — always render it regardless of admin auth state
   444→  if (currentPage === "SelfCare") {
   445→    return (
   446→      <PageErrorBoundary pageName="SelfCare">
   447→        <div className="animate-page-transition">
   448→          <SelfCarePage />
   449→        </div>
   450→      </PageErrorBoundary>
   451→    );
   452→  }
   453→
   454→  // Show login page if not authenticated
   455→  if (!isAuthenticated) {
   456→    return <LoginPage />;
   457→  }
   458→
   459→  // Authenticated — show the app
   460→  return <AuthenticatedApp />;
   461→}
