    18→
    19→function formatBytes(bytes: number): string {
    20→  if (bytes >= 1_073_741_824) return `${(bytes / 1_073_741_824).toFixed(2)} GB`;
    21→  if (bytes >= 1_048_576) return `${(bytes / 1_048_576).toFixed(1)} MB`;
    22→  if (bytes >= 1_024) return `${(bytes / 1_024).toFixed(0)} KB`;
    23→  return `${bytes} B`;
    24→}
    25→
    26→function formatDuration(seconds: number): string {
    27→  if (!seconds || seconds <= 0) return "—";
    28→  if (seconds < 60) return `${seconds}s`;
    29→  if (seconds < 3600) return `${Math.floor(seconds / 60)}m ${seconds % 60}s`;
    30→  const h = Math.floor(seconds / 3600);
    31→  const m = Math.floor((seconds % 3600) / 60);
    32→  return `${h}h ${m}m`;
    33→}
    34→
    35→function formatDate(d: string | null): string {
    36→  if (!d) return "—";
    37→  try { return new Date(d).toLocaleString(); } catch { return d; }
    38→}
