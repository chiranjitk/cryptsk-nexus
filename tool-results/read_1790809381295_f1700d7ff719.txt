   149→// Humanize the cron expressions used by the registry (minute hour dom month dow).
   150→function describeCron(expr: string): string {
   151→  const f = expr.trim().split(/\s+/);
   152→  if (f.length !== 5) return expr;
   153→  const [min, hour, dom, , dow] = f;
   154→  if (min === "*" && hour === "*") return "Every hour";
   155→  if (!/^\d+$/.test(hour) || !/^\d+$/.test(min)) return expr;
   156→  const hhmm = `${hour.padStart(2, "0")}:${min.padStart(2, "0")}`;
   157→  if (dom === "*" && dow === "*") return `Daily at ${hhmm}`;
   158→  if (dom === "*" && /^\d+$/.test(dow)) {
   159→    const days = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"];
   160→    const d = Number(dow) % 7;
   161→    return `Weekly on ${days[d]} at ${hhmm}`;
   162→  }
   163→  if (/^\d+$/.test(dom) && dow === "*") {
   164→    const d = Number(dom);
   165→    if (d >= 1 && d <= 31) {
   166→      const suffix =
   167→        d === 1 || d === 21 || d === 31 ? "st" : d === 2 || d === 22 ? "nd" : d === 3 || d === 23 ? "rd" : "th";
   168→      return `Monthly on the ${d}${suffix} at ${hhmm}`;
   169→    }
   170→  }
   171→  return expr;
   172→}

   174→// Relative countdown until nextRun ("2h 13m", "due now" when overdue).
   175→function formatCountdown(nextRun: string, now: number): string {
   176→  const diff = new Date(nextRun).getTime() - now;
   177→  if (!Number.isFinite(diff)) return "—";
   178→  if (diff <= 0) return "due now";
   179→  const mins = Math.floor(diff / 60000);
   180→  if (mins < 1) return "<1 min";
   181→  const d = Math.floor(mins / 1440);
   182→  const h = Math.floor((mins % 1440) / 60);
   183→  const m = mins % 60;
   184→  if (d > 0) return `${d}d ${h}h`;
   185→  if (h > 0) return `${h}h ${m}m`;
   186→  return `${m}m`;
   187→}
   188→
   189→// Relative time for past timestamps ("just now", "5m ago", "3h ago", "2d ago").
   190→function formatAgo(ts: string, now: number): string {
   191→  const diff = now - new Date(ts).getTime();
   192→  if (!Number.isFinite(diff) || diff < 0) return "just now";
   193→  const mins = Math.floor(diff / 60000);
   194→  if (mins < 1) return "just now";
   195→  if (mins < 60) return `${mins}m ago`;
   196→  const h = Math.floor(mins / 60);
   197→  if (h < 24) return `${h}h ${mins % 60}m ago`;
   198→  return `${Math.floor(h / 24)}d ago`;
   199→}

   201→// 840 → "840ms", 1200 → "1.2s", 125000 → "2m 05s"
   202→function formatDuration(ms?: number): string {
   203→  if (ms == null) return "—";
   204→  if (ms < 1000) return `${ms}ms`;
   205→  if (ms < 60000) return `${(ms / 1000).toFixed(ms < 10000 ? 1 : 0)}s`;
   206→  const m = Math.floor(ms / 60000);
   207→  const s = Math.round((ms % 60000) / 1000);
   208→  return `${m}m ${String(s).padStart(2, "0")}s`;
   209→}

   211→// "archivedAuditLogs" → "archived audit logs"
   212→function prettifyKey(k: string): string {
   213→  return k
   214→    .replace(/([a-z0-9])([A-Z])/g, "$1 $2")
   215→    .replace(/_/g, " ")
   216→    .toLowerCase();
   217→}
   218→
   219→// Compact "key: value" chips (≤3) from a run-result object; nested
   220→// objects/arrays (e.g. retention windows, details[]) are skipped.
   221→function formatLastResult(result?: Record<string, unknown>): string[] {
   222→  if (!result) return [];
   223→  const out: string[] = [];
   224→  for (const [k, v] of Object.entries(result)) {
   225→    if (out.length >= 3) break;
   226→    if (v == null || typeof v === "object") continue;
   227→    const value = typeof v === "number" ? v.toLocaleString("en-IN") : String(v);
   228→    out.push(`${prettifyKey(k)}: ${value}`);
   229→  }
   230→  return out;
   231→}

   233→function cardStatus(job: CronJob): CardStatus {
   234→  if (job.status === "running") return "running";
   235→  const last = job.history?.[0];
   236→  if (last && last.status !== "running") return last.status === "success" ? "success" : "failed";
   237→  return "idle";
   238→}

   240→// Jobs that SUSPEND subscribers get a two-step confirm before Run now.
   241→function needsConfirm(job: CronJob): boolean {
   242→  return /suspend|expir/i.test(`${job.type} ${job.description}`);
   243→}

   272→// ─── Job card ────────────────────────────────────────────────────

   274→function JobCard({ job, now }: { job: CronJob; now: number }) {
   275→  const queryClient = useQueryClient();
   276→  const meta = JOB_META[job.id] ?? JOB_META_FALLBACK;
   277→  const Icon = meta.icon;
   278→  const status = cardStatus(job);
   279→  const badge = STATUS_BADGE[status];
   280→  const lastEntry = job.history?.[0];
   281→  const resultChips = useMemo(() => formatLastResult(lastEntry?.result), [lastEntry]);
   282→  const countdown = formatCountdown(job.nextRun, now);
   283→  const overdue = countdown === "due now";
   284→  const confirmRequired = needsConfirm(job);
   285→
   286→  // Two-step confirm state (suspend/expire jobs)
   287→  const [confirming, setConfirming] = useState(false);
   288→  const confirmTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   289→  useEffect(() => () => {
   290→    if (confirmTimer.current) clearTimeout(confirmTimer.current);
   291→  }, []);
   292→
   293→  // History panel state
   294→  const [historyOpen, setHistoryOpen] = useState(false);
   295→
   296→  const runMutation = useMutation({
   297→    mutationFn: (id: string) => apiFetch<RunResponse>(`/api/jobs/${id}/run?XTransformPort=3004`, { method: "POST" }),
   298→    onSuccess: (res) => {
   299→      toast.success(res.message || `Job "${job.name}" started`);
   300→      // Server-side execution is async — poll a few times so the card
   301→      // flips running → success/failed with fresh result + nextRun.
   302→      [1500, 5000, 10000].forEach((d) =>
   303→        setTimeout(() => queryClient.invalidateQueries({ queryKey: JOBS_KEY }), d)
   304→      );
   305→    },
   306→    onError: (err: Error) => {
   307→      if (err.message.includes("409")) {
   308→        toast.warning(`"${job.name}" is already running`);
   309→      } else if (err.message.includes("400")) {
   310→        toast.warning(`"${job.name}" is disabled — enable it to run manually`);
   311→      } else if (err.message.includes("401") || err.message.includes("403")) {
   312→        toast.error("Session expired or insufficient role — please re-login as admin");
   313→      } else {
   314→        toast.error(`Failed to trigger "${job.name}": ${err.message}`);
   315→      }
   316→    },
   317→  });
   318→
   319→  // Enable/disable — optimistic flip with rollback on failure. Nothing is
   320→  // persisted server-side (in-memory flag in billing-cron), reverts on restart.
   321→  const toggleMutation = useMutation({
   322→    mutationFn: (enabled: boolean) =>
   323→      apiFetch<{ success: boolean; job: { id: string; enabled: boolean } }>(
   324→        `/api/jobs/${job.id}?XTransformPort=3004`,
   325→        { method: "PATCH", body: JSON.stringify({ enabled }) }
   326→      ),
   327→    onMutate: async (enabled: boolean) => {
   328→      await queryClient.cancelQueries({ queryKey: JOBS_KEY });
   329→      const prev = queryClient.getQueryData<JobsResponse>(JOBS_KEY);
   330→      if (prev) {
   331→        queryClient.setQueryData<JobsResponse>(JOBS_KEY, {
   332→          ...prev,
   333→          jobs: prev.jobs.map((j) => (j.id === job.id ? { ...j, enabled } : j)),
   334→          summary: {
   335→            ...prev.summary,
   336→            enabled: Math.max(0, Math.min(prev.summary.total, prev.summary.enabled + (enabled ? 1 : -1))),
   337→          },
   338→        });
   339→      }
   340→      return { prev };
   341→    },
   342→    onSuccess: (_res, enabled) => {
   343→      toast.success(`${job.name} ${enabled ? "enabled" : "disabled"}`);
   344→    },
   345→    onError: (err: Error, _enabled, ctx) => {
   346→      if (ctx?.prev) queryClient.setQueryData(JOBS_KEY, ctx.prev);
   347→      if (err.message.includes("404")) toast.error("Job not found — was the service restarted?");
   348→      else if (err.message.includes("401") || err.message.includes("403")) toast.error("Session expired — please re-login as admin");
   349→      else toast.error(`Failed to update "${job.name}": ${err.message}`);
   350→    },
   351→    onSettled: () => queryClient.invalidateQueries({ queryKey: JOBS_KEY }),
   352→  });
   353→
   354→  const running = status === "running";
   355→  const history = (job.history ?? []).slice(0, 5);
   356→
   357→  const handleRunClick = () => {
   358→    if (confirmRequired && !confirming) {
   359→      setConfirming(true);
   360→      if (confirmTimer.current) clearTimeout(confirmTimer.current);
   361→      confirmTimer.current = setTimeout(() => setConfirming(false), 3000);
   362→      return;
   363→    }
   364→    setConfirming(false);
   365→    if (confirmTimer.current) clearTimeout(confirmTimer.current);
   366→    runMutation.mutate(job.id);
   367→  };
   368→
   369→  return (
   370→    <Card className={cn("flex flex-col transition-shadow hover:shadow-md", !job.enabled && "opacity-80")}>
   371→      <CardContent className="flex flex-1 flex-col gap-4 p-4 sm:p-5">
   372→        {/* Top row: identity + status */}
   373→        <div className="flex items-start gap-3">
   374→          <div
   375→            className={cn(
   376→              "flex h-10 w-10 shrink-0 items-center justify-center rounded-xl",
   377→              meta.chip
   378→            )}
   379→            aria-hidden="true"
   380→          >
   381→            <Icon className="h-5 w-5" />
   382→          </div>
   383→          <div className="min-w-0 flex-1">
   384→            <div className="flex items-center justify-between gap-2">
   385→              <h3 className="truncate text-sm font-semibold text-foreground" title={job.name}>
   386→                {job.id} · {job.name}
   387→              </h3>
   388→            </div>
   389→            <p className="mt-0.5 line-clamp-2 text-xs text-muted-foreground" title={job.description}>
   390→              {job.description}
   391→            </p>
   392→          </div>
   393→          <Badge variant="outline" className={cn("shrink-0 gap-1.5 font-medium", badge.className)}>
   394→            <StatusDot status={status} />
   395→            {badge.label}
   396→            <span className="sr-only">— last status of {job.name}</span>
   397→          </Badge>
   398→        </div>
   399→
   400→        {/* Schedule row */}
   401→        <div className="flex flex-wrap items-center gap-2 text-xs">
   402→          <code className="rounded-md border bg-muted/60 px-1.5 py-0.5 font-mono text-[11px] text-foreground">
   403→            {job.cron}
   404→          </code>
   405→          <span className="text-muted-foreground">{describeCron(job.cron)}</span>
   406→          <span className="text-muted-foreground/50" aria-hidden="true">·</span>
   407→          <span
   408→            className={cn(
   409→              "inline-flex items-center gap-1 font-medium",
   410→              overdue ? "text-amber-600 dark:text-amber-400" : "text-foreground"
   411→            )}
   412→            title={`Next run: ${new Date(job.nextRun).toLocaleString("en-IN")}${job.enabled ? "" : " (paused while disabled)"}`}
   413→          >
   414→            <CalendarClock className="h-3.5 w-3.5 text-muted-foreground" aria-hidden="true" />
   415→            Next run in {countdown}
   416→          </span>
   417→          {!job.enabled && (
   418→            <Badge variant="outline" className="border-transparent bg-muted text-[10px] font-medium uppercase tracking-wide text-muted-foreground">
   419→              Disabled
   420→            </Badge>
   421→          )}
   422→        </div>
   423→
   424→        {/* Stats row: 3 mini tiles */}
   425→        <div className="grid grid-cols-3 gap-2">
   426→          <div className="rounded-lg border bg-muted/30 px-2.5 py-2 min-w-0">
   427→            <p className="text-[11px] font-medium text-muted-foreground">Total runs</p>
   428→            <p className="truncate text-sm font-semibold text-foreground">{job.totalRuns.toLocaleString("en-IN")}</p>
   429→          </div>
   430→          <div className="rounded-lg border bg-muted/30 px-2.5 py-2 min-w-0">
   431→            <p className="text-[11px] font-medium text-muted-foreground">Success rate</p>
   432→            {(() => {
   433→              const rate = job.totalRuns === 0 ? null : Math.round((job.successCount / job.totalRuns) * 100);
   434→              return (
   435→                <p
   436→                  className={cn(
   437→                    "truncate text-sm font-semibold",
   438→                    rate == null
   439→                      ? "text-foreground"
   440→                      : rate >= 90
   441→                        ? "text-emerald-600 dark:text-emerald-400"
   442→                        : rate >= 70
   443→                          ? "text-amber-600 dark:text-amber-400"
   444→                          : "text-red-600 dark:text-red-400"
   445→                  )}
   446→                  title={`${job.successCount} succeeded · ${job.failCount} failed`}
   447→                >
   448→                  {rate == null ? "—" : `${rate}%`}
   449→                </p>
   450→              );
   451→            })()}
   452→          </div>
   453→          <div className="rounded-lg border bg-muted/30 px-2.5 py-2 min-w-0">
   454→            <p className="text-[11px] font-medium text-muted-foreground">Last result</p>
   455→            {resultChips.length > 0 ? (
   456→              <p className="truncate text-xs font-medium text-foreground" title={resultChips.join(" · ")}>
   457→                {resultChips[0]}
   458→                {resultChips.length > 1 && <span className="text-muted-foreground"> +{resultChips.length - 1}</span>}
   459→              </p>
   460→            ) : (
   461→              <p className="truncate text-xs text-muted-foreground">{running ? "running…" : "no runs yet"}</p>
   462→            )}
   463→          </div>
   464→        </div>
   465→        {resultChips.length > 1 && (
   466→          <p className="-mt-2 truncate text-[11px] text-muted-foreground">{resultChips.slice(1).join(" · ")}</p>
   467→        )}
   468→
   469→        {/* Expandable history */}
   470→        {historyOpen && (
   471→          <div className="rounded-lg border bg-muted/20">
   472→            <p className="border-b px-3 py-2 text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   473→              Recent runs
   474→            </p>
   475→            {history.length === 0 ? (
   476→              <p className="px-3 py-3 text-xs text-muted-foreground">No runs recorded yet.</p>
   477→            ) : (
   478→              <ul className="max-h-48 overflow-y-auto nice-scroll divide-y divide-border/60">
   479→                {history.map((entry) => (
