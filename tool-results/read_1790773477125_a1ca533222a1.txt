   106→async function ensurePolicerObject(name: string, cir: number, bc: number) {
   107→  const config = { cir, bc, conformAction: "transmit", violateAction: "drop" };
   108→  return db.vppPolicyObject.upsert({
   109→    where: { name },
   110→    create: {
   111→      name,
   112→      type: "POLICER",
   113→      configJson: JSON.stringify(config),
   114→      description: `auto-created by vpp-adapter for ${name}`,
   115→    },
   116→    update: {
   117→      type: "POLICER",
   118→      configJson: JSON.stringify(config),
   119→      isEnabled: true,
   120→    },
   121→  });
   122→}

   124→async function ensureAclObject(aclProfileId: string, rules: unknown[], defaultAction: string) {
   125→  const profile = await db.vppAclProfile.findUnique({ where: { id: aclProfileId } });
   126→  if (!profile) return null;
   127→  const name = `ACL-${profile.name}`;
   128→  const config = { rules, defaultAction, profileId: aclProfileId };
   129→  return db.vppPolicyObject.upsert({
   130→    where: { name },
   131→    create: {
   132→      name,
   133→      type: "ACL",
   134→      profileId: aclProfileId,
   135→      configJson: JSON.stringify(config),
   136→      description: `auto-created from VppAclProfile ${profile.name}`,
   137→    },
   138→    update: {
   139→      type: "ACL",
   140→      profileId: aclProfileId,
   141→      configJson: JSON.stringify(config),
   142→      isEnabled: true,
   143→    },
   144→  });
   145→}

   199→async function programVpp(s: ProgramInput): Promise<{
   200→  policer: any;
   201→  acl: any;
   202→  natMapping: { inside: string; outside: string; pool: string } | null;
   203→}> {
   204→  const speedDown = Math.max(0, Number(s.speedDownKbps) || 0);
   205→  const speedUp = Math.max(0, Number(s.speedUpKbps) || 0);
   206→  const framedIp = s.framedIp || "";
   207→
   208→  // 1. Policer (one per subscriber, named after sessionId)
   209→  const policerName = `policer-${s.sessionId}`;
   210→  const policerObj = await ensurePolicerObject(policerName, speedDown, 4096);
   211→
   212→  // 2. ACL assignment (only if aclProfileId provided)
   213→  let aclObj: any = null;
   214→  let aclAssignment: SubscriberPolicy["acl"] = null;
   215→  if (s.aclProfileId) {
   216→    const aclProfile = await db.vppAclProfile.findUnique({ where: { id: s.aclProfileId } });
   217→    if (aclProfile) {
   218→      const rules = JSON.parse(aclProfile.rulesJson || "[]");
   219→      aclObj = await ensureAclObject(s.aclProfileId, rules, aclProfile.defaultAction);
   220→      aclAssignment = { profileId: s.aclProfileId, rules };
   221→    }
   222→  }
   223→
   224→  // 3. NAT static mapping (framedIp → first NAT pool's publicIpStart)
   225→  let natMapping: SubscriberPolicy["natMapping"] = null;
   226→  if (framedIp) {
   227→    const pool = await pickNatPool();
   228→    const outside = pool?.publicIpStart || "203.0.113.100";
   229→    await ensureNatMappingObject(framedIp, outside, pool?.name || "default");
   230→    natMapping = {
   231→      inside: framedIp,
   232→      outside,
   233→      pool: pool?.name || "default",
   234→    };
   235→  }
   236→
   237→  // 4. Update in-memory mapping
   238→  const entry: SubscriberPolicy = {
   239→    sessionId: s.sessionId,
   240→    subscriberId: s.subscriberId || "",
   241→    username: s.username || "",
   242→    framedIp,
   243→    mac: s.mac || "",
   244→    nasIp: s.nasIp || "",
   245→    vlanId: s.vlanId,
   246→    vrf: s.vrf,
   247→    policyId: s.policyId,
   248→    aclProfileId: s.aclProfileId,
   249→    qosProfileId: s.qosProfileId,
   250→    natProfileId: s.natProfileId,
   251→    ipPool: s.ipPool,
   252→    speedDownKbps: speedDown,
   253→    speedUpKbps: speedUp,
   254→    timeoutSec: s.timeoutSec,
   255→    circuitId: s.circuitId,
   256→    remoteId: s.remoteId,
   257→    pppoeSessionId: s.pppoeSessionId,
   258→    dhcpClientId: s.dhcpClientId,
   259→    policer: { name: policerName, cir: speedDown, bc: 4096 },
   260→    acl: aclAssignment,
   261→    natMapping,
   262→    vppEpochApplied: inMemory.vppEpoch,
   263→    programmedAt: new Date(),
   264→  };
   265→  inMemory.subscriberPolicies.set(s.sessionId, entry);
   266→
   267→  return { policer: policerObj, acl: aclObj, natMapping };
   268→}

   270→// ─── Upsert SessionSnapshot ────────────────────────────────────
   271→async function upsertSnapshot(s: ProgramInput, recoveryState: "PROGRAMMED" | "RECOVERING" | "VERIFIED" | "STALE" = "PROGRAMMED") {
   272→  const now = new Date();
   273→  const configJson = JSON.stringify({
   274→    speedDownKbps: s.speedDownKbps,
   275→    speedUpKbps: s.speedUpKbps,
   276→    mac: s.mac,
   277→    nasIp: s.nasIp,
   278→    programmedAt: now.toISOString(),
   279→  });
   280→  const data = {
   281→    subscriberId: s.subscriberId || "",
   282→    username: s.username || "",
   283→    nasIp: s.nasIp || "",
   284→    nasPort: s.nasPort || "",
   285→    framedIp: s.framedIp || "",
   286→    mac: s.mac || "",
   287→    vlan: s.vlanId || "",
   288→    vrf: s.vrf || "",
   289→    policyId: s.policyId || "",
   290→    aclProfileId: s.aclProfileId || "",
   291→    qosProfileId: s.qosProfileId || "",
   292→    natProfileId: s.natProfileId || "",
   293→    ipPool: s.ipPool || "",
   294→    circuitId: s.circuitId || "",
   295→    remoteId: s.remoteId || "",
   296→    pppoeSessionId: s.pppoeSessionId || "",
   297→    dhcpClientId: s.dhcpClientId || "",
   298→    speedDownKbps: Math.max(0, Number(s.speedDownKbps) || 0),
   299→    speedUpKbps: Math.max(0, Number(s.speedUpKbps) || 0),
   300→    timeoutSec: Number(s.timeoutSec) || 0,
   301→    vppEpoch: inMemory.vppEpoch,
   302→    vppProgrammedAt: recoveryState === "STALE" ? undefined : now,
   303→    vppRecoveryState: recoveryState,
   304→    configJson,
   305→  };
   306→  return db.sessionSnapshot.upsert({
   307→    where: { sessionId: s.sessionId },
   308→    create: { sessionId: s.sessionId, ...data },
   309→    update: data,
   310→  });
   311→}

   315→async function generateVPPConfig(): Promise<string> {
   316→  const lines: string[] = [];
   317→  lines.push("# ============================================================");
   318→  lines.push("# CRYPTSK Nexus — VPP Dataplane Configuration (auto-generated)");
   319→  lines.push(`# Generated: ${new Date().toISOString()}`);
   320→  lines.push(`# VPP Epoch: ${inMemory.vppEpoch}  VPP Connected: ${inMemory.vppConnected}`);
   321→  lines.push("# ============================================================");
   322→  lines.push("");
   323→
   324→  try {
   325→    // 1. Interfaces
   326→    lines.push("# ─── 1. Interfaces ──────────────────────────────────────");
   327→    const nasList = await db.nas.findMany({ orderBy: { nasname: "asc" }, take: 100 });
   328→    for (const nas of nasList) {
   329→      lines.push(`# NAS: ${nas.shortname} (${nas.nasname}, type=${nas.type})`);
   330→      lines.push(`set interface state ${nas.nasname} up`);
   331→    }
   332→    lines.push("");
   333→
   334→    // 2. NAT
   335→    lines.push("# ─── 2. NAT (Network Address Translation) ──────────────");
   336→    const activeSessions = await db.radacct.findMany({
   337→      where: { acctstoptime: null, framedipaddress: { not: null } },
   338→      take: 1000,
   339→    });
   340→    const pool = await pickNatPool();
   341→    if (activeSessions.length > 0) {
   342→      lines.push(`# ${activeSessions.length} active subscriber sessions — NAT entries`);
   343→      for (const s of activeSessions) {
   344→        const ip = s.framedipaddress;
   345→        if (!ip) continue;
   346→        const outside = pool?.publicIpStart || `203.0.113.${(parseInt(ip.split(".")[3] || "0") % 200) + 10}`;
   347→        lines.push(`nat44 add static address ${ip} -> ${outside}`);
   348→      }
   349→    } else {
   350→      lines.push("# No active subscriber sessions — no NAT entries needed");
   351→    }
   352→    lines.push("");
   353→
   354→    // 3. ACL
   355→    lines.push("# ─── 3. ACL (Access Control Lists) ──────────────────────");
   356→    const aclAttrs = await db.radgroupcheck.findMany({
   357→      where: { attribute: { in: ["Filter-Id", "NAS-Filter-Id"] } },
   358→      orderBy: { groupname: "asc" },
   359→    });
   360→    if (aclAttrs.length > 0) {
   361→      lines.push(`# ${aclAttrs.length} ACL rules from policy engine`);
   362→      for (const a of aclAttrs) {
   363→        lines.push(`# Group: ${a.groupname}, Filter: ${a.value}`);
   364→        lines.push(`acl add index 0 ipv4 permit src any dst any flow-hash 10 filter-id ${a.value}`);
   365→      }
   366→    } else {
   367→      lines.push("# No ACL rules defined — default permit");
   368→    }
   369→    lines.push("");
   370→
   371→    // 4. QoS
   372→    lines.push("# ─── 4. QoS (Bandwidth Policing) ────────────────────────");
   373→    const qosAttrs = await db.radgroupcheck.findMany({
   374→      where: { attribute: "Mikrotik-Rate-Limit" },
   375→      orderBy: { groupname: "asc" },
   376→    });
   377→    if (qosAttrs.length > 0) {
   378→      lines.push(`# ${qosAttrs.length} QoS policer rules`);
   379→      for (const q of qosAttrs) {
   380→        const rates = (q.value || "").split("/")[0];
   381→        lines.push(`policer add name ${q.groupname} cir ${rates} bc 4096 conform-action transmit violate-action drop`);
   382→      }
   383→    } else {
   384→      lines.push("# No QoS policer rules — no bandwidth enforcement");
   385→    }
   386→    lines.push("");
   387→
   388→    // 5. Routing
   389→    lines.push("# ─── 5. Routing ────────────────────────────────────────");
   390→    lines.push("# Default route (configure per your network topology)");
   391→    lines.push("ip route add 0.0.0.0/0 via 203.0.113.1");
   392→    lines.push("");
   393→
   394→    // 6. Summary
   395→    lines.push("# ─── Summary ───────────────────────────────────────────");
   396→    lines.push(`# NAS devices: ${nasList.length}`);
   397→    lines.push(`# Active sessions: ${activeSessions.length}`);
   398→    lines.push(`# ACL rules: ${aclAttrs.length}`);
   399→    lines.push(`# QoS policers: ${qosAttrs.length}`);
   400→    lines.push(`# In-memory subscriber policies: ${inMemory.subscriberPolicies.size}`);
   401→
   402→    stats.configsGenerated++;
   403→    stats.lastGenerateAt = Date.now();
   404→    return lines.join("\n");
   405→  } catch (err: any) {
   406→    stats.errors++;
   407→    return `# Error generating config: ${err.message}`;
   408→  }
   409→}

   411→async function generateSubscriberConfig(subscriberId: string): Promise<string> {
   412→  const lines: string[] = [];
   413→  try {
   414→    const sub = await db.subscriber.findUnique({
   415→      where: { id: subscriberId },
   416→      include: { Plan: true },
   417→    });
   418→    if (!sub) return "# Subscriber not found";
   419→
   420→    const ra = await db.radacct.findFirst({
   421→      where: { username: sub.serviceUsername || "", acctstoptime: null },
   422→    });
   423→
   424→    lines.push(`# ============================================================`);
   425→    lines.push(`# VPP Config for Subscriber: ${sub.serviceUsername || sub.id}`);
   426→    lines.push(`# ============================================================`);
   427→    lines.push("");
   428→    lines.push(`# RADIUS Username: ${sub.serviceUsername || "—"}`);
   429→    lines.push(`# Plan: ${sub.Plan?.name || "—"}`);
   430→    lines.push(`# Framed IP: ${ra?.framedipaddress || "—"}`);
   431→    lines.push(`# NAS IP: ${ra?.nasipaddress || "—"}`);
   432→    lines.push(`# MAC: ${ra?.callingstationid || "—"}`);
   433→    lines.push("");
   434→
   435→    const rg = await db.radusergroup.findFirst({
   436→      where: { username: sub.serviceUsername || "" },
   437→    });
   438→    if (rg) {
   439→      lines.push("# ─── Dataplane Objects ─────────────────────────────────");
   440→      const checks = await db.radgroupcheck.findMany({
   441→        where: { groupname: rg.groupname },
   442→      });
   443→      for (const c of checks) {
   444→        if (c.attribute === "Mikrotik-Rate-Limit") {
   445→          lines.push(`# QoS: ${c.value}`);
   446→          lines.push(`policer add name ${sub.serviceUsername} cir ${(c.value || "").split("/")[0]} bc 4096`);
   447→        }
   448→      }
   449→    }
   450→
   451→    if (ra?.framedipaddress) {
   452→      lines.push("");
   453→      lines.push("# ─── NAT ───────────────────────────────────────────────");
   454→      const pool = await pickNatPool();
   455→      const outside = pool?.publicIpStart || "203.0.113.100";
   456→      lines.push(`nat44 add static address ${ra.framedipaddress} -> ${outside}`);
   457→    }
   458→
   459→    return lines.join("\n");
   460→  } catch (err: any) {
   461→    return `# Error: ${err.message}`;
   462→  }
   463→}

   465→// ─── HTTP Server ────────────────────────────────────────────────
   466→async function main() {
   467→  // ── Startup: initialize epoch + recovery state ──
   468→  inMemory.vppConnected = fs.existsSync(VPP_SOCKET);
   469→
   470→  const last = await db.vppRecoveryLog.findFirst({
   471→    orderBy: { createdAt: "desc" },
   472→  });
   473→  if (last && last.newEpoch) {
   474→    inMemory.vppEpoch = (last.newEpoch || 0) + 1;
   475→  } else {
   476→    inMemory.vppEpoch = 1;
   477→  }
   478→  inMemory.vppLastRestartAt = new Date();
   479→
   480→  console.log(
   481→    `[vpp-adapter] startup complete — epoch=${inMemory.vppEpoch}, vppConnected=${inMemory.vppConnected}, lastRestartAt=${inMemory.vppLastRestartAt.toISOString()}`
   482→  );
   483→
   484→  Bun.serve({
   485→    port: PORT,
   486→    async fetch(req: Request): Promise<Response> {
   487→      const url = new URL(req.url);
   488→      const path = url.pathname;
   489→      const method = req.method;
   490→
   491→      // ── CORS preflight ──
   492→      if (method === "OPTIONS") {
   493→        return new Response(null, {
   494→          status: 204,
   495→          headers: {
   496→            "Access-Control-Allow-Origin": "*",
   497→            "Access-Control-Allow-Methods": "GET, POST, PUT, DELETE, OPTIONS",
   498→            "Access-Control-Allow-Headers": "Content-Type, Authorization",
   499→          },
   500→        });
   501→      }
   502→
   503→      try {
   504→        // ═══ 1. /health ═══
   505→        if (path === "/health" && method === "GET") {
   506→          return json({
   507→            status: "ok",
   508→            port: PORT,
   509→            uptime: Math.floor((Date.now() - startTime) / 1000),
   510→            vppConnected: inMemory.vppConnected,
   511→            vppEpoch: inMemory.vppEpoch,
   512→            vppLastRestartAt: inMemory.vppLastRestartAt,
   513→            stats,
   514→          });
   515→        }
   516→
   517→        // ═══ 2. /vpp/state ═══
   518→        if (path === "/vpp/state" && method === "GET") {
   519→          const grouped = await db.vppPolicyObject.groupBy({
   520→            by: ["type"],
   521→            _count: { _all: true },
   522→            where: { isEnabled: true },
   523→          });
   524→          const policyObjects: Record<string, number> = {
   525→            ACL: 0,
   526→            POLICER: 0,
   527→            NAT: 0,
   528→            VRF: 0,
   529→            QoS: 0,
   530→            CLASSIFICATION: 0,
   531→          };
   532→          for (const g of grouped) {
   533→            policyObjects[g.type] = g._count._all;
   534→          }
   535→          return json({
   536→            vppEpoch: inMemory.vppEpoch,
   537→            vppConnected: inMemory.vppConnected,
   538→            vppLastRestartAt: inMemory.vppLastRestartAt,
   539→            uptime: Math.floor((Date.now() - startTime) / 1000),
   540→            policyObjects,
   541→            subscribersProgrammed: inMemory.subscriberPolicies.size,
   542→            interfaces: [],
   543→            stats,
   544→          });
   545→        }
   546→
   547→        // ═══ 3. /vpp/epoch ═══
   548→        if (path === "/vpp/epoch" && method === "GET") {
   549→          return json({
   550→            epoch: inMemory.vppEpoch,
   551→            lastRestartAt: inMemory.vppLastRestartAt,
   552→            vppConnected: inMemory.vppConnected,
   553→          });
