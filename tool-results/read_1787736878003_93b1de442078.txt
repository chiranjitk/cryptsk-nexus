    21→interface ShellResult {
    22→  success: boolean;

    24→  stderr: string;

    28→async function runShell(
    29→  cmd: string,
    30→  args: string[],
    31→  timeoutMs = 10000
    32→): Promise<ShellResult> {
    33→  try {
    34→    const proc = Bun.spawn([cmd, ...args], {
    35→      stdout: "pipe",
    36→      stderr: "pipe",
    37→    });
    38→    const timeout = setTimeout(() => proc.kill(), timeoutMs);
    39→    const [stdout, stderr] = await Promise.all([
    40→      new Response(proc.stdout).text(),
    41→      new Response(proc.stderr).text(),
    42→    ]);
    43→    clearTimeout(timeout);
    44→    const exitCode = await proc.exited;
    45→    return { success: exitCode === 0, stdout, stderr, exitCode };
    46→  } catch (err: any) {
    47→    return {
    48→      success: false,
    49→      stdout: "",
    50→      stderr: err.message || "Command not found or failed",
    51→      exitCode: -1,
    52→    };
    53→  }
    54→}

   158→    log.warn("nft add element failed", { setName, element, stderr: result.stderr });

   169→async function nftAddRule(table: string, chain: string, rule: string): Promise<number> {
   170→  const result = await nftRun(["add", "rule", "inet", table, chain, ...rule.split(" ")]);
   171→  if (!result.success) {
   172→    log.warn("nft add rule failed", { table, chain, rule, stderr: result.stderr });
   173→    return -1;
   174→  }
   175→  // Get handle of last added rule
   176→  const listResult = await nftRun(["-a", "list", "chain", "inet", table, chain]);
   177→  if (listResult.success) {
   178→    const lines = listResult.stdout.split("\n");
   179→    for (let i = lines.length - 1; i >= 0; i--) {
   180→      const handleMatch = lines[i].match(/handle (\d+)/);
   181→      if (handleMatch) return parseInt(handleMatch[1]);
   182→    }
   183→  }
   184→  return 0;
   185→}

   203→async function nftInitTable(): Promise<{ success: boolean; output: string }> {
   204→  const fullScript = `
   205→table inet ips {
   206→  set blocked_ips { type ipv4_addr; flags timeout; }
   207→  set monitored_ips { type ipv4_addr; }
   208→  set whitelisted_ips { type ipv4_addr; }
   209→
   210→  chain input {
   211→    type filter hook input priority 0; policy accept;
   212→    ip saddr @blocked_ips drop
   213→  }
   214→
   215→  chain forward {
   216→    type filter hook forward priority 0; policy accept;
   217→    ip saddr @blocked_ips drop
   218→    ip daddr @blocked_ips drop
   219→  }
   220→}`;
   221→  // Flush existing table first
   222→  await nftRun(["flush", "table", "inet", NFT_TABLE]).catch(() => {});
   223→  await nftRun(["delete", "table", "inet", NFT_TABLE]).catch(() => {});
   224→  const result = await runShell("nft", ["-f", "-"], 5000);
   225→  // Use echo to pipe the script
   226→  const echoResult = await runShell("bash", ["-c", `echo '${fullScript}' | nft -f -`], 10000);
   227→  if (echoResult.success) {
   228→    log.info("nftables IPS table initialized successfully");
   229→    return { success: true, output: echoResult.stdout };
   230→  } else {
   231→    log.warn("nftables init failed (may need root)", { stderr: echoResult.stderr });
   232→    return { success: false, output: echoResult.stderr };
   233→  }
   234→}

   236→// ─── Block Duration to Seconds ────────────────────────────────────────────
   237→function blockDurationToSeconds(duration: string): number {
   238→  const map: Record<string, number> = {
   239→    TEMP_5M: 300,
   240→    TEMP_15M: 900,
   241→    TEMP_30M: 1800,
   242→    TEMP_1H: 3600,
   243→    TEMP_6H: 21600,
   244→    TEMP_24H: 86400,
   245→    PERMANENT: 0,
   246→  };
   247→  return map[duration] || 1800;
   248→}

   250→// ─── Auto-Block Engine ────────────────────────────────────────────────────
   251→async function autoBlockIp(
   252→  sourceIp: string,
   253→  reason: string,
   254→  action: string,
   255→  duration: string,
   256→  alertId: string
   257→): Promise<string | null> {
   258→  try {
   259→    const timeoutSec = blockDurationToSeconds(duration);
   260→    const expiresAt = timeoutSec > 0 ? new Date(Date.now() + timeoutSec * 1000) : null;
   261→
   262→    // Add to nftables
   263→    if (timeoutSec > 0) {
   264→      await nftAddElement(NFT_BLOCKED_SET, sourceIp, timeoutSec);
   265→    } else {
   266→      await nftAddElement(NFT_BLOCKED_SET, sourceIp);
   267→    }
   268→
   269→    // Create DB record
   270→    const blockRule = await prisma.ipsBlockRule.create({
   271→      data: {
   272→        sourceIp,
   273→        reason,
   274→        action: action as any,
   275→        duration: duration as any,
   276→        alertId,
   277→        expiresAt,
   278→        isActive: true,
   279→        createdBy: "system",
   280→      },
   281→    });
   282→
   283→    // Update alert
   284→    await prisma.ipsAlert.update({
   285→      where: { id: alertId },
   286→      data: {
   287→        actionTaken: `BLOCKED_${action}`,
   288→        blockRuleId: blockRule.id,
   289→      },
   290→    });
   291→
   292→    log.warn("AUTO-BLOCK: IP blocked", {
   293→      ip: sourceIp,
   294→      reason,
   295→      action,
   296→      duration,
   297→      blockRuleId: blockRule.id,
   298→    });
   299→
   300→    return blockRule.id;
   301→  } catch (err: any) {
   302→    log.error("autoBlockIp failed", { ip: sourceIp, error: String(err) });
   303→    return null;
   304→  }
   305→}

   307→// ─── Threat Score Action ──────────────────────────────────────────────────
   308→async function applyThreatScoreAction(ip: string, score: number, reason: string, eventType: string): Promise<void> {
   309→  if (score <= 20) {
   310→    // Monitor only
   311→    await nftAddElement(NFT_MONITORED_SET, ip).catch(() => {});
   312→    log.info(`Threat monitor: ${ip} score ${score} (${eventType})`);
   313→  } else if (score <= 40) {
   314→    // Rate limit + notify
   315→    await nftAddElement(NFT_MONITORED_SET, ip).catch(() => {});
   316→    log.warn(`Threat rate-limit: ${ip} score ${score} (${eventType})`);
   317→  } else if (score <= 70) {
   318→    // Temp block 30 min
   319→    const alert = await prisma.ipsAlert.findFirst({
   320→      where: { sourceIp: ip, eventType: eventType as any },
   321→      orderBy: { createdAt: "desc" },
   322→    });
   323→    if (alert) {
   324→      await autoBlockIp(ip, `Threat score ${score} - ${reason}`, "DROP", "TEMP_30M", alert.id);
   325→    }
   326→  } else {
   327→    // Permanent block
   328→    const alert = await prisma.ipsAlert.findFirst({
   329→      where: { sourceIp: ip, eventType: eventType as any },
   330→      orderBy: { createdAt: "desc" },
   331→    });
   332→    if (alert) {
   333→      await autoBlockIp(ip, `Critical threat score ${score} - ${reason}`, "DROP", "PERMANENT", alert.id);
   334→    }
   335→  }
   336→}

   338→// ─── Conntrack Parser ─────────────────────────────────────────────────────
   339→interface ConntrackEntry {
   340→  srcIp: string;
   341→  dstIp: string;
   342→  srcPort: number;
   343→  dstPort: number;
   344→  protocol: number;
   345→  state: string;
   346→  bytes: number;
   347→  packets: number;
   348→}
   349→
   350→function parseConntrackLine(line: string): ConntrackEntry | null {
   351→  try {
   352→    const srcMatch = line.match(/src=(\S+)/g);
   353→    const dstMatch = line.match(/dst=(\S+)/g);
   354→    const sportMatch = line.match(/sport=(\d+)/g);
   355→    const dportMatch = line.match(/dport=(\d+)/g);
   356→    const bytesMatch = line.match(/bytes=(\d+)/g);
   357→    const packetsMatch = line.match(/packets=(\d+)/g);
   358→    const protoMatch = line.match(/^\s*\w+\s+\d+\s+(\w+)/);
   359→    const stateMatch = line.match(/(\w+)\s+src=/);
   360→
   361→    if (!srcMatch || srcMatch.length < 2 || !dstMatch || dstMatch.length < 2) return null;
   362→
   363→    return {
   364→      srcIp: srcMatch[0].replace("src=", ""),
   365→      dstIp: dstMatch[0].replace("dst=", ""),
   366→      srcPort: sportMatch && sportMatch.length > 0 ? parseInt(sportMatch[0].replace("sport=", "")) : 0,
   367→      dstPort: dportMatch && dportMatch.length > 0 ? parseInt(dportMatch[0].replace("dport=", "")) : 0,
   368→      protocol: protoMatch ? (protoMatch[1] === "tcp" ? 6 : protoMatch[1] === "udp" ? 17 : protoMatch[1] === "icmp" ? 1 : 0) : 0,
   369→      state: stateMatch ? stateMatch[1] : "",
   370→      bytes: bytesMatch && bytesMatch.length > 1 ? parseInt(bytesMatch[1].replace("bytes=", "")) : 0,
   371→      packets: packetsMatch && packetsMatch.length > 1 ? parseInt(packetsMatch[1].replace("packets=", "")) : 0,
   372→    };
   373→  } catch {
   374→    return null;
   375→  }
   376→}

   378→async function readConntrack(): Promise<ConntrackEntry[]> {
   379→  // Try /proc/net/nf_conntrack first
   380→  try {
   381→    const result = await runShell("cat", ["/proc/net/nf_conntrack"], 3000);
   382→    if (result.success && result.stdout.trim()) {
   383→      const entries: ConntrackEntry[] = [];
   384→      for (const line of result.stdout.split("\n")) {
   385→        const entry = parseConntrackLine(line);
   386→        if (entry) entries.push(entry);
   387→      }
   388→      return entries;
   389→    }
   390→  } catch {}
   391→
   392→  // Fallback: conntrack -L command
   393→  try {
   394→    const result = await runShell("conntrack", ["-L"], 5000);
   395→    if (result.success && result.stdout.trim()) {
   396→      const entries: ConntrackEntry[] = [];
   397→      for (const line of result.stdout.split("\n")) {
   398→        const entry = parseConntrackLine(line);
   399→        if (entry) entries.push(entry);
   400→      }
   401→      return entries;
   402→    }
   403→  } catch {}
   404→
   405→  // Fallback: generate simulated data for demo
   406→  log.info("Using simulated conntrack data (no access to /proc/net/nf_conntrack or conntrack command)");
   407→  return generateSimulatedData();
   408→}

   526→// ─── Detection Rules Cache ────────────────────────────────────────────────
   527→let cachedRules: any[] = [];
   528→let rulesCacheTime = 0;
   529→
   530→async function getDetectionRules(): Promise<any[]> {
   531→  const now = Date.now();
   532→  if (cachedRules.length > 0 && now - rulesCacheTime < 30000) {
   533→    return cachedRules;
   534→  }
   535→  try {
   536→    const dbRules = await prisma.ipsDetectionRule.findMany({
   537→      where: { enabled: true },
   538→      orderBy: { sortOrder: "asc" },
   539→    });
   540→    if (dbRules.length > 0) {
   541→      cachedRules = dbRules;
   542→    } else if (cachedRules.length === 0) {
   543→      // Use default rules when DB has none
   544→      cachedRules = getDefaultRules();
   545→      log.info("Using default detection rules (no rules in DB)");
   546→    }
   547→    rulesCacheTime = now;
   548→  } catch (err) {
   549→    log.warn("Failed to load detection rules from DB", { error: String(err) });
   550→    if (cachedRules.length === 0) {
   551→      cachedRules = getDefaultRules();
   552→    }
   553→    rulesCacheTime = now;
   554→  }
   555→  return cachedRules;
   556→}

   573→// ─── Create Alert Helper ──────────────────────────────────────────────────
   574→async function createAlert(entry: {
   575→  eventType: string;
   576→  severity: string;
   577→  sourceIp: string;
   578→  sourceMac?: string;
   579→  destIp?: string;
   580→  destPort?: number;
   581→  protocol?: string;
   582→  pps?: number;
   583→  bps?: number;
   584→  connCount?: number;
   585→  threatScore: number;
   586→  ruleId?: string;
   587→  details?: string;
   588→}): Promise<any> {
   589→  try {
   590→    // Only set ruleId if it refers to a real DB rule (starts with 'c')
   591→    const ruleId = entry.ruleId && entry.ruleId.startsWith("c") ? entry.ruleId : null;
   592→
   593→    const alert = await prisma.ipsAlert.create({
   594→      data: {
   595→        eventType: entry.eventType as any,
   596→        severity: entry.severity as any,
   597→        status: "NEW",
   598→        sourceIp: entry.sourceIp,
   599→        sourceMac: entry.sourceMac || "",
   600→        destIp: entry.destIp || "",
   601→        destPort: entry.destPort || 0,
   602→        protocol: entry.protocol || "",
   603→        pps: entry.pps || 0,
   604→        bps: entry.bps || 0,
   605→        connCount: entry.connCount || 0,
   606→        threatScore: entry.threatScore,
   607→        ruleId,
   608→        details: entry.details || "",
   609→      },
   610→    });
   611→
   612→    log.info("ALERT", {
   613→      type: entry.eventType,
   614→      severity: entry.severity,
   615→      ip: entry.sourceIp,
   616→      score: entry.threatScore,
   617→      alertId: alert.id,
   618→    });
   619→
   620→    // Broadcast via WebSocket
   621→    broadcastAlert({ ...entry, id: alert.id, createdAt: alert.createdAt });
   622→
   623→    return alert;
   624→  } catch (err: any) {
   625→    log.error("createAlert failed", { error: String(err) });
   626→    return null;
   627→  }
   628→}

   630→// ─── Detection Engine ─────────────────────────────────────────────────────
   631→const MALWARE_C2_PORTS = new Set([4444, 5555, 6666, 6667, 8888, 31337, 12345]);

   633→async function runDetectionCycle() {
   634→  if (isShuttingDown) return;
   635→
   636→  const cycleStart = Date.now();
   637→  try {
   638→    const entries = await readConntrack();
   639→    const rules = await getDetectionRules();
   640→    const ruleMap = new Map(rules.map((r) => [r.eventType, r]));
   641→    const now = cycleStart;
   642→    const window60s = now - 60000;
   643→
   644→    // ─── 1. Port Scan Detection ─────────────────────────────────
   645→    const portScanRule = ruleMap.get("PORT_SCAN");
   646→    if (portScanRule) {
   647→      const tcpNewConns = entries.filter((e) => e.protocol === 6);
   648→      for (const entry of tcpNewConns) {
   649→        if (!portScanTracker.has(entry.srcIp)) {
   650→          portScanTracker.set(entry.srcIp, { ports: new Set(), timestamps: [] });
   651→        }
   652→        const tracker = portScanTracker.get(entry.srcIp)!;
   653→        tracker.ports.add(entry.dstPort);
   654→        tracker.timestamps.push(now);
   655→      }
   656→      // Clean old entries and check thresholds
   657→      for (const [ip, tracker] of portScanTracker) {
   658→        tracker.timestamps = tracker.timestamps.filter((t) => t > window60s);
   659→        if (tracker.timestamps.length < 2) {
   660→          tracker.ports.clear();
   661→        }
   662→        if (tracker.ports.size > (portScanRule.thresholdConn || 20)) {
   663→          const score = addThreatScore(ip, portScanRule.scoreImpact || 25);
   664→          await createAlert({
   665→            eventType: "PORT_SCAN",
   666→            severity: score > 40 ? "HIGH" : "MEDIUM",
   667→            sourceIp: ip,
   668→            destPort: 0,
   669→            protocol: "tcp",
   670→            connCount: tracker.ports.size,
   671→            threatScore: score,
   672→            ruleId: portScanRule.id,
   673→            details: `${tracker.ports.size} distinct ports scanned in 60s`,
   674→          });
   675→          if (portScanRule.autoBlock) {
   676→            const alert = await prisma.ipsAlert.findFirst({
   677→              where: { sourceIp: ip, eventType: "PORT_SCAN" as any },
   678→              orderBy: { createdAt: "desc" },
   679→            });
   680→            if (alert) {
   681→              await autoBlockIp(ip, "Port scan detected", portScanRule.blockAction, portScanRule.blockDuration, alert.id);
   682→            }
   683→          }
   684→          await applyThreatScoreAction(ip, score, "Port scan", "PORT_SCAN");
   685→          tracker.ports.clear();
   686→          tracker.timestamps = [];
   687→        }
   688→      }
   689→    }
   690→
   691→    // ─── 2. SYN Flood Detection ──────────────────────────────────
   692→    const synFloodRule = ruleMap.get("SYN_FLOOD");
   693→    if (synFloodRule) {
   694→      const synEntries = entries.filter((e) => e.protocol === 6 && (e.state === "SYN_SENT" || e.state === "SYN_RECV"));
   695→      for (const entry of synEntries) {
   696→        if (!synTracker.has(entry.srcIp)) {
   697→          synTracker.set(entry.srcIp, { count: 0, timestamps: [] });
   698→        }
   699→        synTracker.get(entry.srcIp)!.count++;
   700→        synTracker.get(entry.srcIp)!.timestamps.push(now);
   701→      }
   702→      for (const [ip, tracker] of synTracker) {
   703→        tracker.timestamps = tracker.timestamps.filter((t) => t > window60s);
   704→        tracker.count = tracker.timestamps.length;
   705→        if (tracker.count > (synFloodRule.thresholdPps || 1000)) {
   706→          const score = addThreatScore(ip, synFloodRule.scoreImpact || 40);
   707→          await createAlert({
   708→            eventType: "SYN_FLOOD",
   709→            severity: score > 40 ? "CRITICAL" : "HIGH",
   710→            sourceIp: ip,
   711→            destPort: 80,
   712→            protocol: "tcp",
   713→            pps: tracker.count,
   714→            connCount: tracker.count,
   715→            threatScore: score,
   716→            ruleId: synFloodRule.id,
   717→            details: `${tracker.count} SYN packets in window`,
   718→          });
   719→          if (synFloodRule.autoBlock) {
   720→            const alert = await prisma.ipsAlert.findFirst({
   721→              where: { sourceIp: ip, eventType: "SYN_FLOOD" as any },
   722→              orderBy: { createdAt: "desc" },
   723→            });
   724→            if (alert) {
   725→              await autoBlockIp(ip, "SYN flood detected", synFloodRule.blockAction, synFloodRule.blockDuration, alert.id);
   726→            }
   727→          }
   728→          await applyThreatScoreAction(ip, score, "SYN flood", "SYN_FLOOD");
   729→          tracker.count = 0;
   730→          tracker.timestamps = [];
   731→        }
   732→      }
   733→    }
   734→
   735→    // ─── 3. UDP Flood Detection ──────────────────────────────────

   737→    if (udpFloodRule) {

   773→        }

   819→
   820→    // ─── 5. Connection Flood ─────────────────────────────────────
   821→    const connFloodRule = ruleMap.get("CONNECTION_FLOOD");

   980→    const c2Rule = ruleMap.get("MALWARE_C2");
   981→    if (c2Rule) {
   982→      const c2Entries = entries.filter((e) => MALWARE_C2_PORTS.has(e.dstPort) && e.protocol === 6);

  1010→    // ─── 10. Brute Force ─────────────────────────────────────────
  1011→    const bfRule = ruleMap.get("BRUTE_FORCE");
  1012→    if (bfRule) {
  1013→      const synOnly = entries.filter((e) => e.protocol === 6 && e.state === "SYN_SENT");
  1014→      for (const entry of synOnly) {
  1015→        if (!bruteForceTracker.has(entry.srcIp)) {
  1016→          bruteForceTracker.set(entry.srcIp, { count: 0, timestamps: [] });
  1017→        }
  1018→        bruteForceTracker.get(entry.srcIp)!.count++;
  1019→        bruteForceTracker.get(entry.srcIp)!.timestamps.push(now);
  1020→      }
  1021→      for (const [ip, tracker] of bruteForceTracker) {
  1022→        tracker.timestamps = tracker.timestamps.filter((t) => t > window60s);
  1023→        tracker.count = tracker.timestamps.length;
  1024→        if (tracker.count > (bfRule.thresholdConn || 50)) {
  1025→          const score = addThreatScore(ip, bfRule.scoreImpact || 30);
  1026→          await createAlert({
  1027→            eventType: "BRUTE_FORCE",
  1028→            severity: score > 40 ? "HIGH" : "MEDIUM",
  1029→            sourceIp: ip,
  1030→            protocol: "tcp",
  1031→            connCount: tracker.count,
  1032→            threatScore: score,
  1033→            ruleId: bfRule.id,
  1034→            details: `${tracker.count} failed connection attempts in 60s`,
  1035→          });
  1036→          if (bfRule.autoBlock) {
  1037→            const alert = await prisma.ipsAlert.findFirst({
  1038→              where: { sourceIp: ip, eventType: "BRUTE_FORCE" as any },
  1039→              orderBy: { createdAt: "desc" },
  1040→            });
  1041→            if (alert) {
  1042→              await autoBlockIp(ip, "Brute force detected", bfRule.blockAction, bfRule.blockDuration, alert.id);
  1043→            }
  1044→          }
  1045→          await applyThreatScoreAction(ip, score, "Brute force", "BRUTE_FORCE");
  1046→          tracker.count = 0;
  1047→          tracker.timestamps = [];
  1048→        }
  1049→      }
  1050→    }
  1051→
  1052→    lastDetectionRun = new Date();
  1053→    log.debug("Detection cycle complete", {
  1054→      entriesProcessed: entries.length,
  1055→      duration: `${Date.now() - cycleStart}ms`,
  1056→    });
  1057→  } catch (err: any) {
  1058→    log.error("Detection cycle failed", { error: String(err) });
  1059→  }
  1060→}

  1062→// ─── Route Matcher ────────────────────────────────────────────────────────
  1063→function matchRoute(
  1064→  url: URL,
  1065→  method: string
  1066→): { handler: (req: Request, params: Record<string, string>) => Promise<Response>; params: Record<string, string> } | null {
  1067→  const path = url.pathname.replace(/\/+/g, "/");
  1068→
  1069→  // ─── Daemon Status ────────────────────────────────────────────
  1070→  if (method === "GET" && path === "/") {
  1071→    return { handler: handleGetStatus, params: {} };
  1072→  }
  1073→
  1074→  // ─── Alerts ───────────────────────────────────────────────────
  1075→  if (method === "GET" && path === "/alerts") {
  1076→    return { handler: handleGetAlerts, params: {} };
  1077→  }
  1078→  if (method === "GET" && path === "/alerts/stats") {
  1079→    return { handler: handleGetAlertStats, params: {} };
  1080→  }
  1081→  let m = path.match(/^\/alerts\/([^/]+)\/status$/);
  1082→  if (method === "PUT" && m) {
  1083→    return { handler: handleUpdateAlertStatus, params: { id: m[1] } };
  1084→  }
  1085→  if (method === "DELETE" && path === "/alerts") {
  1086→    return { handler: handleDeleteAlerts, params: {} };
  1087→  }
  1088→
  1089→  // ─── Detection Rules ──────────────────────────────────────────
  1090→  if (method === "GET" && path === "/rules") {
  1091→    return { handler: handleGetRules, params: {} };
  1092→  }
  1093→  if (method === "POST" && path === "/rules") {
  1094→    return { handler: handleCreateRule, params: {} };
  1095→  }
  1096→  m = path.match(/^\/rules\/([^/]+)$/);
  1097→  if (method === "PUT" && m) {
  1098→    return { handler: handleUpdateRule, params: { id: m[1] } };
  1099→  }
  1100→  if (method === "DELETE" && m) {
  1101→    return { handler: handleDeleteRule, params: { id: m[1] } };
  1102→  }
  1103→
  1104→  // ─── Block Rules ──────────────────────────────────────────────
  1105→  if (method === "GET" && path === "/block-rules") {
  1106→    return { handler: handleGetBlockRules, params: {} };
  1107→  }
  1108→  if (method === "POST" && path === "/block-rules") {
  1109→    return { handler: handleCreateBlockRule, params: {} };
  1110→  }
  1111→  m = path.match(/^\/block-rules\/unblock\/([^/]+)$/);
  1112→  if (method === "POST" && m) {
  1113→    return { handler: handleUnblockByIp, params: { ip: m[1] } };
  1114→  }
  1115→  m = path.match(/^\/block-rules\/([^/]+)$/);
  1116→  if (method === "DELETE" && m) {
  1117→    return { handler: handleDeleteBlockRule, params: { id: m[1] } };
  1118→  }
  1119→
  1120→  // ─── Threat Scores ────────────────────────────────────────────
  1121→  if (method === "GET" && path === "/threat-scores") {
  1122→    return { handler: handleGetThreatScores, params: {} };
  1123→  }
  1124→  if (method === "POST" && path === "/threat-scores/reset") {
  1125→    return { handler: handleResetThreatScores, params: {} };
  1126→  }
  1127→
  1128→  // ─── Top Offenders ────────────────────────────────────────────
  1129→  if (method === "GET" && path === "/top-offenders") {
  1130→    return { handler: handleGetTopOffenders, params: {} };
  1131→  }
  1132→
  1133→  // ─── nftables ─────────────────────────────────────────────────
  1134→  if (method === "POST" && path === "/nftables/init") {
  1135→    return { handler: handleNftablesInit, params: {} };
  1136→  }
  1137→
  1138→  return null;
  1139→}

  1528→      _max: { threatScore: true },
  1529→      orderBy: { _count: { id: "desc" } },
  1530→      take: 20,
  1531→    });
  1532→
  1533→    const offenders = alertCounts.map((g) => ({
  1534→      ip: g.sourceIp,

  1536→      maxThreatScore: g._max.threatScore || 0,

  1540→    return json({ success: true, data: offenders });
  1541→  } catch (err: any) {

  1572→async function main() {

  1578→
  1579→  // Start detection engine (every 5 seconds)
  1580→  detectionInterval = setInterval(runDetectionCycle, 5000);
  1581→
  1582→  // Start server
  1583→  const server = Bun.serve({
  1584→    port: SERVICE_PORT,
  1585→    fetch: async (req: Request) => {
  1586→      const url = new URL(req.url);
  1587→
  1588→      // CORS preflight
  1589→      if (req.method === "OPTIONS") {
  1590→        return new Response(null, { status: 204, headers: corsHeaders });
  1591→      }
  1592→
  1593→      // WebSocket upgrade
  1594→      if (url.pathname === "/ws") {
  1595→        if (server.upgrade(req)) return;
  1596→        return jsonErr("WebSocket upgrade failed", 500);
  1597→      }
  1598→
  1599→      // Route matching
  1600→      const matched = matchRoute(url, req.method);
  1601→      if (matched) {
  1602→        try {
  1603→          return await matched.handler(req, matched.params);
  1604→        } catch (err: any) {
  1605→          if (err.message?.includes("Unauthorized")) {
  1606→            return json({ success: false, error: "Unauthorized" }, 401);
  1607→          }
  1608→          log.error("Handler error", { path: url.pathname, error: String(err) });
  1609→          return jsonErr(err.message || "Internal server error", 500);
  1610→        }
  1611→      }
  1612→
  1613→      return jsonErr("Not found", 404);
  1614→    },
  1615→    websocket: {
  1616→      open(ws) {
  1617→        wsClients.add(ws);
  1618→        log.info("WebSocket client connected", { totalClients: wsClients.size });
  1619→      },
  1620→      close(ws) {
  1621→        wsClients.delete(ws);
  1622→        log.info("WebSocket client disconnected", { totalClients: wsClients.size });
  1623→      },
  1624→      message(ws, message) {
  1625→        // Echo back for keepalive
  1626→        if (message === "ping") {
  1627→          ws.send("pong");
  1628→        }
  1629→      },
  1630→    },
  1631→  });
  1632→
  1633→  log.info(`IPS Daemon listening on port ${SERVICE_PORT}`, {
  1634→    detectionInterval: "5s",
  1635→    nftablesTable: NFT_TABLE,
  1636→    simulatedData: true,
  1637→  });
  1638→
  1639→  // Keep process alive
  1640→  process.stdin.resume();
  1641→
  1642→  // Graceful shutdown
  1643→  const shutdown = async (signal: string) => {
  1644→    if (isShuttingDown) return;
  1645→    isShuttingDown = true;
  1646→    log.info(`Received ${signal}, shutting down...`);
  1647→
  1648→    if (detectionInterval) {
  1649→      clearInterval(detectionInterval);
  1650→      detectionInterval = null;
  1651→    }
  1652→
  1653→    // Close all WebSocket clients
  1654→    for (const client of wsClients) {
  1655→      try { client.close(); } catch {}
  1656→    }
  1657→    wsClients.clear();
  1658→
  1659→    await prisma.$disconnect();
  1660→    log.info("IPS Daemon stopped");
  1661→    process.exit(0);
  1662→  };
  1663→
  1664→  process.on("SIGTERM", () => shutdown("SIGTERM"));
  1665→  process.on("SIGINT", () => shutdown("SIGINT"));
  1666→}

  1670→  process.exit(1);
