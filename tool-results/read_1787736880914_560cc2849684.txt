    20→// ─── Shell Command Helper ─────────────────────────────────────────────────
    21→interface ShellResult {
    22→  success: boolean;
    23→  stdout: string;
    24→  stderr: string;
    25→  exitCode: number;
    26→}
    27→
    28→async function runShell(
    29→  cmd: string,
    30→  args: string[],
    31→  timeoutMs = 10000
    32→): Promise<ShellResult> {
    33→  try {
    34→    const proc = Bun.spawn([cmd, ...args], {
    35→      stdout: "pipe",
    36→      stderr: "pipe",
    37→    });
    38→    const timeout = setTimeout(() => proc.kill(), timeoutMs);
    39→    const [stdout, stderr] = await Promise.all([
    40→      new Response(proc.stdout).text(),
    41→      new Response(proc.stderr).text(),
    42→    ]);
    43→    clearTimeout(timeout);
    44→    const exitCode = await proc.exited;
    45→    return { success: exitCode === 0, stdout, stderr, exitCode };
    46→  } catch (err: any) {
    47→    return {
    48→      success: false,
    49→      stdout: "",
    50→      stderr: err.message || "Command not found or failed",
    51→      exitCode: -1,
    52→    };
    53→  }
    54→}

    56→// ─── JSON Helpers ─────────────────────────────────────────────────────────
    57→function json(data: unknown, status = 200) {
    58→  return new Response(JSON.stringify(data, (_, v) => typeof v === 'bigint' ? v.toString() : v), {
    59→    status,
    60→    headers: { "Content-Type": "application/json", ...corsHeaders },
    61→  });
    62→}
    63→
    64→function jsonErr(message: string, status = 400) {
    65→  return json({ success: false, error: message }, status);
    66→}

    68→// ─── Audit Logger ─────────────────────────────────────────────────────────
    69→async function auditLog(
    70→  userId: string,
    71→  action: string,
    72→  entity: string,
    73→  entityId: string,
    74→  details: Record<string, unknown> = {},
    75→  req?: Request
    76→) {
    77→  try {
    78→    await prisma.auditLog.create({
    79→      data: {
    80→        userId,
    81→        userName: "",
    82→        action,
    83→        entity,
    84→        entityId,
    85→        details: JSON.stringify(details, (_, v) => typeof v === 'bigint' ? v.toString() : v),
    86→        endpoint: req?.url || "",
    87→        method: req?.method || "",
    88→        ipAddress: req?.headers.get("x-forwarded-for") || "",
    89→        userAgent: req?.headers.get("user-agent") || "",
    90→      },
    91→    });
    92→  } catch (e) {
    93→    log.error("auditLog failed", { error: String(e) });
    94→  }
    95→}

   608→async function handleGetInterfaces(req: Request) {
   609→  const auth = requireAuth(req);
   610→  try {
   611→    // Run shell commands to discover interfaces
   612→    const [linkResult, addrResult] = await Promise.all([
   613→      runShell("ip", ["link", "show"]),
   614→      runShell("ip", ["addr", "show"]),
   615→    ]);
   616→
   617→    // Read /sys/class/net/ for physical interface info
   618→    let sysInterfaces: string[] = [];
   619→    try {
   620→      const lsResult = await runShell("ls", ["/sys/class/net/"]);
   621→      sysInterfaces = lsResult.stdout
   622→        .split("\n")
   623→        .map((s) => s.trim())
   624→        .filter(Boolean);
   625→    } catch {
   626→      // ignore
   627→    }
   628→
   629→    // Parse interfaces from ip output
   630→    const interfaces: Record<string, any> = {};
   631→
   632→    // Parse ip link show
   633→    const linkLines = linkResult.stdout.split("\n");
   634→    let currentIface = "";
   635→    for (const line of linkLines) {
   636→      const ifaceMatch = line.match(/^\d+:\s+([^:@]+)/);
   637→      if (ifaceMatch) {
   638→        currentIface = ifaceMatch[1].trim();
   639→        interfaces[currentIface] = { name: currentIface };
   640→      }
   641→      if (currentIface && interfaces[currentIface]) {
   642→        if (line.includes("link/ether")) {
   643→          const macMatch = line.match(/link\/ether\s+([0-9a-f:]+)/i);
   644→          if (macMatch) interfaces[currentIface].macAddress = macMatch[1];
   645→        }
   646→        const mtuMatch = line.match(/mtu\s+(\d+)/);
   647→        if (mtuMatch) interfaces[currentIface].mtu = parseInt(mtuMatch[1]);
   648→        const stateMatch = line.match(/state\s+(\w+)/);
   649→        if (stateMatch) {
   650→          interfaces[currentIface].carrierStatus =
   651→            stateMatch[1].toUpperCase() === "UP";
   652→        }
   653→      }
   654→    }
   655→
   656→    // Parse ip addr show for addresses
   657→    const addrLines = addrResult.stdout.split("\n");
   658→    currentIface = "";
   659→    const ifaceAddrs: Record<string, { ipv4: string[]; ipv6: string[] }> = {};
   660→    for (const line of addrLines) {
   661→      const ifaceMatch = line.match(/^\d+:\s+([^:@]+)/);
   662→      if (ifaceMatch) {
   663→        currentIface = ifaceMatch[1].trim();
   664→        if (!ifaceAddrs[currentIface])
   665→          ifaceAddrs[currentIface] = { ipv4: [], ipv6: [] };
   666→      }
   667→      if (currentIface && ifaceAddrs[currentIface]) {
   668→        const inetMatch = line.match(
   669→          /inet\s+(\S+)/
   670→        );
   671→        if (inetMatch) {
   672→          ifaceAddrs[currentIface].ipv4.push(inetMatch[1]);
   673→        }
   674→        const inet6Match = line.match(
   675→          /inet6\s+(\S+)/
   676→        );
   677→        if (inet6Match) {
   678→          ifaceAddrs[currentIface].ipv6.push(inet6Match[1]);
   679→        }
   680→      }
   681→    }
   682→
   683→    // Build final interface list
   684→    const result = [];
   685→    for (const [name, info] of Object.entries(interfaces)) {
   686→      const addrs = ifaceAddrs[name] || { ipv4: [], ipv6: [] };
   687→      const isPhysical = sysInterfaces.includes(name);
   688→      const ifaceData = {
   689→        name,
   690→        type: isPhysical ? "PHYSICAL" : "VIRTUAL",
   691→        macAddress: (info as any).macAddress || "",
   692→        mtu: (info as any).mtu || 1500,
   693→        carrierStatus: (info as any).carrierStatus || false,
   694→        ipv4Addresses: addrs.ipv4,
   695→        ipv6Addresses: addrs.ipv6,
   696→      };
   697→      result.push(ifaceData);
   698→
   699→      // Sync to DB
   700→      try {
   701→        await prisma.systemInterface.upsert({
   702→          where: { name },
   703→          update: {
   704→            macAddress: ifaceData.macAddress,
   705→            mtu: ifaceData.mtu,
   706→            carrierStatus: ifaceData.carrierStatus,
   707→            ipv4Address: addrs.ipv4[0]?.split("/")[0] || "",
   708→            ipv6Address: addrs.ipv6[0]?.split("/")[0] || "",
   709→          },
   710→          create: {
   711→            name,
   712→            type: isPhysical ? "PHYSICAL" : "VIRTUAL",
   713→            macAddress: ifaceData.macAddress,
   714→            mtu: ifaceData.mtu,
   715→            carrierStatus: ifaceData.carrierStatus,
   716→            ipv4Address: addrs.ipv4[0]?.split("/")[0] || "",
   717→            ipv6Address: addrs.ipv6[0]?.split("/")[0] || "",
   718→          },
   719→        });
   720→      } catch (dbErr) {
   721→        log.warn("Failed to sync interface to DB", { name, error: String(dbErr) });
   722→      }
   723→    }
   724→
   725→    return json({ success: true, data: result, total: result.length });
   726→  } catch (err: any) {
   727→    log.error("handleGetInterfaces failed", { error: String(err) });
   728→    return jsonErr("Failed to discover interfaces: " + String(err.message), 500);
   729→  }
   730→}

   732→async function handleUpdateInterface(req: Request, params: Record<string, string>) {
   733→  const auth = requireAuth(req);
   734→  try {
   735→    const body = await req.json();
   736→    const { role, enabled, ipv4Address, ipv4Netmask, ipv4Gateway, ipv4Dns } = body;
   737→    const iface = await prisma.systemInterface.findUnique({
   738→      where: { id: params.id },
   739→    });
   740→    if (!iface) return jsonErr("Interface not found", 404);
   741→
   742→    const updateData: any = {};
   743→    if (role !== undefined) updateData.role = role;
   744→    if (enabled !== undefined) updateData.enabled = enabled;
   745→    if (ipv4Address !== undefined) updateData.ipv4Address = ipv4Address;
   746→    if (ipv4Netmask !== undefined) updateData.ipv4Netmask = ipv4Netmask;
   747→    if (ipv4Gateway !== undefined) updateData.ipv4Gateway = ipv4Gateway;
   748→    if (ipv4Dns !== undefined) updateData.ipv4Dns = ipv4Dns;
   749→
   750→    // Apply system changes
   751→    if (enabled !== undefined) {
   752→      const cmd = enabled ? "up" : "down";
   753→      await runShell("ip", ["link", "set", iface.name, cmd]);
   754→    }
   755→    if (ipv4Address && ipv4Netmask) {
   756→      const cidr = `${ipv4Address}/${ipv4Netmask}`;
   757→      await runShell("ip", ["addr", "add", cidr, "dev", iface.name]);
   758→    }
   759→    if (ipv4Gateway) {
   760→      await runShell("ip", ["route", "add", "default", "via", ipv4Gateway, "dev", iface.name]);
   761→    }
   762→
   763→    const updated = await prisma.systemInterface.update({
   764→      where: { id: params.id },
   765→      data: updateData,
   766→    });
   767→
   768→    await auditLog(auth.userId, "UPDATE", "SystemInterface", params.id, updateData, req);
   769→    log.info("Interface updated", { name: iface.name, ...updateData });
   770→    return json({ success: true, data: updated });
   771→  } catch (err: any) {
   772→    return jsonErr(err.message || "Update failed", 500);
   773→  }
   774→}

  1396→// ─── dnsmasq Config Generation Helpers ─────────────────────────────────
  1397→async function regenerateDnsmasqConfig(): Promise<string> {
  1398→  const records = await prisma.dnsRecord.findMany({ where: { enabled: true } });
  1399→  const gwConfig = await prisma.gatewayConfig.findUnique({ where: { id: "default" } });
  1400→
  1401→  let config = `# Auto-generated by Cryptsk Gateway Service\n`;
  1402→  config += `# DO NOT EDIT MANUALLY\n\n`;
  1403→  config += `listen-address=127.0.0.1\n`;
  1404→  if (gwConfig?.enableDnsCache !== false) {
  1405→    config += `cache-size=1000\n`;
  1406→  }
  1407→  if (gwConfig?.dnsForwarders) {
  1408→    for (const dns of gwConfig.dnsForwarders.split(",")) {
  1409→      const trimmed = dns.trim();
  1410→      if (trimmed) config += `server=${trimmed}\n`;
  1411→    }
  1412→  }
  1413→  config += `\n# DNS Records\n`;
  1414→
  1415→  for (const rec of records) {
  1416→    switch (rec.type) {
  1417→      case "A":
  1418→        config += `address=/${rec.name}/${rec.value}\n`;
  1419→        break;
  1420→      case "AAAA":
  1421→        config += `address=/${rec.name}/${rec.value}\n`;
  1422→        break;
  1423→      case "CNAME":
  1424→        config += `cname=${rec.name},${rec.value}\n`;
  1425→        break;
  1426→      case "MX":
  1427→        config += `mx-host=${rec.name},${rec.value},${rec.priority || 10}\n`;
  1428→        break;
  1429→      case "TXT":
  1430→        config += `txt-record=${rec.name},${rec.value}\n`;
  1431→        break;
  1432→      case "SRV":
  1433→        config += `srv-host=${rec.name},${rec.value},${rec.priority || 0}\n`;
  1434→        break;
  1435→    }
  1436→  }
  1437→
  1438→  // Write config file
  1439→  const writeResult = await runShell("sh", [
  1440→    "-c", `mkdir -p /etc/dnsmasq.d && echo '${config.replace(/'/g, "'\"'\"'")}' > /etc/dnsmasq.d/cryptsk-hosts.conf`,
  1441→  ]);
  1442→  if (!writeResult.success) {
  1443→    log.warn("Failed to write dnsmasq config", { stderr: writeResult.stderr });
  1444→  }
  1445→
  1446→  return config;
  1447→}

  1594→async function handleGatewayInit(req: Request) {
  1595→  const auth = requireAuth(req);
  1596→  // Check gateway mode before executing gateway operations
  1597→  const gwSettings = await prisma.ispSettings.findFirst();
  1598→  if (!gwSettings?.gatewayModeEnabled) {
  1599→    return jsonErr("Gateway mode is not enabled");
  1600→  }
  1601→  try {
  1602→    const results: Record<string, any> = {};
  1603→
  1604→    // 1. Apply interface configs
  1605→    const interfaces = await prisma.systemInterface.findMany({ where: { enabled: true, isManaged: true } });
  1606→    for (const iface of interfaces) {
  1607→      await runShell("ip", ["link", "set", iface.name, "up"]);
  1608→      if (iface.ipv4Address && iface.ipv4Netmask) {
  1609→        await runShell("ip", ["addr", "add", `${iface.ipv4Address}/${iface.ipv4Netmask}`, "dev", iface.name]).catch(() => {});
  1610→      }
  1611→    }
  1612→    results.interfaces = { configured: interfaces.length };
  1613→
  1614→    // 2. Enable IP forwarding
  1615→    await runShell("sh", ["-c", "echo 1 > /proc/sys/net/ipv4/ip_forward"]).catch(() => {});
  1616→    await runShell("sh", ["-c", "echo 1 > /proc/sys/net/ipv6/conf/all/forwarding"]).catch(() => {});
  1617→
  1618→    // 3. Setup nftables basics
  1619→    await runShell("nft", ["flush", "ruleset"]).catch(() => {});
  1620→    await runShell("nft", ["add", "table", "inet", "cryptsk"]).catch(() => {});
  1621→    await runShell("nft", ["add", "chain", "inet", "cryptsk", "filter", "{ type filter hook forward priority 0 \\; policy drop \\;}"]).catch(() => {});
  1622→    await runShell("nft", ["add", "chain", "inet", "cryptsk", "nat", "{ type nat hook postrouting priority 100 \\;}"]).catch(() => {});
  1623→    // Allow established/related
  1624→    await runShell("nft", ["add", "rule", "inet", "cryptsk", "filter", "ct", "state", "established,related", "accept"]).catch(() => {});
  1625→    // MASQUERADE on WAN
  1626→    const wanIface = interfaces.find((i) => i.role === "WAN");
  1627→    if (wanIface) {
  1628→      await runShell("nft", ["add", "rule", "inet", "cryptsk", "nat", "outdev", wanIface.name, "masquerade"]).catch(() => {});
  1629→    }
  1630→    // Allow LAN forward
  1631→    const lanIface = interfaces.find((i) => i.role === "LAN");
  1632→    if (lanIface) {
  1633→      await runShell("nft", ["add", "rule", "inet", "cryptsk", "filter", "iifname", lanIface.name, "accept"]).catch(() => {});
  1634→    }
  1635→
  1636→    // 4. Setup TC on WAN
  1637→    if (wanIface) {
  1638→      await runShell("tc", ["qdisc", "add", "dev", wanIface.name, "root", "handle", "1:", "htb"]).catch(() => {});
  1639→      await runShell("tc", ["class", "add", "dev", wanIface.name, "parent", "1:", "classid", "1:1", "htb", "rate", "1gbit"]).catch(() => {});
  1640→      results.tc = { wanInterface: wanIface.name, initialized: true };
  1641→    }
  1642→
  1643→    results.status = "initialized";
  1644→    await auditLog(auth.userId, "INIT", "Gateway", "system", results, req);
  1645→    log.info("Gateway initialized", results);
  1646→    return json({ success: true, data: results });
  1647→  } catch (err: any) {
  1648→    return jsonErr("Gateway init failed: " + String(err.message), 500);
  1649→  }

  1765→async function applySingleNftRule(rule: any) {
  1766→  const criteria = JSON.parse(rule.matchCriteria || "{}");
  1767→  const chain = rule.chain || "forward";
  1768→  const base = ["nft", "add", "rule", "inet", "cryptsk", chain];
  1769→
  1770→  const parts: string[] = [];
  1771→
  1772→  if (criteria.protocol) parts.push("ip", "protocol", criteria.protocol);
  1773→  if (criteria.srcIp) parts.push("ip", "saddr", criteria.srcIp);
  1774→  if (criteria.dstIp) parts.push("ip", "daddr", criteria.dstIp);
  1775→  if (criteria.srcPort) parts.push(...(criteria.protocol === "tcp" || criteria.protocol === "udp" ? ["th", "sport"] : ["sport"]), criteria.srcPort);
  1776→  if (criteria.dstPort) parts.push(...(criteria.protocol === "tcp" || criteria.protocol === "udp" ? ["th", "dport"] : ["dport"]), criteria.dstPort);
  1777→  if (criteria.ifaceIn) parts.push("iifname", criteria.ifaceIn);
  1778→  if (criteria.ifaceOut) parts.push("oifname", criteria.ifaceOut);
  1779→
  1780→  // Action
  1781→  const action = (rule.action || "ACCEPT").toLowerCase();
  1782→  if (action === "snat" || action === "dnat") {
  1783→    if (rule.natTarget) {
  1784→      parts.push(action, rule.natTarget);
  1785→    }
  1786→  } else {
  1787→    parts.push(action);
  1788→  }
  1789→
  1790→  if (rule.logEnabled) {
  1791→    parts.push("log");
  1792→    if (rule.logPrefix) parts.push(`prefix "${rule.logPrefix}"`);
  1793→  }
  1794→
  1795→  await runShell(base[0], [...base.slice(1), ...parts]);
  1796→}
