   129→function formatBytes(bytes: number | string | undefined | null): string {
   130→  if (bytes === undefined || bytes === null) return "—";
   131→  const n = typeof bytes === "string" ? parseInt(bytes, 10) : bytes;
   132→  if (isNaN(n) || n === 0) return "0 B";
   133→  const units = ["B", "KB", "MB", "GB", "TB"];
   134→  const i = Math.floor(Math.log(n) / Math.log(1024));
   135→  return `${(n / Math.pow(1024, i)).toFixed(1)} ${units[i]}`;
   136→}

   138→function formatTime(iso: string | number | undefined | null): string {
   139→  if (!iso) return "—";
   140→  try {
   141→    const d = typeof iso === "number" ? new Date(iso) : new Date(iso);
   142→    if (isNaN(d.getTime())) return String(iso);
   143→    return d.toLocaleString(undefined, {
   144→      year: "numeric",
   145→      month: "short",
   146→      day: "2-digit",
   147→      hour: "2-digit",
   148→      minute: "2-digit",
   149→      second: "2-digit",
   150→    });
   151→  } catch {
   152→    return String(iso);
   153→  }
   154→}

   156→function formatDuration(ms: number | undefined | null): string {
   157→  if (ms === undefined || ms === null) return "—";
   158→  if (ms < 1000) return `${ms}ms`;
   159→  if (ms < 60000) return `${(ms / 1000).toFixed(2)}s`;
   160→  return `${(ms / 60000).toFixed(1)}m`;
   161→}

   166→}

   168→// Status badge colors (slate/emerald/amber/red — no indigo/blue)
   169→function statusBadgeVariant(
   170→  status: string | undefined | null
   171→): { className: string; variant: "default" | "secondary" | "destructive" | "outline" } {
   172→  const s = (status || "").toUpperCase();
   173→  if (["UP", "ACTIVE", "OK", "CONNECTED", "HEALTHY", "VERIFIED", "FRESH", "PROGRAMMED"].includes(s)) {
   174→    return { className: "bg-emerald-100 text-emerald-800 border-emerald-200", variant: "outline" };
   175→  }
   176→  if (["DOWN", "ERROR", "FAILED", "STALE", "DISCONNECTED"].includes(s)) {
   177→    return { className: "bg-red-100 text-red-800 border-red-200", variant: "outline" };
   178→  }
   179→  if (["RECOVERING", "PENDING", "WARNING", "WARN"].includes(s)) {
   180→    return { className: "bg-amber-100 text-amber-800 border-amber-200", variant: "outline" };
   181→  }
   182→  return { className: "bg-slate-100 text-slate-700 border-slate-200", variant: "outline" };
   183→}

   307→// Format adapter uptime (seconds → "5m 30s" / "1h 12m" / "45s")
   308→function formatUptime(seconds: number | string | undefined | null): string {
   309→  if (seconds === undefined || seconds === null) return "—";
   310→  const s = typeof seconds === "string" ? parseInt(seconds, 10) : seconds;
   311→  if (isNaN(s) || s < 0) return "—";
   312→  if (s < 60) return `${s}s`;
   313→  const m = Math.floor(s / 60);
   314→  const sec = s % 60;
   315→  if (m < 60) return `${m}m ${sec}s`;
   316→  const h = Math.floor(m / 60);
   317→  const mm = m % 60;
   318→  return `${h}h ${mm}m`;
   319→}

   321→// Tiny inline sparkline for StatCard trend display (recharts Area mini-chart)

   326→}: {
   327→  data: number[];

   329→  id: string;

   337→        <defs>
   338→          <linearGradient id={gradientId} x1="0" y1="0" x2="0" y2="1">
   339→            <stop offset="5%" stopColor={color} stopOpacity={0.4} />
   340→            <stop offset="95%" stopColor={color} stopOpacity={0} />

   344→          type="monotone"

   346→          stroke={color}

   366→// Enhanced StatCard: framer-motion hover lift + optional sparkline + optional delta arrow + tooltip
   367→function StatCard({
   368→  icon: Icon,
   369→  label,
   370→  value,
   371→  color,
   372→  loading,
   373→  sub,
   374→  trend,
   375→  delta,
   376→  tooltip,
   377→  index = 0,
   378→}: {
   379→  icon: React.ElementType;
   380→  label: string;
   381→  value: React.ReactNode;
   382→  color: string;
   383→  loading?: boolean;
   384→  sub?: React.ReactNode;
   385→  trend?: { value: number; label: string }[];
   386→  delta?: { value: number; positive: boolean };
   387→  tooltip?: string;
   388→  index?: number;
   389→}) {
   390→  const sparkColor = STAT_COLOR_HEX[color] || "#475569";
   391→  const hoverBorder = STAT_COLOR_BORDER[color] || "hover:border-slate-300";
   392→  const sparkId = React.useId();
   393→  const trendValues = trend?.map((t) => t.value) || [];
   394→
   395→  const card = (
   396→    <Card className={`transition-colors border ${hoverBorder} h-full`}>
   397→      <CardHeader className="pb-2">
   398→        <CardTitle className="text-sm flex items-center gap-2 font-medium text-muted-foreground">
   399→          <Icon className={`h-4 w-4 ${color}`} />
   400→          {label}
   401→          {tooltip && <Info className="h-3 w-3 text-muted-foreground/50 ml-auto" />}
   402→        </CardTitle>
   403→      </CardHeader>
   404→      <CardContent>
   405→        {loading ? (
   406→          <>
   407→            <Skeleton className="h-7 w-20 mb-2 animate-pulse" />
   408→            <Skeleton className="h-1.5 w-full animate-pulse" />
   409→          </>
   410→        ) : (
   411→          <div className="space-y-1.5">
   412→            <div className="flex items-baseline gap-2">

   421→                    <ArrowUp className="h-3 w-3" />
   422→                  ) : (
   423→                    <ArrowDown className="h-3 w-3" />
   424→                  )}
   425→                  {Math.abs(delta.value)}
   426→                </span>
   427→              )}
   428→            </div>
   429→            {sub && <div className="text-xs text-muted-foreground">{sub}</div>}

   431→              <div className="mt-2 -mb-1">

   433→              </div>
   434→            )}
   435→          </div>
   436→        )}
   437→      </CardContent>
   438→    </Card>
   439→  );
   440→
   441→  return (
   442→    <motion.div
   443→      initial={{ opacity: 0, y: 10 }}
   444→      animate={{ opacity: 1, y: 0 }}
   445→      transition={{ delay: index * 0.05, type: "spring", stiffness: 200 }}
   446→      className="h-full"
   447→    >
   448→      {tooltip ? (
   449→        <UITooltip>
   450→          <TooltipTrigger asChild>
   451→            <motion.div
   452→              whileHover={{ y: -2 }}
   453→              transition={{ type: "spring", stiffness: 200 }}
   454→              className="h-full"
   455→            >
   456→              {card}
   457→            </motion.div>
   458→          </TooltipTrigger>
   459→          <TooltipContent side="top">
   460→            <p className="max-w-[220px] text-xs">{tooltip}</p>
   461→          </TooltipContent>
   462→        </UITooltip>
   463→      ) : (
   464→        <motion.div
   465→          whileHover={{ y: -2 }}
   466→          transition={{ type: "spring", stiffness: 200 }}
   467→          className="h-full"
   468→        >
   469→          {card}
   470→        </motion.div>
   471→      )}
   472→    </motion.div>
   473→  );
   474→}
