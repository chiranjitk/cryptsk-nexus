    17→// ─── vtysh Path Detection ──────────────────────────────────────────────────
    18→
    19→interface VtyshInfo {
    20→  path: string;
    21→  portable: boolean;
    22→}
    23→
    24→let _cachedVtyshInfo: VtyshInfo | null = null;
    25→
    26→/**
    27→ * Auto-detect the vtysh binary path and whether it's a portable install.
    28→ * Checks in order: portable wrapper, `which vtysh`, /usr/bin/vtysh, /usr/sbin/vtysh, /usr/lib/frr/vtysh.
    29→ * Returns null when FRR is not installed.
    30→ */
    31→function getVtyshInfo(): VtyshInfo | null {
    32→  if (_cachedVtyshInfo !== null) return _cachedVtyshInfo;
    33→
    34→  const candidates: Array<{ path: string; portable: boolean }> = [
    35→    // Try portable install paths first
    36→    { path: PORTABLE_VTYSH_WRAPPER, portable: true },
    37→    { path: "/home/z/frr-portable/bin/vtysh", portable: true },
    38→    // Try `which` first (respects PATH) for system install
    39→    ...(() => {
    40→      try {
    41→        const out = execFileSync("which", ["vtysh"], {
    42→          timeout: 3000,
    43→          encoding: "utf-8",
    44→          stdio: ["pipe", "pipe", "pipe"],
    45→        }).trim();
    46→        return out ? [{ path: out, portable: false }] : [];
    47→      } catch {
    48→        return [];
    49→      }
    50→    })(),
    51→    { path: "/usr/bin/vtysh", portable: false },
    52→    { path: "/usr/sbin/vtysh", portable: false },
    53→    { path: "/usr/lib/frr/vtysh", portable: false },
    54→  ];
    55→
    56→  for (const { path, portable } of candidates) {
    57→    try {
    58→      if (existsSync(path)) {
    59→        // Verify it's executable by trying --version
    60→        const env = portable
    61→          ? {
    62→              ...process.env,
    63→              LD_LIBRARY_PATH: `${PORTABLE_FRR_HOME}/lib:${process.env.LD_LIBRARY_PATH || ""}`,
    64→            }
    65→          : process.env;
    66→        execFileSync(path, ["--version"], {
    67→          timeout: 3000,
    68→          env,
    69→        });
    70→        _cachedVtyshInfo = { path, portable };
    71→        return _cachedVtyshInfo;
    72→      }
    73→    } catch {
    74→      continue;
    75→    }
    76→  }
    77→
    78→  _cachedVtyshInfo = null;
    79→  return null;
    80→}

   105→/**
   106→ * Run a single vtysh command and return trimmed stdout.
   107→ * Returns empty string on any failure. Timeout: 3 seconds.
   108→ */
   109→function vtysh(command: string): string {
   110→  const info = getVtyshInfo();
   111→  if (!info) return "";
   112→  try {
   113→    const env = info.portable
   114→      ? {
   115→          ...process.env,
   116→          LD_LIBRARY_PATH: `${PORTABLE_FRR_HOME}/lib:${process.env.LD_LIBRARY_PATH || ""}`,
   117→        }
   118→      : process.env;
   119→    // Use execFileSync with argument array to prevent shell injection
   120→    return execFileSync(info.path, ["-c", command], {
   121→      timeout: 3000,
   122→      encoding: "utf-8",
   123→      env,
   124→    }).trim();
   125→  } catch {
   126→    return "";
   127→  }
   128→}

   131→ * Pipe multiple configuration lines into vtysh via printf.

   134→function vtyshConfigure(lines: string[]): boolean {
   135→  const info = getVtyshInfo();
   136→  if (!info) return false;
   137→  try {
   138→    const env = info.portable
   139→      ? {
   140→          ...process.env,
   141→          LD_LIBRARY_PATH: `${PORTABLE_FRR_HOME}/lib:${process.env.LD_LIBRARY_PATH || ""}`,
   142→        }
   143→      : process.env;
   144→    // Use execFileSync with argument array — no shell interpolation
   145→    execFileSync(info.path, lines, {
   146→      timeout: 3000,
   147→      encoding: "utf-8",
   148→      env,
   149→      stdio: ["pipe", "pipe", "pipe"],

   153→    console.error("[FRR] vtysh configure error:", err.message);
   154→    return false;
   155→  }

   171→/**
   172→ * Parse running daemons from `show running-config` header area.
   173→ * Looks for lines like "router bgp 65001", "router ospf", "router ospf6", "router rip", etc.
   174→ */
   175→function parseEnabledDaemons(output: string): string[] {
   176→  const daemons: string[] = [];
   177→  const patterns: [RegExp, string][] = [
   178→    [/^router\s+bgp\s+\d+/, "bgpd"],
   179→    [/^router\s+ospf\b/, "ospfd"],
   180→    [/^router\s+ospf6\b/, "ospf6d"],
   181→    [/^router\s+rip\b/, "ripd"],
   182→    [/^router\s+ripng\b/, "ripngd"],
   183→    [/^bfd\b/, "bfdd"],
   184→    [/^router\s+isis\b/, "isisd"],
   185→  ];
   186→  for (const line of output.split("\n")) {
   187→    const trimmed = line.trim();
   188→    for (const [rx, name] of patterns) {
   189→      if (rx.test(trimmed) && !daemons.includes(name)) {
   190→        daemons.push(name);
   191→      }
   192→    }
   193→  }
   194→  return daemons;
   195→}

   197→/**
   198→ * Extract router ID and hostname from `show running-config`.
   199→ */
   200→function parseRouterIdAndHostname(
   201→  output: string
   202→): { routerId: string; hostname: string } {
   203→  let routerId = "";
   204→  let hostname = "";
   205→  for (const line of output.split("\n")) {
   206→    const t = line.trim();
   207→    if (!routerId) {
   208→      const m = t.match(/frr\s+default\s+route-id\s+(\S+)/);
   209→      if (m) routerId = m[1];
   210→    }
   211→    if (!routerId) {
   212→      const m = t.match(/router-id\s+(\S+)/);
   213→      if (m) routerId = m[1];
   214→    }
   215→    if (!hostname) {
   216→      const m = t.match(/^hostname\s+(\S+)/);
   217→      if (m) hostname = m[1];
   218→    }
   219→  }
   220→  return { routerId, hostname };
   221→}

   225→/**
   226→ * Parse `show ip bgp summary` output.
   227→ *
   228→ * Typical output:
   229→ * BGP router identifier 1.2.3.4, local AS number 65001
   230→ * BGP table version is 42
   231→ * ...
   232→ * Neighbor        V    AS MsgRcvd MsgSent   TblVer  InQ OutQ Up/Down  State/PfxRcd
   233→ * 10.0.0.2        4 65002    1234    1234        0    0    0 01:23:45       123
   234→ * 192.168.1.1     4 65003       0       0        0    0    0 Never    Active
   235→ */
   236→function parseBgpSummary(
   237→  output: string
   238→): {
   239→  localAs: string;
   240→  routerId: string;
   241→  neighbors: Array<{
   242→    ip: string;
   243→    as: string;
   244→    state: string;
   245→    uptime: string;
   246→    prefixRx: number;
   247→    prefixTx: number;
   248→  }>;
   249→} {
   250→  let localAs = "";
   251→  let routerId = "";
   252→  const neighbors: Array<{
   253→    ip: string;
   254→    as: string;
   255→    state: string;
   256→    uptime: string;
   257→    prefixRx: number;
   258→    prefixTx: number;
   259→  }> = [];
   260→
   261→  const lines = output.split("\n");
   262→  for (const line of lines) {
   263→    const t = line.trim();
   264→
   265→    // Local AS and router ID from header
   266→    const asMatch = t.match(/local AS number\s+(\d+)/);
   267→    if (asMatch) localAs = asMatch[1];
   268→
   269→    const ridMatch = t.match(/BGP router identifier\s+(\S+)/);
   270→    if (ridMatch) routerId = ridMatch[1];
   271→
   272→    // Skip column headers and separator lines
   273→    if (
   274→      t.startsWith("Neighbor") ||
   275→      t.startsWith("---") ||
   276→      t.startsWith("BGP") ||
   277→      t === ""
   278→    ) {
   279→      continue;
   280→    }
   281→
   282→    // Data lines: columns separated by whitespace
   283→    // Columns: Neighbor, V, AS, MsgRcvd, MsgSent, TblVer, InQ, OutQ, Up/Down, State/PfxRcd
   284→    const parts = t.split(/\s+/);
   285→    if (parts.length >= 10) {
   286→      const ip = parts[0];
   287→      const as = parts[2];
   288→      const uptime = parts[8];
   289→      const stateOrPfx = parts.slice(9).join(" ");
   290→
   291→      // State can be "Idle", "Active", "OpenSent", etc. or a number (prefix count)
   292→      const stateNum = parseInt(stateOrPfx);
   293→      const isEstablished = !isNaN(stateNum);
   294→      const state = isEstablished ? "Established" : stateOrPfx;
   295→      const prefixRx = isEstablished ? stateNum : 0;
   296→
   297→      neighbors.push({
   298→        ip,
   299→        as,
   300→        state,
   301→        uptime: uptime === "Never" ? "" : uptime,
   302→        prefixRx,
   303→        prefixTx: 0,
   304→      });
   305→    }
   306→  }
   307→
   308→  return { localAs, routerId, neighbors };
   309→}

   311→/**
   312→ * Parse `show ip bgp` output for route summary counts.
   313→ * Returns { routesReceived, routesAccepted }.
   314→ */
   315→function parseBgpRoutes(output: string): {
   316→  routesReceived: number;
   317→  routesAccepted: number;
   318→} {
   319→  let routesReceived = 0;
   320→  let routesAccepted = 0;
   321→
   322→  if (!output) return { routesReceived: 0, routesAccepted: 0 };
   323→
   324→  // Count data lines (those starting with an IP/network prefix)
   325→  // Filter out header lines
   326→  const lines = output.split("\n");
   327→  for (const line of lines) {
   328→    const t = line.trim();
   329→    // BGP table lines start with a network prefix (e.g., "10.0.0.0/24")
   330→    if (/^\d+\.\d+\.\d+\.\d+/.test(t) || /^::/.test(t)) {
   331→      routesReceived++;
   332→    }
   333→  }
   334→  routesAccepted = routesReceived; // Accepted routes = displayed routes
   335→
   336→  return { routesReceived, routesAccepted };
   337→}

   339→/**
   340→ * Parse BGP neighbor detail from `show ip bgp neighbors <ip>` or aggregate output.
   341→ * We also parse `show ip bgp neighbors` to extract prefixTx (advertised).
   342→ * However, to keep things simpler, we use the summary for the main data
   343→ * and this function can be used for detailed info. For now it augments with advertised prefixes.
   344→ */
   345→function parseBgpNeighborDetail(output: string): Record<string, number> {
   346→  const result: Record<string, number> = {};
   347→  let currentNeighbor = "";
   348→
   349→  for (const line of output.split("\n")) {
   350→    const t = line.trim();
   351→
   352→    // Neighbor header line: "BGP neighbor is 10.0.0.2, remote AS 65002, ..."
   353→    const neighborMatch = t.match(
   354→      /BGP neighbor is (\S+),\s+remote AS (\d+)/
   355→    );
   356→    if (neighborMatch) {
   357→      currentNeighbor = neighborMatch[1];
   358→      continue;
   359→    }
   360→
   361→    // Prefixes advertised: "  Local Policy refused prefixes: 0"
   362→    // or "  Advertised to all peers (in all address families): X"
   363→    // Better: "  For address family: IPv4 Unicast"
   364→    //          "  Community: (standard) ..."
   365→    // Look for "Prefixes Suppressed:" or "Local Policy refused prefixes"
   366→    // Also look for "  Local host: ..." section
   367→    if (
   368→      currentNeighbor &&
   369→      (t.includes("advertised") || t.includes("Advertised"))
   370→    ) {
   371→      const advMatch = t.match(/(\d+)\s+advertised/i);
   372→      if (advMatch && !result[currentNeighbor]) {
   373→        result[currentNeighbor] = parseInt(advMatch[1]) || 0;
   374→      }
   375→    }
   376→
   377→    // Alternative: "  Last update: ..." followed by "  Prefixes Received:"
   378→    if (currentNeighbor && t.match(/Prefixes (Received|Advertised)/i)) {
   379→      // Format: "  Prefixes Received: 123" or "  Prefixes Advertised: 456"
   380→      const valMatch = t.match(/(\d+)/);
   381→      if (valMatch) {
   382→        if (t.toLowerCase().includes("advertised")) {
   383→          result[currentNeighbor] = parseInt(valMatch[1]) || 0;
   384→        }
   385→      }
   386→    }
   387→  }
   388→
   389→  return result;
   390→}

   392→/**
   393→ * Parse BGP networks from running-config output.
   394→ * Looks for "  network X.X.X.X/Y" within a "router bgp" block.
   395→ */
   396→function parseBgpNetworks(runningConfig: string): Array<{ prefix: string }> {
   397→  const networks: Array<{ prefix: string }> = [];
   398→  let inBgp = false;
   399→
   400→  for (const line of runningConfig.split("\n")) {
   401→    const t = line.trim();
   402→    if (t.startsWith("router bgp")) {
   403→      inBgp = true;
   404→      continue;
   405→    }
   406→    if (inBgp && t.startsWith("router ")) {
   407→      inBgp = false;
   408→      continue;
   409→    }
   410→    if (inBgp) {
   411→      const m = t.match(/^network\s+(\S+)/);
   412→      if (m) networks.push({ prefix: m[1] });
   413→    }
   414→  }
   415→
   416→  return networks;
   417→}

   419→// ── OSPF Parsers ───────────────────────────────────────────────────────────
   420→
   421→/**
   422→ * Parse `show ip ospf` output.
   423→ *
   424→ * Typical output:
   425→ * OSPF Routing Process, Router ID: 1.2.3.4
   426→ * ...
   427→ * Number of areas is 1, normal is 1, stub is 0, nssa is 0
   428→ * ...
   429→ * Area BACKBONE (0.0.0.0)
   430→ *   Number of interfaces in this area: 3
   431→ *   ...
   432→ * Area 0.0.0.1
   433→ *   Number of interfaces in this area: 1
   434→ */
   435→function parseOspfSummary(output: string): {
   436→  routerId: string;
   437→  areas: Array<{ id: string; interfaces: string[]; neighborCount: number }>;
   438→} {
   439→  let routerId = "";
   440→  const areas: Array<{ id: string; interfaces: string[]; neighborCount: number }> =
   441→    [];
   442→  let currentArea: { id: string; interfaces: string[]; neighborCount: number } | null =
   443→    null;
   444→
   445→  for (const line of output.split("\n")) {
   446→    const t = line.trim();
   447→
   448→    const ridMatch = t.match(/Router ID:\s*(\S+)/);
   449→    if (ridMatch) routerId = ridMatch[1];
   450→
   451→    // Area line: "Area BACKBONE (0.0.0.0)" or "Area 0.0.0.1"
   452→    const areaMatch = t.match(
   453→      /Area\s+(?:BACKBONE\s+)?\(([0-9.]+)\)|Area\s+([0-9.]+)/
   454→    );
   455→    if (areaMatch) {
   456→      if (currentArea) {
   457→        areas.push(currentArea);
   458→      }
   459→      const id = areaMatch[1] || areaMatch[2] || "0.0.0.0";
   460→      currentArea = { id, interfaces: [], neighborCount: 0 };
   461→      continue;
   462→    }
   463→
   464→    // Interfaces count within an area
   465→    if (currentArea && t.match(/Number of interfaces/i)) {
   466→      const intMatch = t.match(/(\d+)/);
   467→      if (intMatch) {
   468→        // We'll populate actual interface names from the interface command
   469→        currentArea.neighborCount = 0; // Will fill from neighbor data
   470→      }
   471→    }
   472→  }
   473→
   474→  if (currentArea) {
   475→    areas.push(currentArea);
   476→  }
   477→
   478→  return { routerId, areas };
   479→}

   481→/**
   482→ * Parse `show ip ospf neighbor` output.
   483→ * Returns map of areaId -> neighborCount.
   484→ */
   485→function parseOspfNeighbors(output: string): Record<string, number> {
   486→  const areaNeighborCount: Record<string, number> = {};
   487→  const lines = output.split("\n");
   488→
   489→  for (const line of lines) {
   490→    const t = line.trim();
   491→    if (
   492→      t.startsWith("Neighbor ID") ||
   493→      t.startsWith("---") ||
   494→      t === "" ||
   495→      t.startsWith("OSPF")
   496→    ) {
   497→      continue;
   498→    }
   499→    // Data lines: "2.3.4.5        10.0.0.2    1    Full/DR     00:01:23    eth0"
   500→    // Not easy to associate with areas from this output alone, we'll just count per interface
   501→  }
   502→
   503→  return areaNeighborCount;
   504→}

   506→/**
   507→ * Parse `show ip ospf interface` output.
   508→ *
   509→ * Typical:
   510→ * eth0 is up, line protocol is up
   511→ *   Internet Address 10.0.0.1/24, Area 0.0.0.0, MTU 1500
   512→ *   Router ID 1.2.3.4, Network Type BROADCAST, Cost: 10
   513→ *   Hello 10, Dead 40, Retransmit 5
   514→ * ...
   515→ */
   516→function parseOspfInterfaces(output: string): Array<{
   517→  name: string;
   518→  area: string;
   519→  state: string;
   520→  cost: number;
   521→  hello: number;
   522→  dead: number;
   523→}> {
   524→  const interfaces: Array<{
   525→    name: string;
   526→    area: string;
   527→    state: string;
   528→    cost: number;
   529→    hello: number;
   530→    dead: number;
   531→  }> = [];
   532→
   533→  if (!output) return interfaces;
   534→
   535→  const lines = output.split("\n");
   536→  let current: {
   537→    name: string;
   538→    area: string;
   539→    state: string;
   540→    cost: number;
   541→    hello: number;
   542→    dead: number;
   543→  } | null = null;
   544→
   545→  for (const line of lines) {
   546→    const t = line.trim();
   547→
   548→    // Interface header: "eth0 is up" or "eth0 is down"
   549→    const ifaceMatch = t.match(/^(\S+)\s+is\s+(\S+)/);
   550→    if (ifaceMatch) {
   551→      if (current) {
   552→        interfaces.push(current);
   553→      }
   554→      current = {
   555→        name: ifaceMatch[1],
   556→        area: "",
   557→        state: ifaceMatch[2],
   558→        cost: 0,
   559→        hello: 10,
   560→        dead: 40,
   561→      };
   562→      continue;
   563→    }
   564→
   565→    if (!current) continue;
   566→
   567→    // Area: "Internet Address 10.0.0.1/24, Area 0.0.0.0, ..."
   568→    const areaMatch = t.match(/Area\s+([0-9.]+)/);
   569→    if (areaMatch) current.Area = areaMatch[1];
   570→
   571→    // Cost: "Cost: 10" or "Cost 10"
   572→    const costMatch = t.match(/Cost[:\s]+(\d+)/);
   573→    if (costMatch) current.cost = parseInt(costMatch[1]) || 0;
   574→
   575→    // Hello interval: "Hello 10"
   576→    const helloMatch = t.match(/Hello\s+(\d+)/);
   577→    if (helloMatch) current.hello = parseInt(helloMatch[1]) || 10;
   578→
   579→    // Dead interval: "Dead 40"
   580→    const deadMatch = t.match(/Dead\s+(\d+)/);
   581→    if (deadMatch) current.dead = parseInt(deadMatch[1]) || 40;
   582→  }
   583→
   584→  if (current) {
   585→    interfaces.push(current);
   586→  }
   587→
   588→  return interfaces;
   589→}

   591→/**
   592→ * Cross-reference OSPF areas with interface names.
   593→ */
   594→function buildOspfAreasWithInterfaces(
   595→  ospfSummary: ReturnType<typeof parseOspfSummary>,
   596→  ospfInterfaces: ReturnType<typeof parseOspfInterfaces>,
   597→  ospfNeighborOutput: string
   598→): Array<{ id: string; interfaces: string[]; neighborCount: number }> {
   599→  // Count neighbors per interface from neighbor output
   600→  const neighborIfaces = new Set<string>();
   601→  for (const line of ospfNeighborOutput.split("\n")) {
   602→    const t = line.trim();
   603→    if (
   604→      t.startsWith("Neighbor ID") ||
   605→      t.startsWith("---") ||
   606→      t === "" ||
   607→      t.startsWith("OSPF")
   608→    )
   609→      continue;
   610→    // "2.3.4.5        10.0.0.2    1    Full/DR     00:01:23    eth0"
   611→    const parts = t.split(/\s+/);
   612→    if (parts.length >= 7) {
   613→      neighborIfaces.add(parts[6]);
   614→    }
   615→  }
   616→
   617→  // Build area -> interfaces map
   618→  const areaInterfaceMap: Record<string, string[]> = {};
   619→  for (const iface of ospfInterfaces) {
   620→    if (!areaInterfaceMap[iface.Area]) {
   621→      areaInterfaceMap[iface.Area] = [];
   622→    }
   623→    areaInterfaceMap[iface.Area].push(iface.name);
   624→  }
   625→
   626→  // Count neighbors per area
   627→  const areaNeighborCount: Record<string, number> = {};
   628→  for (const iface of ospfInterfaces) {
   629→    if (neighborIfaces.has(iface.name)) {
   630→      areaNeighborCount[iface.Area] =
   631→        (areaNeighborCount[iface.Area] || 0) + 1;
   632→    }
   633→  }
   634→
   635→  return ospfSummary.areas.map((a) => ({
   636→    id: a.id,
   637→    interfaces: areaInterfaceMap[a.id] || [],
   638→    neighborCount: areaNeighborCount[a.id] || 0,
   639→  }));
   640→}
   641→
   642→// ── OSPFv3 (OSPF6) Parsers ────────────────────────────────────────────────
   643→
   644→/**
   645→ * Parse `show ipv6 ospf6 interface` output.
   646→ *
   647→ * Typical:
   648→ * eth0 is up, type BROADCAST
   649→ *   Interface ID: 3
   650→ *   Area ID: 0.0.0.0 (BACKBONE)
   651→ *   State: DR, Cost: 10
   652→ * ...
   653→ */
   654→function parseOspf6Interfaces(output: string): Array<{
   655→  name: string;
   656→  area: string;
   657→  state: string;
   658→}> {
   659→  const interfaces: Array<{ name: string; area: string; state: string }> = [];
   660→
   661→  if (!output) return interfaces;
   662→
   663→  const lines = output.split("\n");
   664→  let current: { name: string; area: string; state: string } | null = null;
   665→
   666→  for (const line of lines) {
   667→    const t = line.trim();
   668→
   669→    const ifaceMatch = t.match(/^(\S+)\s+is\s+(\S+)/);
   670→    if (ifaceMatch) {
   671→      if (current) interfaces.push(current);
   672→      current = { name: ifaceMatch[1], area: "", state: ifaceMatch[2] };
   673→      continue;
   674→    }
   675→
   676→    if (!current) continue;
   677→
   678→    // Area: "Area ID: 0.0.0.0"
   679→    const areaMatch = t.match(/Area ID:\s+([0-9.]+)/);
   680→    if (areaMatch) current.Area = areaMatch[1];
   681→
   682→    // State: "State: DR" or "State: BDR"
   683→    const stateMatch = t.match(/State:\s+(\S+)/);
   684→    if (stateMatch) current.state = stateMatch[1];
   685→  }
   686→
   687→  if (current) interfaces.push(current);
   688→  return interfaces;
   689→}

   692→ * Build OSPFv3 areas from interfaces.
   693→ */
   694→function buildOspf6Areas(
   695→  interfaces: Array<{ name: string; area: string; state: string }>
   696→): Array<{ id: string; interfaces: string[] }> {
   697→  const areaMap: Record<string, string[]> = {};
   698→  for (const iface of interfaces) {
   699→    if (!areaMap[iface.Area]) areaMap[iface.Area] = [];
   700→    areaMap[iface.Area].push(iface.name);
   701→  }
   702→  return Object.entries(areaMap).map(([id, ifaces]) => ({
   703→    id,
   704→    interfaces: ifaces,
   705→  }));
   706→}

   887→ */
   888→function parseBfdPeers(output: string): Array<{
   889→  neighbor: string;
   890→  interface: string;
   891→  state: string;
   892→  rxInterval: number;
   893→  txInterval: number;
   894→}> {
   895→  const peers: Array<{
   896→    neighbor: string;
   897→    interface: string;
   898→    state: string;
   899→    rxInterval: number;

   908→    if (
   909→      t === "" ||
   910→      t.startsWith("BFD") ||
   911→      t.toLowerCase().startsWith("peer") ||
   912→      t.startsWith("---")
   913→    )
   914→      continue;
   915→
   916→    // Try to parse data lines — columns vary by FRR version
   917→    const parts = t.split(/\s+/);

   925→      let txInterval = 0;

   928→      const numericParts = parts
   929→        .slice(3)
   930→        .map((p) => parseInt(p))
   931→        .filter((n) => !isNaN(n) && n >= 10 && n <= 100000);
   932→
   933→      if (numericParts.length >= 2) {
   934→        rxInterval = numericParts[numericParts.length - 2];
   935→        txInterval = numericParts[numericParts.length - 1];
   936→      } else if (numericParts.length === 1) {
   937→        txInterval = numericParts[0];
   938→        rxInterval = numericParts[0];
   939→      }
   940→
   941→      peers.push({ neighbor, interface: iface, state, rxInterval, txInterval });
   942→    }
   943→  }
   944→
   945→  return peers;

  1065→// ── Route Table Parser ─────────────────────────────────────────────────────
  1066→
  1067→/**
  1068→ * Parse `show ip route` output.
  1069→ *
  1070→ * Typical:
  1071→ * Codes: C - connected, S - static, R - RIP, B - BGP, O - OSPF, ...
  1072→ * ...
  1073→ * C>* 10.0.0.0/24 is directly connected, eth0
  1074→ * S>* 0.0.0.0/0 [1/0] via 192.168.1.1, eth0
  1075→ * B>* 172.16.0.0/16 [200/0] via 10.0.0.2, eth0, weight 1
  1076→ * O>* 192.168.100.0/24 [110/20] via 10.0.0.3, eth0, label 42
  1077→ */
  1078→function parseRouteTable(output: string): Array<{
  1079→  type: string;
  1080→  prefix: string;
  1081→  nextHop: string;
  1082→  interface: string;
  1083→  metric: number;
  1084→  age: string;
  1085→}> {
  1086→  const routes: Array<{
  1087→    type: string;
  1088→    prefix: string;
  1089→    nextHop: string;
  1090→    interface: string;
  1091→    metric: number;
  1092→    age: string;
  1093→  }> = [];
  1094→
  1095→  if (!output) return routes;
  1096→
  1097→  const codeMap: Record<string, string> = {
  1098→    C: "connected",
  1099→    S: "static",
  1100→    R: "rip",
  1101→    B: "bgp",
  1102→    O: "ospf",
  1103→    "O IA": "ospf-inter",
  1104→    "O E1": "ospf-external1",
  1105→    "O E2": "ospf-external2",
  1106→    K: "kernel",
  1107→    E: "ecmp",
  1108→    N: "nhrp",
  1109→    M: "ospf6",
  1110→  };
  1111→
  1112→  const lines = output.split("\n");
  1113→
  1114→  for (const line of lines) {
  1115→    const t = line.trim();
  1116→
  1117→    // Skip header/comment lines
  1118→    if (
  1119→      t === "" ||
  1120→      t.startsWith("Codes:") ||
  1121→      t.startsWith("Gateway") ||
  1122→      t.startsWith("---")
  1123→    )
  1124→      continue;
  1125→
  1126→    // Match route lines. Pattern: "C>* 10.0.0.0/24 is directly connected, eth0"
  1127→    // or "B>* 172.16.0.0/16 [200/0] via 10.0.0.2, eth0"
  1128→    const routeMatch = t.match(
  1129→      /^([A-Z][A-Z\s]?)\*?\s*>\s*(\S+)\s+(?:\[.+?\]\s+)?(?:via\s+(\S+),?\s+)?.*?(?:,\s*(\S+))?$/
  1130→    );
  1131→
  1132→    if (routeMatch) {
  1133→      let rawType = routeMatch[1].trim();
  1134→      const prefix = routeMatch[2];
  1135→      let nextHop = routeMatch[3] || "directly connected";
  1136→      const iface = routeMatch[4] || "";
  1137→
  1138→      // Parse metric from [admin/metric]
  1139→      let metric = 0;
  1140→      const metricMatch = t.match(/\[(\d+)\/(\d+)\]/);
  1141→      if (metricMatch) metric = parseInt(metricMatch[2]) || 0;
  1142→
  1143→      // Map code to friendly name
  1144→      const type =
  1145→        codeMap[rawType] || rawType.toLowerCase().replace(/\s+/g, "-");
  1146→
  1147→      // Extract age from the rest of the line (e.g., "00:05:23")
  1148→      let age = "";
  1149→      const ageMatch = t.match(/(\d{2}:\d{2}:\d{2})/);
  1150→      if (ageMatch) age = ageMatch[1];
  1151→
  1152→      routes.push({ type, prefix, nextHop, interface: iface, metric, age });
  1153→    }
  1154→  }
  1155→
  1156→  return routes;
  1157→}

  1159→// ── Route Maps Parser ──────────────────────────────────────────────────────

  1161→interface RouteMapEntry {
  1162→  name: string;
  1163→  action: string;
  1164→  sequence: number;
  1165→  matchRules: string[];
  1166→  setRules: string[];
  1167→}
  1168→
  1169→/**
  1170→ * Parse route-maps from `show running-config` output.
  1171→ * Looks for blocks like:
  1172→ *   route-map RM_SET_LOCAL permit 10
  1173→ *    match ip address PREFIX_LIST_1
  1174→ *    set local-preference 200
  1175→ */
  1176→function parseRouteMaps(runningConfig: string): RouteMapEntry[] {
  1177→  const entries: RouteMapEntry[] = [];
  1178→  const lines = runningConfig.split("\n");
  1179→
  1180→  for (const line of lines) {
  1181→    const t = line.trim();
  1182→    const rmMatch = t.match(/^route-map\s+(\S+)\s+(\S+)\s+(\d+)/);
  1183→    if (rmMatch) {
  1184→      const name = rmMatch[1];
  1185→      const action = rmMatch[2]; // permit or deny
  1186→      const sequence = parseInt(rmMatch[3]);
  1187→      const matchRules: string[] = [];
  1188→      const setRules: string[] = [];
  1189→
  1190→      // Collect subsequent match/set lines that belong to this route-map entry
  1191→      // Walk forward in the lines array
  1192→      const idx = lines.indexOf(line);
  1193→      for (let i = idx + 1; i < lines.length; i++) {
  1194→        const inner = lines[i].trim();
  1195→        // Stop if we hit another route-map, a blank line after indented lines,
  1196→        // or a non-indented line (new config section)
  1197→        if (inner === "") {
  1198→          if (matchRules.length > 0 || setRules.length > 0) break;
  1199→          continue;
  1200→        }
  1201→        if (
  1202→          inner.startsWith("route-map ") ||
  1203→          inner.startsWith("router ") ||
  1204→          inner.startsWith("interface ") ||
  1205→          inner.startsWith("ip ") ||
  1206→          inner.startsWith("ipv6 ") ||
  1207→          inner.startsWith("bfd ") ||
  1208→          inner.startsWith("access-list ") ||
  1209→          inner.startsWith("no ") ||
  1210→          inner.startsWith("line ") ||
  1211→          inner.startsWith("log ") ||
  1212→          inner.startsWith("hostname ") ||
  1213→          inner.startsWith("frr ") ||
  1214→          inner.startsWith("service ") ||
  1215→          inner.startsWith("password ") ||
  1216→          inner.startsWith("enable ") ||
  1217→          inner.startsWith("username ")
  1218→        ) {
  1219→          break;
  1220→        }
  1221→        if (inner.startsWith("match ")) {
  1222→          matchRules.push(inner);
  1223→        } else if (inner.startsWith("set ")) {
  1224→          setRules.push(inner);
  1225→        }
  1226→      }
  1227→
  1228→      entries.push({ name, action, sequence, matchRules, setRules });
  1229→    }
  1230→  }
  1231→
  1232→  return entries;
  1233→}
  1234→
  1235→// ── Prefix Lists Parser ────────────────────────────────────────────────────
  1236→
  1237→interface PrefixListEntry {
  1238→  name: string;
  1239→  seq: number;
  1240→  action: string; // permit or deny
  1241→  prefix: string;
  1242→  le?: number; // less-than-or-equal (max prefix length)
  1243→  ge?: number; // greater-than-or-equal (min prefix length)
  1244→}
  1245→
  1246→/**
  1247→ * Parse prefix-lists from `show running-config` output.
  1248→ * Looks for lines like:
  1249→ *   ip prefix-list PREFIX_LIST_1 seq 5 permit 10.0.0.0/8 le 24
  1250→ *   ip prefix-list PREFIX_LIST_1 seq 10 deny 0.0.0.0/0 le 32
  1251→ */
  1252→function parsePrefixLists(runningConfig: string): PrefixListEntry[] {
  1253→  const entries: PrefixListEntry[] = [];
  1254→
  1255→  for (const line of runningConfig.split("\n")) {
  1256→    const t = line.trim();
  1257→    // Match: ip prefix-list <name> seq <seq> <action> <prefix> [ge <n>] [le <n>]
  1258→    const match = t.match(
  1259→      /^ip prefix-list\s+(\S+)\s+seq\s+(\d+)\s+(\S+)\s+(\S+)(?:\s+ge\s+(\d+))?(?:\s+le\s+(\d+))?/
  1260→    );
  1261→    if (match) {
  1262→      entries.push({
  1263→        name: match[1],
  1264→        seq: parseInt(match[2]),
  1265→        action: match[3],
  1266→        prefix: match[4],
  1267→        ge: match[5] ? parseInt(match[5]) : undefined,
  1268→        le: match[6] ? parseInt(match[6]) : undefined,
  1269→      });
  1270→    }
  1271→  }
  1272→
  1273→  return entries;
  1274→}

  1276→// ── Access Lists Parser ────────────────────────────────────────────────────
  1277→
  1278→interface AccessListEntry {
  1279→  name: string;
  1280→  type: string; // standard, extended
  1281→  rules: Array<{
  1282→    seq: number;
  1283→    action: string; // permit or deny
  1284→    source: string;
  1285→    destination?: string;
  1286→    protocol?: string;
  1287→  }>;
  1288→}
  1289→
  1290→/**
  1291→ * Parse access-lists from `show running-config` output.
  1292→ * Standard ACLs:
  1293→ *   access-list ACL_DENY deny 10.0.0.0/8
  1294→ *   access-list ACL_PERMIT permit any
  1295→ * Numbered ACLs:
  1296→ *   access-list 10 permit 10.0.0.0/8
  1297→ *   access-list 10 deny any
  1298→ */
  1299→function parseAccessLists(runningConfig: string): AccessListEntry[] {
  1300→  const map: Record<string, AccessListEntry> = {};
  1301→
  1302→  for (const line of runningConfig.split("\n")) {
  1303→    const t = line.trim();
  1304→
  1305→    // Standard named ACL: access-list <name> <action> <source>
  1306→    const namedMatch = t.match(
  1307→      /^access-list\s+(\S+)\s+(permit|deny)\s+(\S+)(?:\s+(\S+)(?:\s+(\S+))?)?/
  1308→    );
  1309→    if (namedMatch) {
  1310→      const name = namedMatch[1];
  1311→      const action = namedMatch[2];
  1312→      const source = namedMatch[3];
  1313→      // Check if there are additional fields (protocol, destination for extended)
  1314→      const maybeProtocol = namedMatch[4];
  1315→      const maybeDest = namedMatch[5];
  1316→
  1317→      if (!map[name]) {
  1318→        map[name] = { name, type: "standard", rules: [] };
  1319→      }
  1320→
  1321→      const rule: AccessListEntry["rules"][0] = {
  1322→        seq: map[name].rules.length + 10,
  1323→        action,
  1324→        source,
  1325→      };
  1326→
  1327→      // If we have protocol and dest, it's extended
  1328→      if (maybeProtocol && maybeProtocol !== "any" && maybeDest) {
  1329→        map[name].type = "extended";
  1330→        rule.protocol = maybeProtocol;
  1331→        rule.destination = maybeDest;
  1332→      }
  1333→
  1334→      map[name].rules.push(rule);
  1335→    }
  1336→  }
  1337→
  1338→  return Object.values(map);
  1339→}

  1604→// ─── FRR Service Management Helper ─────────────────────────────────────────
  1605→
  1606→/**
  1607→ * Execute an FRR service management command.
  1608→ * Returns { success, operation, status, message }.
  1609→ */
  1610→function manageFrrService(
  1611→  operation: string
  1612→): {
  1613→  success: boolean;
  1614→  operation: string;
  1615→  status: string;
  1616→  message: string;
  1617→} {
  1618→  const vtyshInfo = getVtyshInfo();
  1619→  const isPortable = vtyshInfo?.portable ?? existsSync(PORTABLE_VTYSH_WRAPPER);
  1620→
  1621→  const ALLOWED_OPS = ["start", "stop", "restart", "enable", "disable"];
  1622→  if (!ALLOWED_OPS.includes(operation)) {
  1623→    return { success: false, operation, status: "error", message: `Invalid operation: ${operation}` };
  1624→  }
  1625→
  1626→  try {
  1627→    const env = isPortable
  1628→      ? {
  1629→          ...process.env,
  1630→          LD_LIBRARY_PATH: `${PORTABLE_FRR_HOME}/lib:${process.env.LD_LIBRARY_PATH || ""}`,
  1631→        }
  1632→      : process.env;
  1633→
  1634→    if (isPortable) {
  1635→      if (!existsSync(PORTABLE_FRR_WRAPPER)) {
  1636→        return {
  1637→          success: false,
  1638→          operation,
  1639→          status: "error",
  1640→          message: `Portable FRR wrapper not found at ${PORTABLE_FRR_WRAPPER}`,
  1641→        };
  1642→      }
  1643→      execFileSync(PORTABLE_FRR_WRAPPER, [operation], { timeout: 5000, encoding: "utf-8", env });
  1644→    } else {
  1645→      execFileSync("systemctl", [operation, "frr"], { timeout: 5000, encoding: "utf-8", env });
  1646→    }
  1647→
  1648→    // Determine current status after operation
  1649→    let currentStatus = "unknown";
  1650→    if (operation === "stop") {
  1651→      currentStatus = "stopped";
  1652→    } else if (operation === "start" || operation === "restart") {
  1653→      // Give daemons a moment to start
  1654→      const startCheck = Date.now();
  1655→      while (Date.now() - startCheck < 3000) {
  1656→        if (checkFrrDaemonsRunning()) {
  1657→          currentStatus = "running";
  1658→          break;
  1659→        }
  1660→        execFileSync("sleep", ["0.5"], { timeout: 1000 });
  1661→      }
  1662→      if (currentStatus === "unknown") currentStatus = "starting";
  1663→    }
  1664→
  1665→    return {
  1666→      success: true,
  1667→      operation,
  1668→      status: currentStatus,
  1669→      message: `FRR ${operation} successful (${isPortable ? "portable" : "systemd"})`,
  1670→    };
  1671→  } catch (err: any) {
  1672→    return {
  1673→      success: false,
  1674→      operation,
  1675→      status: checkFrrDaemonsRunning() ? "running" : "stopped",
  1676→      message: `FRR ${operation} failed: ${err.message}`,
  1677→    };
  1678→  }
  1679→}

  1681→// ─── POST Handler ──────────────────────────────────────────────────────────
  1682→
  1683→export async function POST(request: NextRequest) {
  1684→  try {
  1685→    try {
  1686→      await requireAuth(request);
  1687→    } catch (error: any) {
  1688→      if (error instanceof AuthError)
  1689→        return NextResponse.json(
  1690→          { error: error.message },
  1691→          { status: error.statusCode }
  1692→        );
  1693→    }
  1694→
  1695→    const vtyshPath = getVtyshPath();
  1696→    if (!vtyshPath) {
  1697→      return NextResponse.json(
  1698→        { error: "FRR is not installed on this system" },
  1699→        { status: 400 }
  1700→      );
  1701→    }
  1702→
  1703→    const body = await request.json();
  1704→    const { action, protocol, ...params } = body;
  1705→
  1706→    switch (action) {
  1707→      // ────────────────────── FRR Service Management ──────────────────────
  1708→      case "service-frr": {
  1709→        const { operation } = params;
  1710→        if (
  1711→          !operation ||
  1712→          !["start", "stop", "restart", "status"].includes(operation)
  1713→        ) {
  1714→          return NextResponse.json(
  1715→            { error: "operation must be start, stop, restart, or status" },
  1716→            { status: 400 }
  1717→          );
  1718→        }
  1719→
  1720→        if (operation === "status") {
  1721→          const running = checkFrrDaemonsRunning();
  1722→          return NextResponse.json({
  1723→            success: true,
  1724→            operation: "status",
  1725→            status: running ? "running" : "stopped",
  1726→            frrDaemonsRunning: running,
  1727→          });
  1728→        }
  1729→
  1730→        const result = manageFrrService(operation);
  1731→        return NextResponse.json(result);
  1732→      }
  1733→
  1734→      // ────────────────────── BGP ──────────────────────
  1735→      case "configure-bgp": {
  1736→        const { localAs, routerId, networks } = params;
  1737→        if (!localAs) {
  1738→          return NextResponse.json(
  1739→            { error: "localAs is required" },
  1740→            { status: 400 }
  1741→          );
  1742→        }
  1743→
  1744→        const lines: string[] = [
  1745→          "configure terminal",
  1746→          `router bgp ${localAs}`,
  1747→        ];
  1748→
  1749→        if (routerId) {
  1750→          lines.push(`  bgp router-id ${routerId}`);
  1751→        }
  1752→
  1753→        if (Array.isArray(networks)) {
  1754→          for (const net of networks) {
  1755→            if (net) lines.push(`  network ${net}`);
  1756→          }
  1757→        }
  1758→
  1759→        lines.push("  end", "write");
  1760→
  1761→        const ok = vtyshConfigure(lines);
  1762→        return ok
  1763→          ? NextResponse.json({
  1764→              success: true,
  1765→              message: `BGP configured for AS ${localAs}`,
  1766→            })
  1767→          : NextResponse.json(
  1768→              { error: "Failed to configure BGP" },
  1769→              { status: 500 }
  1770→            );
  1771→      }
  1772→
  1773→      case "add-bgp-neighbor": {
  1774→        const { ip, remoteAs, description } = params;
  1775→        if (!ip || !remoteAs) {
  1776→          return NextResponse.json(
  1777→            { error: "ip and remoteAs are required" },
  1778→            { status: 400 }
  1779→          );
  1780→        }
  1781→
  1782→        const lines: string[] = [
  1783→          "configure terminal",
  1784→          "router bgp", // FRR will use the existing BGP process
  1785→          `  neighbor ${ip} remote-as ${remoteAs}`,
  1786→        ];
  1787→
  1788→        if (description) {
  1789→          lines.push(`  neighbor ${ip} description ${description}`);
  1790→        }
  1791→
  1792→        lines.push("  end", "write");
  1793→
  1794→        const ok = vtyshConfigure(lines);
  1795→        return ok
  1796→          ? NextResponse.json({
  1797→              success: true,
  1798→              message: `BGP neighbor ${ip} added`,
  1799→            })
  1800→          : NextResponse.json(
  1801→              { error: `Failed to add BGP neighbor ${ip}` },
  1802→              { status: 500 }
  1803→            );
  1804→      }
  1805→
  1806→      case "remove-bgp-neighbor": {
  1807→        const { ip } = params;
  1808→        if (!ip) {
  1809→          return NextResponse.json(
  1810→            { error: "ip is required" },
  1811→            { status: 400 }
  1812→          );
  1813→        }
  1814→
