   116→function timeAgo(dateStr: string | null | undefined): string {
   117→  if (!dateStr) return "—";
   118→  const diff = Date.now() - new Date(dateStr).getTime();
   119→  if (diff < 60000) return "Just now";
   120→  if (diff < 3600000) return `${Math.floor(diff / 60000)}m ago`;
   121→  if (diff < 86400000) return `${Math.floor(diff / 3600000)}h ago`;
   122→  if (diff < 2592000000) return `${Math.floor(diff / 86400000)}d ago`;
   123→  return `${Math.floor(diff / 2592000000)}mo ago`;
   124→}
   125→
   126→function formatDate(dateStr: string | null | undefined): string {
   127→  if (!dateStr) return "—";
   128→  try { return new Date(dateStr).toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" }); }
   129→  catch { return "—"; }
   130→}
   131→
   132→function formatDateTime(dateStr: string | null | undefined): string {
   133→  if (!dateStr) return "—";
   134→  try { return new Date(dateStr).toLocaleString("en-IN", { day: "2-digit", month: "short", year: "numeric", hour: "2-digit", minute: "2-digit" }); }
   135→  catch { return "—"; }
   136→}
   137→
   138→function truncate(str: string | null | undefined, len: number = 60): string {
   139→  if (!str) return "—";
   140→  return str.length > len ? str.slice(0, len) + "…" : str;
   141→}
   142→
   143→function getInitials(name: string): string {
   144→  return name.split(" ").map(n => n[0]).filter(Boolean).slice(0, 2).join("").toUpperCase() || "?";
   145→}

   147→function fileUrl(path: string): string {
