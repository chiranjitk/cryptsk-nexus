   488→export default function LoginPage() {
   489→  const { login, isLoggingIn, error, clearError, checkAuth, isAuthenticated } =
   490→    useAuthStore();
   491→  const { setUser } = useAppStore();
   492→  const abortControllerRef = useRef<AbortController | null>(null);
   493→
   494→  // Form state
   495→  const [email, setEmail] = useState("");
   496→  const [password, setPassword] = useState("");
   497→  const [showPassword, setShowPassword] = useState(false);
   498→  const [rememberMe, setRememberMe] = useState(false);
   499→  const [fieldErrors, setFieldErrors] = useState<{ email?: string; password?: string }>({});
   500→  const [shakeKey, setShakeKey] = useState(0);
   501→  const [networkRetry, setNetworkRetry] = useState(0);
   502→  const year = new Date().getFullYear().toString();
   503→
   504→  // Check auth on mount
   505→  useEffect(() => {
   506→    checkAuth();
   507→  }, [checkAuth]);
   508→
   509→  // Cleanup abort controller on unmount
   510→  useEffect(() => {
   511→    return () => {
   512→      abortControllerRef.current?.abort();
   513→    };
   514→  }, []);
   515→
   516→  // If already authenticated, skip showing login (parent handles redirect)
   517→  useEffect(() => {
   518→    if (isAuthenticated) {
   519→      const user = useAuthStore.getState().user;
   520→      if (user) {
   521→        setUser(user);
   522→      }
   523→    }
   524→  }, [isAuthenticated, setUser]);
   525→
   526→  // Clear global error when user types
   527→  const handleEmailChange = useCallback(
   528→    (e: React.ChangeEvent<HTMLInputElement>) => {
   529→      setEmail(e.target.value);
   530→      if (error) clearError();
   531→      if (fieldErrors.email) setFieldErrors((prev) => ({ ...prev, email: undefined }));
   532→      if (networkRetry > 0) setNetworkRetry(0);
   533→    },
   534→    [error, clearError, fieldErrors.email, networkRetry]
   535→  );
   536→
   537→  const handlePasswordChange = useCallback(
   538→    (e: React.ChangeEvent<HTMLInputElement>) => {
   539→      setPassword(e.target.value);
   540→      if (error) clearError();
   541→      if (fieldErrors.password) setFieldErrors((prev) => ({ ...prev, password: undefined }));
   542→      if (networkRetry > 0) setNetworkRetry(0);
   543→    },
   544→    [error, clearError, fieldErrors.password, networkRetry]
   545→  );
   546→
   547→  // ── Welcome TTS after login ──────────────────────────────────────
   548→  const playWelcomeTTS = useCallback((userName?: string) => {
   549→    try {
   550→      const name = userName || "there";
   551→      const text = `Welcome back, ${name}. How can I help you today?`;
   552→      fetch("/api/voice/speak", {
   553→        method: "POST",
   554→        headers: { "Content-Type": "application/json" },
   555→        body: JSON.stringify({ text }),
   556→      })
   557→        .then((res) => {
   558→          if (!res.ok) throw new Error("TTS failed");
   559→          return res.blob();
   560→        })
   561→        .then((blob) => {
   562→          const url = URL.createObjectURL(blob);
   563→          const audio = new Audio(url);
   564→          audio.play().catch(() => {});
   565→          audio.onended = () => URL.revokeObjectURL(url);
   566→          audio.onerror = () => URL.revokeObjectURL(url);
   567→        })
   568→        .catch(() => {
   569→          // TTS failure is non-critical — silent fallback
   570→        });
   571→    } catch {
   572→      // Ignore TTS errors during login
   573→    }
   574→  }, []);
   575→
   576→  // Form submission with timeout & retry
   577→  const handleSubmit = async (e: FormEvent<HTMLFormElement>) => {
   578→    e.preventDefault();
   579→
   580→    // Abort any previous in-flight request
   581→    abortControllerRef.current?.abort();
   582→    const controller = new AbortController();
   583→    abortControllerRef.current = controller;
   584→
   585→    // Client-side validation
   586→    const errors: { email?: string; password?: string } = {};
   587→    if (!email.trim()) {
   588→      errors.email = "Email is required";
   589→    } else if (!isValidEmail(email.trim())) {
   590→      errors.email = "Please enter a valid email address";
   591→    }
   592→    if (!password.trim()) {
   593→      errors.password = "Password is required";
   594→    } else if (password.length < 6) {
   595→      errors.password = "Password must be at least 6 characters";
   596→    }
   597→
   598→    if (Object.keys(errors).length > 0) {
   599→      setFieldErrors(errors);
   600→      setShakeKey((prev) => prev + 1);
   601→      return;
   602→    }
   603→
   604→    setFieldErrors({});
   605→
   606→    // Attempt login with built-in timeout (15s) and auto-retry
   607→    const maxRetries = 1;
   608→    let attempt = 0;
   609→    let success = false;
   610→
   611→    while (attempt <= maxRetries && !success) {
   612→      attempt++;
   613→
   614→      if (attempt > 1) {
   615→        setNetworkRetry(attempt - 1);
   616→      }
   617→
   618→      try {
   619→        success = await login(email.trim(), password, { signal: controller.signal, timeout: 15000 });
   620→      } catch {
   621→        break;
   622→      }
   623→
   624→      if (!success) {
   625→        const storeError = useAuthStore.getState().error;
   626→        const isNetworkError =
   627→          storeError?.includes("network") ||
   628→          storeError?.includes("fetch") ||
   629→          storeError?.includes("timeout") ||
   630→          storeError?.includes("Failed to fetch");
   631→
   632→        if (!isNetworkError || attempt > maxRetries || controller.signal.aborted) {
   633→          break;
   634→        }
   635→
   636→        await new Promise((resolve) => setTimeout(resolve, 1500));
   637→      }
   638→    }
   639→
   640→    setNetworkRetry(0);
   641→
   642→    if (success) {
   643→      const user = useAuthStore.getState().user;
   644→      if (user) {
   645→        setUser(user);
   646→      }
   647→      // Show welcome notification using a custom element instead of sonner
   648→      // to guarantee readable text color in both light and dark modes
   649→      const welcomeEl = document.createElement("div");
   650→      welcomeEl.setAttribute("role", "status");
   651→      welcomeEl.setAttribute("aria-live", "polite");
   652→      welcomeEl.style.cssText = "position:fixed;bottom:24px;right:24px;z-index:999999999;padding:14px 18px;background:#FFFFFF;color:#111827;border:1px solid #E5E7EB;border-radius:10px;box-shadow:0 4px 16px rgba(0,0,0,0.12),0 1px 4px rgba(0,0,0,0.06);font-family:ui-sans-serif,system-ui,-apple-system,sans-serif;font-size:14px;max-width:380px;animation:toastSlideIn 0.35s ease forwards;cursor:pointer;";
   653→      welcomeEl.innerHTML = `<div style="display:flex;align-items:center;gap:8px;"><svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="#059669" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round"><path d="M22 11.08V12a10 10 0 1 1-5.93-9.14"/><polyline points="22 4 12 14.01 9 11.01"/></svg><div><div style="font-weight:600;font-size:14px;color:#111827;">Welcome back!</div><div style="font-size:13px;color:#4B5563;margin-top:2px;">Signed in as ${escapeHtml(user?.name || email)}</div></div></div>`;
   654→      welcomeEl.onclick = () => { welcomeEl.style.animation = "toastSlideOut 0.3s ease forwards"; setTimeout(() => welcomeEl.remove(), 300); };
   655→      document.body.appendChild(welcomeEl);
   656→      setTimeout(() => { if (welcomeEl.parentNode) { welcomeEl.style.animation = "toastSlideOut 0.3s ease forwards"; setTimeout(() => welcomeEl.remove(), 300); } }, 4000);
   657→
   658→      // Speak welcome message via TTS
   659→      playWelcomeTTS(user?.name);
   660→    } else {
   661→      setShakeKey((prev) => prev + 1);
   662→      const storeError = useAuthStore.getState().error;
   663→      toast.error("Login failed", {
   664→        description: storeError || "Invalid credentials",
   665→        style: {
   666→          background: "#FFFFFF",
   667→          color: "#111827",
   668→          border: "1px solid #FCA5A5",
   669→        },
   670→        descriptionClassName: "!text-gray-600",
   671→      });
   672→    }
   673→  };
   674→
   675→  // If already authenticated, render nothing (parent will show dashboard)
   676→  if (isAuthenticated) {
   677→    return null;
   678→  }
   679→
   680→  return (
   681→    <div className="fixed inset-0 z-50 flex flex-col justify-center items-center login-scanlines overflow-hidden">
   682→      {/* Background layers */}
   683→      <AnimatedBackground />
   684→      <LoginAnimations />
   685→      <StatusIndicator />
   686→
   687→      {/* Main login card container */}
   688→      <div className="relative z-10 w-full max-w-[440px] mx-3 sm:mx-4">
   689→        {/* Logo + Brand */}
   690→        <LoginLogo />
   691→
   692→        {/* Login Card */}
   693→        <div
   694→          key={shakeKey}
   695→          className={`${shakeKey > 0 ? "login-shake" : ""} login-fade-in`}
   696→          style={{ animationDelay: "0.2s" }}
   697→        >
   698→          <div className="login-animated-border rounded-2xl">
   699→            <div
   700→              className="rounded-2xl p-6 sm:p-8 border relative overflow-hidden login-glass-card"
   701→              style={{
   702→                background:
   703→                  "linear-gradient(180deg, rgba(15, 23, 42, 0.88) 0%, rgba(8, 12, 24, 0.94) 100%)",
   704→                borderColor: "rgba(220, 38, 38, 0.08)",
   705→              }}
   706→            >
   707→              {/* Loading progress bar */}
   708→              <LoginProgressBar visible={isLoggingIn} />
   709→
   710→              {/* Subtle card glow */}
   711→              <div
   712→                className="absolute inset-0 pointer-events-none rounded-2xl"
   713→                style={{
   714→                  boxShadow: "inset 0 0 80px rgba(220, 38, 38, 0.03)",
   715→                }}
   716→              />
   717→
   718→              {/* Top accent line */}
   719→              <div
   720→                className="absolute top-0 left-8 right-8 h-px"
   721→                style={{
   722→                  background:
   723→                    "linear-gradient(90deg, transparent 0%, rgba(220, 38, 38, 0.4) 50%, transparent 100%)",
   724→                }}
   725→              />
   726→
   727→              {/* Noise texture overlay for frosted glass */}
   728→              <div
   729→                className="absolute inset-0 pointer-events-none rounded-2xl opacity-[0.015]"

   731→                  backgroundImage: `url("data:image/svg+xml,%3Csvg viewBox='0 0 256 256' xmlns='http://www.w3.org/2000/svg'%3E%3Cfilter id='n'%3E%3CfeTurbulence type='fractalNoise' baseFrequency='0.9' numOctaves='4' stitchTiles='stitch'/%3E%3C/filter%3E%3Crect width='100%25' height='100%25' filter='url(%23n)'/%3E%3C/svg%3E")`,

   733→                  backgroundSize: "128px 128px",
   734→                }}

   737→              {/* Card header */}

   740→                  Access Your Control Panel

   742→                <p className="mt-1.5 text-sm text-slate-400">
   743→                  Authenticate to manage your ISP network
   744→                </p>
   745→              </div>
   746→
   747→              <form onSubmit={handleSubmit} className="space-y-5 relative z-[1]" noValidate>
   748→                {/* Global error */}
   749→                {(error || networkRetry > 0) && (
   750→                  <div
   751→                    className="flex items-start gap-2.5 rounded-xl border p-3.5 text-sm login-fade-in"
   752→                    style={{
   753→                      background: "rgba(220, 38, 38, 0.08)",
   754→                      borderColor: "rgba(220, 38, 38, 0.25)",
   755→                    }}
   756→                  >
   757→                    <div
   758→                      className="mt-0.5 w-5 h-5 rounded-full flex items-center justify-center flex-shrink-0"
