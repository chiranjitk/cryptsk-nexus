    64→    64→function formatBytes(bytes: number | string | undefined | null): string {
    65→    65→  if (bytes === undefined || bytes === null) return "—";
    66→    66→  const n = typeof bytes === "string" ? parseInt(bytes, 10) : bytes;
    67→    67→  if (isNaN(n) || n === 0) return "0 B";
    68→    68→  const units = ["B", "KB", "MB", "GB", "TB"];
    69→    69→  const i = Math.floor(Math.log(n) / Math.log(1024));
    70→    70→  return `${(n / Math.pow(1024, i)).toFixed(1)} ${units[i]}`;
    71→    71→}
    72→    72→
    73→    73→function formatTime(iso: string | number | undefined | null): string {
    74→    74→  if (!iso) return "—";
    75→    75→  try {
    76→    76→    const d = typeof iso === "number" ? new Date(iso) : new Date(iso);
    77→    77→    if (isNaN(d.getTime())) return String(iso);
    78→    78→    return d.toLocaleString(undefined, {
    79→    79→      year: "numeric",
    80→    80→      month: "short",
    81→    81→      day: "2-digit",
    82→    82→      hour: "2-digit",
    83→    83→      minute: "2-digit",
    84→    84→      second: "2-digit",
    85→    85→    });
    86→    86→  } catch {
    87→    87→    return String(iso);
    88→    88→  }
    89→    89→}
    90→    90→
    91→    91→function formatDuration(ms: number | undefined | null): string {
    92→    92→  if (ms === undefined || ms === null) return "—";
    93→    93→  if (ms < 1000) return `${ms}ms`;
    94→    94→  if (ms < 60000) return `${(ms / 1000).toFixed(2)}s`;
    95→    95→  return `${(ms / 60000).toFixed(1)}m`;
    96→    96→}

    98→    98→function truncate(s: string | undefined | null, n = 12): string {
    99→    99→  if (!s) return "—";
   100→   100→  return s.length > n ? `${s.slice(0, n)}…` : s;
   101→   101→}

   104→   104→function statusBadgeVariant(
   105→   105→  status: string | undefined | null
   106→   106→): { className: string; variant: "default" | "secondary" | "destructive" | "outline" } {
   107→   107→  const s = (status || "").toUpperCase();
   108→   108→  if (["UP", "ACTIVE", "OK", "CONNECTED", "HEALTHY", "VERIFIED", "FRESH", "PROGRAMMED"].includes(s)) {
   109→   109→    return { className: "bg-emerald-100 text-emerald-800 border-emerald-200", variant: "outline" };
   110→   110→  }
   111→   111→  if (["DOWN", "ERROR", "FAILED", "STALE", "DISCONNECTED"].includes(s)) {
   112→   112→    return { className: "bg-red-100 text-red-800 border-red-200", variant: "outline" };
   113→   113→  }
   114→   114→  if (["RECOVERING", "PENDING", "WARNING", "WARN"].includes(s)) {
   115→   115→    return { className: "bg-amber-100 text-amber-800 border-amber-200", variant: "outline" };
   116→   116→  }
   117→   117→  return { className: "bg-slate-100 text-slate-700 border-slate-200", variant: "outline" };
   118→   118→}

   120→   120→function riskBadge(level: string | undefined | null): React.ReactNode {
   121→   121→  const l = (level || "").toUpperCase();
   122→   122→  if (l === "CRITICAL") {
   123→   123→    return <Badge className="bg-red-600 text-white border-red-700 font-bold">CRITICAL</Badge>;
   124→   124→  }
   125→   125→  if (l === "HIGH") {
   126→   126→    return <Badge className="bg-red-100 text-red-800 border-red-200">HIGH</Badge>;
   127→   127→  }
   128→   128→  if (l === "MEDIUM") {
   129→   129→    return <Badge className="bg-amber-100 text-amber-800 border-amber-200">MEDIUM</Badge>;
   130→   130→  }
   131→   131→  if (l === "LOW") {
   132→   132→    return <Badge className="bg-emerald-100 text-emerald-800 border-emerald-200">LOW</Badge>;
   133→   133→  }
   134→   134→  return <Badge variant="secondary">{level || "UNKNOWN"}</Badge>;
   135→   135→}
