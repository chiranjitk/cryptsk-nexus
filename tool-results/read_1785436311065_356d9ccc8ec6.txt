    19→    19→function serializeRows(rows: Record<string, unknown>[]): Record<string, unknown>[] {
    20→    20→  return rows.map((row) => {
    21→    21→    const serialized: Record<string, unknown> = {};
    22→    22→    for (const [key, value] of Object.entries(row)) {
    23→    23→      serialized[key] = typeof value === "bigint" ? Number(value) : value;
    24→    24→    }
    25→    25→    return serialized;
    26→    26→  });
    27→    27→}

    60→    60→// ─── GET /api/aaa/groups ─────────────────────────────────────────────────────

    62→    62→export async function GET(req: NextRequest) {
    63→    63→  try {
    64→    64→    await requireAuth(req);
    65→    65→
    66→    66→    const { searchParams } = new URL(req.url);
    67→    67→    const search = searchParams.get("search")?.trim() || "";
    68→    68→    const page = Math.max(1, parseInt(searchParams.get("page") || "1"));
    69→    69→    const limit = Math.min(200, Math.max(1, parseInt(searchParams.get("limit") || "50")));
    70→    70→    const offset = (page - 1) * limit;
    71→    71→
    72→    72→    // Build WHERE clause for search
    73→    73→    const whereClause = search
    74→    74→      ? `WHERE gname ILIKE $1`
    75→    75→      : "";
    76→    76→    const params: unknown[] = search ? [`%${search}%`] : [];
    77→    77→
    78→    78→    // Main query: distinct groupnames from both tables + enrichment
    79→    79→    const [groupRows, totalResult] = await Promise.all([
    80→    80→      db.$queryRawUnsafe<Record<string, unknown>[]>(
    81→    81→        `SELECT * FROM (
    82→    82→          SELECT DISTINCT groupname AS gname FROM radgroupcheck
    83→    83→          UNION
    84→    84→          SELECT DISTINCT groupname AS gname FROM radgroupreply
    85→    85→        ) AS distinct_groups
    86→    86→        ${whereClause}
    87→    87→        ORDER BY gname ASC
    88→    88→        LIMIT ${limit} OFFSET ${offset}`,
    89→    89→        ...params
    90→    90→      ),
    91→    91→      db.$queryRawUnsafe<{ count: string }[]>(
    92→    92→        `SELECT COUNT(*)::text AS count FROM (
    93→    93→          SELECT DISTINCT groupname FROM radgroupcheck
    94→    94→          UNION
    95→    95→          SELECT DISTINCT groupname FROM radgroupreply
    96→    96→        ) AS all_groups
    97→    97→        ${whereClause}`,
    98→    98→        ...params
    99→    99→      ),
   100→   100→    ]);
   101→   101→
   102→   102→    const totalGroups = Number(totalResult[0]?.count || 0);
   103→   103→
   104→   104→    if (groupRows.length === 0) {
   105→   105→      return NextResponse.json({
   106→   106→        success: true,
   107→   107→        groups: [],
   108→   108→        pagination: { page, limit, total: totalGroups, totalPages: 0 },
   109→   109→        stats: { totalGroups: 0, totalUsersAcrossGroups: 0 },
   110→   110→      });
   111→   111→    }
   112→   112→
   113→   113→    // Fetch check attrs, reply attrs, user counts, and RadiusGroup enrichment for all groups
   114→   114→    const groupNames = groupRows.map((r) => r.gname as string);
   115→   115→
   116→   116→    // Check attributes
   117→   117→    const checkRows = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   118→   118→      `SELECT id, groupname, attribute, op, value
   119→   119→       FROM radgroupcheck
   120→   120→       WHERE groupname = ANY($1)
   121→   121→       ORDER BY groupname, attribute`,
   122→   122→      groupNames
   123→   123→    );
   124→   124→
   125→   125→    // Reply attributes
   126→   126→    const replyRows = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   127→   127→      `SELECT id, groupname, attribute, op, value
   128→   128→       FROM radgroupreply
   129→   129→       WHERE groupname = ANY($1)
   130→   130→       ORDER BY groupname, attribute`,
   131→   131→      groupNames
   132→   132→    );
   133→   133→
   134→   134→    // User counts from radusergroup
   135→   135→    const userCountRows = await db.$queryRawUnsafe<{ groupname: string; user_count: string }[]>(
   136→   136→      `SELECT groupname, COUNT(*)::text AS user_count
   137→   137→       FROM radusergroup
   138→   138→       WHERE groupname = ANY($1)
   139→   139→       GROUP BY groupname`,
   140→   140→      groupNames
   141→   141→    );
   142→   142→
   143→   143→    // RadiusGroup Prisma enrichment
   144→   144→    const enrichedGroups = await db.radiusGroup.findMany({
   145→   145→      where: { name: { in: groupNames } },
   146→   146→      select: {
   147→   147→        id: true,
   148→   148→        name: true,
   149→   149→        description: true,
   150→   150→        priority: true,
   151→   151→        speedLimitDown: true,
   152→   152→        speedLimitUp: true,
   153→   153→        dataLimit: true,
   154→   154→        sessionTimeout: true,
   155→   155→        framedIpv6Pool: true,
   156→   156→        delegatedIpv6PrefixPool: true,
   157→   157→      },
   158→   158→    });
   159→   159→
   160→   160→    // Build maps
   161→   161→    const checkMap = new Map<string, Record<string, unknown>[]>();
   162→   162→    for (const row of checkRows) {
   163→   163→      const gn = row.groupname as string;
   164→   164→      if (!checkMap.has(gn)) checkMap.set(gn, []);
   165→   165→      checkMap.get(gn)!.push(row);
   166→   166→    }
   167→   167→
   168→   168→    const replyMap = new Map<string, Record<string, unknown>[]>();
   169→   169→    for (const row of replyRows) {
   170→   170→      const gn = row.groupname as string;
   171→   171→      if (!replyMap.has(gn)) replyMap.set(gn, []);
   172→   172→      replyMap.get(gn)!.push(row);
   173→   173→    }
   174→   174→
   175→   175→    const userCountMap = new Map<string, number>();
   176→   176→    let totalUsersAcrossGroups = 0;
   177→   177→    for (const row of userCountRows) {
   178→   178→      const count = Number(row.user_count);
   179→   179→      userCountMap.set(row.groupname, count);
   180→   180→      totalUsersAcrossGroups += count;
   181→   181→    }
   182→   182→
   183→   183→    const enrichedMap = new Map<string, Record<string, unknown>>();
   184→   184→    for (const eg of enrichedGroups) {
   185→   185→      enrichedMap.set(eg.name, eg as unknown as Record<string, unknown>);
   186→   186→    }
   187→   187→
   188→   188→    // Assemble final groups array
   189→   189→    const groups = groupRows.map((row) => {
   190→   190→      const gname = row.gname as string;
   191→   191→      const checks = serializeRows(checkMap.get(gname) || []);
   192→   192→      const replies = serializeRows(replyMap.get(gname) || []);
   193→   193→      const userCount = userCountMap.get(gname) || 0;
   194→   194→      const enrichment = enrichedMap.get(gname) || null;
   195→   195→
   196→   196→      return {
   197→   197→        groupname: gname,
   198→   198→        checkAttrs: checks,
   199→   199→        replyAttrs: replies,
   200→   200→        userCount,
   201→   201→        checkCount: checks.length,
   202→   202→        replyCount: replies.length,
   203→   203→        enrichment,
   204→   204→        description: enrichment?.description || "",
   205→   205→        priority: enrichment?.priority || 0,
   206→   206→      };
   207→   207→    });
   208→   208→
   209→   209→    const totalPages = Math.ceil(totalGroups / limit);
   210→   210→
   211→   211→    return NextResponse.json({
   212→   212→      success: true,
   213→   213→      groups,
   214→   214→      pagination: { page, limit, total: totalGroups, totalPages },
   215→   215→      stats: {
   216→   216→        totalGroups,
   217→   217→        totalUsersAcrossGroups,
   218→   218→      },
   219→   219→    });
   220→   220→  } catch (error: unknown) {
   221→   221→    if (error instanceof AuthError) {
   222→   222→      return NextResponse.json({ success: false, error: error.message }, { status: error.statusCode });
   223→   223→    }
   224→   224→    console.error("[aaa/groups] GET error:", error);
   225→   225→    const msg = error instanceof Error ? error.message : "Internal server error";
   226→   226→    return NextResponse.json({ success: false, error: msg }, { status: 500 });
   227→   227→  }
   228→   228→}

   230→   230→// ─── POST /api/aaa/groups ────────────────────────────────────────────────────
   231→   231→
   232→   232→interface CreateGroupBody {
   233→   233→  groupname: string;
   234→   234→  description?: string;
   235→   235→  priority?: number;
   236→   236→  speedLimitDown?: number;
   237→   237→  speedLimitUp?: number;
   238→   238→  dataLimit?: number;
   239→   239→  sessionTimeout?: number;
   240→   240→  checkAttrs?: { attribute: string; op?: string; value: string }[];
   241→   241→  replyAttrs?: { attribute: string; op?: string; value: string }[];
   242→   242→}
   243→   243→
   244→   244→export async function POST(req: NextRequest) {
   245→   245→  try {
   246→   246→    const userId = await requireAuth(req);
   247→   247→    const body: CreateGroupBody = await req.json();
   248→   248→
   249→   249→    const { groupname, description, priority, speedLimitDown, speedLimitUp, dataLimit, sessionTimeout, checkAttrs, replyAttrs } = body;
   250→   250→
   251→   251→    // ── Validate groupname
   252→   252→    if (!groupname?.trim()) {
   253→   253→      return NextResponse.json({ success: false, error: "groupname is required" }, { status: 400 });
   254→   254→    }
   255→   255→
   256→   256→    const trimmedName = groupname.trim();
   257→   257→
   258→   258→    if (trimmedName.length > 64) {
   259→   259→      return NextResponse.json({ success: false, error: "groupname must not exceed 64 characters" }, { status: 400 });
   260→   260→    }
   261→   261→
   262→   262→    // ── Validate no duplicate
   263→   263→    const existing = await db.$queryRawUnsafe<{ exists: boolean }[]>(
   264→   264→      `SELECT EXISTS(
   265→   265→        SELECT 1 FROM radgroupcheck WHERE groupname = $1
   266→   266→        UNION
   267→   267→        SELECT 1 FROM radgroupreply WHERE groupname = $1
   268→   268→      )::text::boolean AS exists`,
   269→   269→      trimmedName
   270→   270→    );
   271→   271→
   272→   272→    if (existing[0]?.exists) {
   273→   273→      return NextResponse.json({ success: false, error: `Group '${trimmedName}' already exists in FreeRADIUS tables` }, { status: 409 });
   274→   274→    }
   275→   275→
   276→   276→    // Also check RadiusGroup Prisma model
   277→   277→    const existingPrisma = await db.radiusGroup.findUnique({ where: { name: trimmedName } });
   278→   278→    if (existingPrisma) {
   279→   279→      return NextResponse.json({ success: false, error: `Group '${trimmedName}' already exists in RadiusGroup model` }, { status: 409 });
   280→   280→    }
   281→   281→
   282→   282→    // ── Validate and insert check attributes
   283→   283→    if (checkAttrs && checkAttrs.length > 0) {
   284→   284→      for (const attr of checkAttrs) {
   285→   285→        if (!attr.attribute?.trim()) {
   286→   286→          return NextResponse.json({ success: false, error: "Each check attribute must have a non-empty 'attribute'" }, { status: 400 });
   287→   287→        }
   288→   288→        if (!attr.value?.trim()) {
   289→   289→          return NextResponse.json({ success: false, error: `Check attribute '${attr.attribute}' must have a non-empty 'value'` }, { status: 400 });
   290→   290→        }
   291→   291→        const op = attr.op || "==";
   292→   292→        if (!VALID_CHECK_OPS.includes(op)) {
   293→   293→          return NextResponse.json({ success: false, error: `Invalid operator '${op}' for check attribute '${attr.attribute}'. Valid: ${VALID_CHECK_OPS.join(", ")}` }, { status: 400 });
   294→   294→        }
   295→   295→        if (attr.attribute.length > 64 || attr.value.length > 253) {
   296→   296→          return NextResponse.json({ success: false, error: `Attribute '${attr.attribute}' exceeds field length limits (attribute: 64, value: 253)` }, { status: 400 });
   297→   297→        }
   298→   298→      }
   299→   299→
   300→   300→      // Batch insert check attributes
   301→   301→      for (const attr of checkAttrs) {
   302→   302→        const op = attr.op || "==";
   303→   303→        await db.$queryRawUnsafe(
   304→   304→          `INSERT INTO radgroupcheck (groupname, attribute, op, value) VALUES ($1, $2, $3, $4)`,
   305→   305→          trimmedName, attr.attribute.trim(), op, attr.value.trim()
   306→   306→        );
   307→   307→      }
   308→   308→    }
   309→   309→
   310→   310→    // ── Validate and insert reply attributes
   311→   311→    if (replyAttrs && replyAttrs.length > 0) {
   312→   312→      for (const attr of replyAttrs) {
   313→   313→        if (!attr.attribute?.trim()) {
   314→   314→          return NextResponse.json({ success: false, error: "Each reply attribute must have a non-empty 'attribute'" }, { status: 400 });
   315→   315→        }
   316→   316→        if (!attr.value?.trim()) {
   317→   317→          return NextResponse.json({ success: false, error: `Reply attribute '${attr.attribute}' must have a non-empty 'value'` }, { status: 400 });
   318→   318→        }
   319→   319→        const op = attr.op || "=";
   320→   320→        if (!VALID_REPLY_OPS.includes(op)) {
   321→   321→          return NextResponse.json({ success: false, error: `Invalid operator '${op}' for reply attribute '${attr.attribute}'. Valid: ${VALID_REPLY_OPS.join(", ")}` }, { status: 400 });
   322→   322→        }
   323→   323→        if (attr.attribute.length > 64 || attr.value.length > 253) {
   324→   324→          return NextResponse.json({ success: false, error: `Attribute '${attr.attribute}' exceeds field length limits (attribute: 64, value: 253)` }, { status: 400 });
   325→   325→        }
   326→   326→      }
   327→   327→
   328→   328→      // Batch insert reply attributes
   329→   329→      for (const attr of replyAttrs) {
   330→   330→        const op = attr.op || "=";
   331→   331→        await db.$queryRawUnsafe(
   332→   332→          `INSERT INTO radgroupreply (groupname, attribute, op, value) VALUES ($1, $2, $3, $4)`,
   333→   333→          trimmedName, attr.attribute.trim(), op, attr.value.trim()
   334→   334→        );
   335→   335→      }
   336→   336→    }
   337→   337→
   338→   338→    // ── Create/update RadiusGroup Prisma model for enrichment
   339→   339→    await db.radiusGroup.upsert({
   340→   340→      where: { name: trimmedName },
   341→   341→      create: {
   342→   342→        name: trimmedName,
   343→   343→        description: description || "",
   344→   344→        priority: priority || 0,
   345→   345→        speedLimitDown: speedLimitDown || 0,
   346→   346→        speedLimitUp: speedLimitUp || 0,
   347→   347→        dataLimit: dataLimit ?? null,
   348→   348→        sessionTimeout: sessionTimeout ?? null,
   349→   349→      },
   350→   350→      update: {
   351→   351→        description: description || "",
   352→   352→        priority: priority || 0,
   353→   353→        speedLimitDown: speedLimitDown || 0,
   354→   354→        speedLimitUp: speedLimitUp || 0,
   355→   355→        dataLimit: dataLimit ?? null,
   356→   356→        sessionTimeout: sessionTimeout ?? null,
   357→   357→      },
   358→   358→    });
   359→   359→
   360→   360→    // ── Fetch the created data back for response
   361→   361→    const [checks, replies] = await Promise.all([
   362→   362→      db.$queryRawUnsafe<Record<string, unknown>[]>(
   363→   363→        `SELECT id, groupname, attribute, op, value FROM radgroupcheck WHERE groupname = $1 ORDER BY attribute`,
   364→   364→        trimmedName
   365→   365→      ),
   366→   366→      db.$queryRawUnsafe<Record<string, unknown>[]>(
   367→   367→        `SELECT id, groupname, attribute, op, value FROM radgroupreply WHERE groupname = $1 ORDER BY attribute`,
   368→   368→        trimmedName
   369→   369→      ),
   370→   370→    ]);
   371→   371→
   372→   372→    const radiusGroup = await db.radiusGroup.findUnique({ where: { name: trimmedName } });
   373→   373→
   374→   374→    return NextResponse.json({
   375→   375→      success: true,
   376→   376→      group: {
   377→   377→        groupname: trimmedName,
   378→   378→        checkAttrs: serializeRows(checks),
   379→   379→        replyAttrs: serializeRows(replies),
   380→   380→        checkCount: checks.length,
   381→   381→        replyCount: replies.length,
   382→   382→        userCount: 0,
   383→   383→        enrichment: radiusGroup,
   384→   384→      },
   385→   385→    });
   386→   386→  } catch (error: unknown) {
   387→   387→    if (error instanceof AuthError) {
   388→   388→      return NextResponse.json({ success: false, error: error.message }, { status: error.statusCode });
   389→   389→    }
   390→   390→    console.error("[aaa/groups] POST error:", error);
   391→   391→    const msg = error instanceof Error ? error.message : "Internal server error";
   392→   392→    return NextResponse.json({ success: false, error: msg }, { status: 500 });
   393→   393→  }
   394→   394→}

   433→   433→  checkAttrs?: { attribute: string; op?: string; value: string }[];
   434→   434→  replyAttrs?: { attribute: string; op?: string; value: string }[];
   435→   435→}
   436→   436→
   437→   437→export async function PUT(req: NextRequest) {
   438→   438→  try {
   439→   439→    await requireAuth(req);
   440→   440→    const body: UpdateGroupAction = await req.json();
   441→   441→
   442→   442→    const { groupname, action } = body;
   443→   443→
   444→   444→    if (!groupname?.trim()) {
   445→   445→      return NextResponse.json({ success: false, error: "groupname is required" }, { status: 400 });
   446→   446→    }
   447→   447→
   448→   448→    if (!action) {
   449→   449→      return NextResponse.json({ success: false, error: "action is required. Valid: add-check, remove-check, add-reply, remove-reply, update-check, update-reply, rename, update-enrichment, replace-checks, replace-replies" }, { status: 400 });
   450→   450→    }
   451→   451→
   452→   452→    const trimmedName = groupname.trim();

   812→   812→        }, { status: 400 });
   813→   813→    }
   814→   814→  } catch (error: unknown) {

   820→   820→    return NextResponse.json({ success: false, error: msg }, { status: 500 });
   821→   821→  }
