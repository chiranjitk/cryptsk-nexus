    14→    14→    url = f"{BASE}/{path}"
    15→    15→    data = json.dumps(body).encode() if body else None
    16→    16→    r = urllib.request.Request(url, data=data, method=method)
    17→    17→    r.add_header("Content-Type", "application/json")
    18→    18→    if auth:
    19→    19→        r.add_header("Authorization", f"Bearer {TOKEN}")
    20→    20→    try:
    21→    21→        resp = urllib.request.urlopen(r, timeout=10)
    22→    22→        return resp.status, json.loads(resp.read())
    23→    23→    except urllib.error.HTTPError as e:
    24→    24→        try:
    25→    25→            return e.code, json.loads(e.read())
    26→    26→        except:
    27→    27→            return e.code, {}

    47→    47→            v = v[k] if k < len(v) else default
    48→    48→        else:
    49→    49→            return default
