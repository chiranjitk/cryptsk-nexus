   262→    // ── Validate no duplicate
   263→    const existing = await db.$queryRawUnsafe<{ exists: boolean }[]>(
   264→      `SELECT CAST(CAST(EXISTS(
   265→        SELECT 1 FROM radgroupcheck WHERE groupname = $1
   266→        UNION
   267→        SELECT 1 FROM radgroupreply WHERE groupname = $1
   268→      ) AS text) AS boolean) AS exists`,
   269→      trimmedName
   270→    );
   271→
   272→    if (existing[0]?.exists) {
   273→      return NextResponse.json({ success: false, error: `Group '${trimmedName}' already exists in FreeRADIUS tables` }, { status: 409 });
   274→    }
   275→
   276→    // Also check RadiusGroup Prisma model
   277→    const existingPrisma = await db.radiusGroup.findUnique({ where: { name: trimmedName } });
   278→    if (existingPrisma) {
   279→      return NextResponse.json({ success: false, error: `Group '${trimmedName}' already exists in RadiusGroup model` }, { status: 409 });
   280→    }
   281→
   282→    // ── Validate and insert check attributes
   283→    if (checkAttrs && checkAttrs.length > 0) {
   284→      for (const attr of checkAttrs) {
   285→        if (!attr.attribute?.trim()) {
   286→          return NextResponse.json({ success: false, error: "Each check attribute must have a non-empty 'attribute'" }, { status: 400 });
   287→        }
   288→        if (!attr.value?.trim()) {
   289→          return NextResponse.json({ success: false, error: `Check attribute '${attr.attribute}' must have a non-empty 'value'` }, { status: 400 });
   290→        }
   291→        const op = attr.op || "==";
   292→        if (!VALID_CHECK_OPS.includes(op)) {
   293→          return NextResponse.json({ success: false, error: `Invalid operator '${op}' for check attribute '${attr.attribute}'. Valid: ${VALID_CHECK_OPS.join(", ")}` }, { status: 400 });
   294→        }
   295→        if (attr.attribute.length > 64 || attr.value.length > 253) {
   296→          return NextResponse.json({ success: false, error: `Attribute '${attr.attribute}' exceeds field length limits (attribute: 64, value: 253)` }, { status: 400 });
   297→        }
   298→      }
   299→
   300→      // Batch insert check attributes
   301→      for (const attr of checkAttrs) {
   302→        const op = attr.op || "==";
   303→        await db.$queryRawUnsafe(
   304→          `INSERT INTO radgroupcheck (groupname, attribute, op, value) VALUES ($1, $2, $3, $4)`,
   305→          trimmedName, attr.attribute.trim(), op, attr.value.trim()
   306→        );
   307→      }
   308→    }
   309→
   310→    // ── Validate and insert reply attributes
   311→    if (replyAttrs && replyAttrs.length > 0) {
   312→      for (const attr of replyAttrs) {
   313→        if (!attr.attribute?.trim()) {
   314→          return NextResponse.json({ success: false, error: "Each reply attribute must have a non-empty 'attribute'" }, { status: 400 });
   315→        }
   316→        if (!attr.value?.trim()) {
   317→          return NextResponse.json({ success: false, error: `Reply attribute '${attr.attribute}' must have a non-empty 'value'` }, { status: 400 });
   318→        }
   319→        const op = attr.op || "=";
   320→        if (!VALID_REPLY_OPS.includes(op)) {
   321→          return NextResponse.json({ success: false, error: `Invalid operator '${op}' for reply attribute '${attr.attribute}'. Valid: ${VALID_REPLY_OPS.join(", ")}` }, { status: 400 });
   322→        }
   323→        if (attr.attribute.length > 64 || attr.value.length > 253) {
   324→          return NextResponse.json({ success: false, error: `Attribute '${attr.attribute}' exceeds field length limits (attribute: 64, value: 253)` }, { status: 400 });
   325→        }
   326→      }
   327→
   328→      // Batch insert reply attributes
   329→      for (const attr of replyAttrs) {
   330→        const op = attr.op || "=";
   331→        await db.$queryRawUnsafe(
   332→          `INSERT INTO radgroupreply (groupname, attribute, op, value) VALUES ($1, $2, $3, $4)`,
   333→          trimmedName, attr.attribute.trim(), op, attr.value.trim()
   334→        );
   335→      }
   336→    }
   337→
   338→    // ── Create/update RadiusGroup Prisma model for enrichment
   339→    await db.radiusGroup.upsert({
   340→      where: { name: trimmedName },
   341→      create: {
   342→        name: trimmedName,
   343→        description: description || "",
   344→        priority: priority || 0,
   345→        speedLimitDown: speedLimitDown || 0,
   346→        speedLimitUp: speedLimitUp || 0,
   347→        dataLimit: dataLimit ?? null,
   348→        sessionTimeout: sessionTimeout ?? null,
   349→      },
   350→      update: {
   351→        description: description || "",
   352→        priority: priority || 0,
   353→        speedLimitDown: speedLimitDown || 0,
   354→        speedLimitUp: speedLimitUp || 0,
   355→        dataLimit: dataLimit ?? null,
   356→        sessionTimeout: sessionTimeout ?? null,
   357→      },
   358→    });
   359→
   360→    // ── Fetch the created data back for response
   361→    const [checks, replies] = await Promise.all([
   362→      db.$queryRawUnsafe<Record<string, unknown>[]>(
   363→        `SELECT id, groupname, attribute, op, value FROM radgroupcheck WHERE groupname = $1 ORDER BY attribute`,
   364→        trimmedName
   365→      ),
   366→      db.$queryRawUnsafe<Record<string, unknown>[]>(
   367→        `SELECT id, groupname, attribute, op, value FROM radgroupreply WHERE groupname = $1 ORDER BY attribute`,
   368→        trimmedName
   369→      ),
   370→    ]);
   371→
   372→    const radiusGroup = await db.radiusGroup.findUnique({ where: { name: trimmedName } });
   373→
   374→    return NextResponse.json({
   375→      success: true,
   376→      RadiusGroup: {
   377→        groupname: trimmedName,
   378→        checkAttrs: serializeRows(checks),
   379→        replyAttrs: serializeRows(replies),
   380→        checkCount: checks.length,
   381→        replyCount: replies.length,
   382→        userCount: 0,
   383→        enrichment: radiusGroup,
   384→      },
   385→    }, { status: 201 });
   386→  } catch (error: unknown) {
   387→    if (error instanceof AuthError) {
   388→      return NextResponse.json({ success: false, error: error.message }, { status: error.statusCode });
   389→    }
   390→    console.error("[aaa/groups] POST error:", error);
   391→    const msg = error instanceof Error ? error.message : "Internal server error";
   392→    return NextResponse.json({ success: false, error: msg }, { status: 500 });
   393→  }

   690→        // Rename in radgroupreply

   700→        }

   710→        if (body.speedLimitDown !== undefined) updateData.speedLimitDown = Number(body.speedLimitDown);

   712→        if (body.dataLimit !== undefined) updateData.dataLimit = body.dataLimit ?? null;

   714→        if (body.framedIpv6Pool !== undefined) updateData.framedIpv6Pool = body.framedIpv6Pool;
   715→        if (body.delegatedIpv6PrefixPool !== undefined) updateData.delegatedIpv6PrefixPool = body.delegatedIpv6PrefixPool;

   718→        const enriched = await db.radiusGroup.upsert({
   719→          where: { name: trimmedName },
   720→          create: {
   721→            name: trimmedName,
   722→            description: (body.description as string) || "",
   723→            priority: Number(body.priority) || 0,
   724→            speedLimitDown: Number(body.speedLimitDown) || 0,
   725→            speedLimitUp: Number(body.speedLimitUp) || 0,
   726→            dataLimit: body.dataLimit ?? null,
   727→            sessionTimeout: body.sessionTimeout ?? null,
   728→            framedIpv6Pool: (body.framedIpv6Pool as string) || "",
   729→            delegatedIpv6PrefixPool: (body.delegatedIpv6PrefixPool as string) || "",

   738→      case "replace-checks": {

   747→          }
   748→          const op = attr.op || "==";
   749→          if (!VALID_CHECK_OPS.includes(op)) {
   750→            return NextResponse.json({ success: false, error: `Invalid check operator '${op}'` }, { status: 400 });
   751→          }
   752→        }
   753→
   754→        // Delete all existing checks
   755→        await db.$queryRawUnsafe(`DELETE FROM radgroupcheck WHERE groupname = $1`, trimmedName);
   756→
   757→        // Insert new ones
   758→        for (const attr of body.checkAttrs) {
   759→          const op = attr.op || "==";
   760→          await db.$queryRawUnsafe(
   761→            `INSERT INTO radgroupcheck (groupname, attribute, op, value) VALUES ($1, $2, $3, $4)`,
   762→            trimmedName, attr.attribute.trim(), op, attr.value.trim()
   763→          );
   764→        }
   765→
   766→        const newChecks = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   767→          `SELECT id, groupname, attribute, op, value FROM radgroupcheck WHERE groupname = $1 ORDER BY attribute`,
   768→          trimmedName
   769→        );
   770→
   771→        return NextResponse.json({ success: true, action: "replace-checks", checkAttrs: serializeRows(newChecks), count: newChecks.length });
   772→      }
   773→
   774→      // ── REPLACE ALL REPLY ATTRIBUTES ──────────────────────────────────
   775→      case "replace-replies": {
   776→        if (!Array.isArray(body.replyAttrs)) {
   777→          return NextResponse.json({ success: false, error: "replyAttrs array is required for replace-replies" }, { status: 400 });
   778→        }
   779→
   780→        for (const attr of body.replyAttrs) {
   781→          if (!attr.attribute?.trim() || (!attr.value?.trim() && attr.value !== "0")) {
   782→            return NextResponse.json({ success: false, error: "Each reply attribute must have non-empty attribute and value" }, { status: 400 });
   783→          }
   784→          const op = attr.op || "=";
   785→          if (!VALID_REPLY_OPS.includes(op)) {
   786→            return NextResponse.json({ success: false, error: `Invalid reply operator '${op}'` }, { status: 400 });
   787→          }
   788→        }
   789→
   790→        await db.$queryRawUnsafe(`DELETE FROM radgroupreply WHERE groupname = $1`, trimmedName);
   791→
   792→        for (const attr of body.replyAttrs) {
   793→          const op = attr.op || "=";
   794→          await db.$queryRawUnsafe(
   795→            `INSERT INTO radgroupreply (groupname, attribute, op, value) VALUES ($1, $2, $3, $4)`,
   796→            trimmedName, attr.attribute.trim(), op, attr.value.trim()
   797→          );
   798→        }
   799→
   800→        const newReplies = await db.$queryRawUnsafe<Record<string, unknown>[]>(
   801→          `SELECT id, groupname, attribute, op, value FROM radgroupreply WHERE groupname = $1 ORDER BY attribute`,
   802→          trimmedName
   803→        );
   804→
   805→        return NextResponse.json({ success: true, action: "replace-replies", replyAttrs: serializeRows(newReplies), count: newReplies.length });
   806→      }
   807→
   808→      default:
   809→        return NextResponse.json({
   810→          success: false,
   811→          error: `Unknown action '${action}'. Valid actions: add-check, remove-check, add-reply, remove-reply, update-check, update-reply, rename, update-enrichment, replace-checks, replace-replies`,
   812→        }, { status: 400 });
   813→    }
   814→  } catch (error: unknown) {
   815→    if (error instanceof AuthError) {
   816→      return NextResponse.json({ success: false, error: error.message }, { status: error.statusCode });
   817→    }
   818→    console.error("[aaa/groups] PUT error:", error);
   819→    const msg = error instanceof Error ? error.message : "Internal server error";
   820→    return NextResponse.json({ success: false, error: msg }, { status: 500 });
   821→  }
   822→}

   826→export async function DELETE(req: NextRequest) {
   827→  try {
   828→    await requireAuth(req);
   829→
   830→    const { searchParams } = new URL(req.url);
   831→    const groupname = searchParams.get("groupname")?.trim();
   832→
   833→    if (!groupname) {
   834→      return NextResponse.json({ success: false, error: "groupname query parameter is required" }, { status: 400 });
   835→    }
   836→
   837→    // Verify group exists
   838→    const exists = await db.$queryRawUnsafe<{ exists: boolean }[]>(
   839→      `SELECT CAST(CAST(EXISTS(
   840→        SELECT 1 FROM radgroupcheck WHERE groupname = $1
   841→        UNION
   842→        SELECT 1 FROM radgroupreply WHERE groupname = $1
   843→      ) AS text) AS boolean) AS exists`,
   844→      groupname
   845→    );
   846→
   847→    if (!exists[0]?.exists) {
   848→      return NextResponse.json({ success: false, error: `Group '${groupname}' not found` }, { status: 404 });
   849→    }
   850→
   851→    // Count users assigned to this group before deletion
   852→    const userCountResult = await db.$queryRawUnsafe<{ count: string }[]>(
   853→      `SELECT CAST(COUNT(*) AS text) AS count FROM radusergroup WHERE groupname = $1`,
   854→      groupname
   855→    );
   856→    const affectedUsers = Number(userCountResult[0]?.count || 0);
   857→
   858→    // Count check/reply attrs
   859→    const [checkCountResult, replyCountResult] = await Promise.all([
   860→      db.$queryRawUnsafe<{ count: string }[]>(
   861→        `SELECT CAST(COUNT(*) AS text) AS count FROM radgroupcheck WHERE groupname = $1`,
   862→        groupname
   863→      ),
   864→      db.$queryRawUnsafe<{ count: string }[]>(
   865→        `SELECT CAST(COUNT(*) AS text) AS count FROM radgroupreply WHERE groupname = $1`,
   866→        groupname
   867→      ),
   868→    ]);
   869→    const deletedChecks = Number(checkCountResult[0]?.count || 0);
   870→    const deletedReplies = Number(replyCountResult[0]?.count || 0);
   871→
   872→    // Execute all deletions in a logical sequence
   873→    // 1. Delete all radgroupcheck entries
   874→    await db.$queryRawUnsafe(`DELETE FROM radgroupcheck WHERE groupname = $1`, groupname);
   875→
   876→    // 2. Delete all radgroupreply entries
   877→    await db.$queryRawUnsafe(`DELETE FROM radgroupreply WHERE groupname = $1`, groupname);
   878→
   879→    // 3. Remove all radusergroup assignments for this group
   880→    await db.$queryRawUnsafe(`DELETE FROM radusergroup WHERE groupname = $1`, groupname);
   881→
   882→    // 4. Remove RadiusGroup Prisma model if exists
   883→    try {
   884→      await db.radiusGroup.deleteMany({ where: { name: groupname } });
   885→    } catch {
   886→      // RadiusGroup might not exist, that's fine
   887→    }
   888→
   889→    return NextResponse.json({
   890→      success: true,
   891→      deleted: {
   892→        groupname,
   893→        deletedChecks,
   894→        deletedReplies,
   895→        unassignedUsers: affectedUsers,
   896→      },
   897→    });
   898→  } catch (error: unknown) {
   899→    if (error instanceof AuthError) {
   900→      return NextResponse.json({ success: false, error: error.message }, { status: error.statusCode });
   901→    }
   902→    console.error("[aaa/groups] DELETE error:", error);
   903→    const msg = error instanceof Error ? error.message : "Internal server error";
   904→    return NextResponse.json({ success: false, error: msg }, { status: 500 });
   905→  }
   906→}
